"use client";

import {
  ArrowLeft,
  Copy,
  Download,
  Edit3,
  Eye,
  FolderOpen,
  Image as ImageIcon,
  Pause,
  Play,
  Plus,
  Save,
  Settings,
  Sparkles,
  Trash2,
  Upload,
  X,
} from "lucide-react";

import {
  ChangeEvent,
  useEffect,
  useRef,
  useState,
} from "react";

/* =========================================================
   TYPES
========================================================= */

type Quiz = {
  id: string;
  time: number;
  question: string;
  options: string[];
  correctIndex: number;
  points: number;
  feedbackCorrect: string;
  feedbackWrong: string;
};

type QuizDraft = {
  question: string;
  options: [string, string, string, string];
  correctIndex: number;
  points: number;
  feedbackCorrect: string;
  feedbackWrong: string;
};

type SavedProject = {
  version: number;
  title: string;
  videoName: string;
  duration: number;
  quizzes: Quiz[];
  savedAt: string;
};

/* =========================================================
   CONSTANT
========================================================= */

const STORAGE_KEY = "vlearning-studio-demo-project";

/* =========================================================
   HELPERS
========================================================= */

function createEmptyDraft(): QuizDraft {
  return {
    question: "",
    options: ["", "", "", ""],
    correctIndex: 0,
    points: 10,
    feedbackCorrect: "Chính xác! Bạn đã trả lời đúng.",
    feedbackWrong: "Chưa chính xác. Hãy xem lại nội dung.",
  };
}

function formatTime(seconds: number) {
  if (!Number.isFinite(seconds)) {
    return "0:00";
  }

  const minutes = Math.floor(seconds / 60);
  const secs = Math.floor(seconds % 60);

  return `${minutes}:${secs.toString().padStart(2, "0")}`;
}

/* =========================================================
   MAIN
========================================================= */

export default function EditorPage() {
  const videoRef = useRef<HTMLVideoElement>(null);
  const previousTimeRef = useRef(0);

  /* PROJECT */

  const [projectTitle] = useState(
    "Bài giảng tương tác đầu tiên"
  );

  const [saveStatus, setSaveStatus] =
    useState("Chưa lưu");

  /* VIDEO */

  const [videoUrl, setVideoUrl] =
    useState<string | null>(null);

  const [videoName, setVideoName] =
    useState("");

  const [duration, setDuration] =
    useState(10);

  const [currentTime, setCurrentTime] =
    useState(0);

  const [isPlaying, setIsPlaying] =
    useState(false);

  /* QUIZ */

  const [quizzes, setQuizzes] =
    useState<Quiz[]>([]);

  const [activeQuiz, setActiveQuiz] =
    useState<Quiz | null>(null);

  const [answeredIds, setAnsweredIds] =
    useState<string[]>([]);

  const [score, setScore] =
    useState(0);

  const [selectedAnswer, setSelectedAnswer] =
    useState<number | null>(null);

  const [quizResult, setQuizResult] =
    useState<"correct" | "wrong" | null>(null);

  /* QUESTION EDITOR */

  const [showQuestionEditor, setShowQuestionEditor] =
    useState(false);

  const [editingQuizId, setEditingQuizId] =
    useState<string | null>(null);

  const [questionTime, setQuestionTime] =
    useState(0);

  const [draft, setDraft] =
    useState<QuizDraft>(
      createEmptyDraft()
    );

  /* =========================================================
     LOAD PROJECT
  ========================================================= */

  useEffect(() => {
    try {
      const saved =
        localStorage.getItem(STORAGE_KEY);

      if (!saved) {
        return;
      }

      const data =
        JSON.parse(saved) as SavedProject;

      if (Array.isArray(data.quizzes)) {
        setQuizzes(data.quizzes);
      }

      if (typeof data.videoName === "string") {
        setVideoName(data.videoName);
      }

      if (typeof data.duration === "number") {
        setDuration(data.duration);
      }

      if (data.savedAt) {
        const date =
          new Date(data.savedAt);

        setSaveStatus(
          `Đã tải bản lưu lúc ${date.toLocaleTimeString(
            "vi-VN"
          )}`
        );
      }
    } catch (error) {
      console.error(error);

      setSaveStatus(
        "Không đọc được bản lưu"
      );
    }
  }, []);

  /* =========================================================
     VIDEO
  ========================================================= */

  function handleVideoUpload(
    event: ChangeEvent<HTMLInputElement>
  ) {
    const file =
      event.target.files?.[0];

    if (!file) {
      return;
    }

    setVideoUrl((oldUrl) => {
      if (oldUrl) {
        URL.revokeObjectURL(oldUrl);
      }

      return URL.createObjectURL(file);
    });

    setVideoName(file.name);

    setCurrentTime(0);
    setAnsweredIds([]);
    setScore(0);
    setActiveQuiz(null);
    setQuizResult(null);
    setSelectedAnswer(null);

    previousTimeRef.current = 0;

    setSaveStatus(
      "Có thay đổi chưa lưu"
    );
  }

  function handleLoadedMetadata() {
    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    setDuration(video.duration);
  }

  function togglePlay() {
    const video =
      videoRef.current;

    if (!video) {
      alert(
        "Hãy tải video bài giảng trước."
      );

      return;
    }

    if (activeQuiz) {
      return;
    }

    if (video.paused) {
      video.play();
    } else {
      video.pause();
    }
  }

  function handleSeek(
    event: ChangeEvent<HTMLInputElement>
  ) {
    const time =
      Number(event.target.value);

    const video =
      videoRef.current;

    setCurrentTime(time);

    previousTimeRef.current =
      time;

    if (video) {
      video.currentTime = time;
    }
  }

  function handleTimeUpdate() {
    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    const time =
      video.currentTime;

    setCurrentTime(time);

    if (!activeQuiz) {
      const previous =
        previousTimeRef.current;

      const quiz =
        quizzes.find(
          (item) =>
            !answeredIds.includes(
              item.id
            ) &&
            item.time >= previous &&
            item.time <= time
        );

      if (quiz) {
        video.pause();

        setIsPlaying(false);
        setActiveQuiz(quiz);
        setSelectedAnswer(null);
        setQuizResult(null);
      }
    }

    previousTimeRef.current =
      time;
  }

  /* =========================================================
     ADD QUESTION
  ========================================================= */

  function openQuestionEditor() {
    if (!videoUrl) {
      alert(
        "Bạn cần tải video trước khi thêm câu hỏi."
      );

      return;
    }

    videoRef.current?.pause();

    setEditingQuizId(null);

    setQuestionTime(
      currentTime
    );

    setDraft(
      createEmptyDraft()
    );

    setShowQuestionEditor(
      true
    );
  }

  /* =========================================================
     EDIT QUESTION
  ========================================================= */

  function editQuiz(
    quiz: Quiz
  ) {
    videoRef.current?.pause();

    setEditingQuizId(
      quiz.id
    );

    setQuestionTime(
      quiz.time
    );

    setDraft({
      question:
        quiz.question,

      options: [
        quiz.options[0] ?? "",
        quiz.options[1] ?? "",
        quiz.options[2] ?? "",
        quiz.options[3] ?? "",
      ],

      correctIndex:
        quiz.correctIndex,

      points:
        quiz.points,

      feedbackCorrect:
        quiz.feedbackCorrect,

      feedbackWrong:
        quiz.feedbackWrong,
    });

    setShowQuestionEditor(
      true
    );
  }

  /* =========================================================
     SAVE / UPDATE QUESTION
  ========================================================= */

  function saveQuestion() {
    if (!draft.question.trim()) {
      alert(
        "Hãy nhập nội dung câu hỏi."
      );

      return;
    }

    const hasEmptyOption =
      draft.options.some(
        (option) =>
          !option.trim()
      );

    if (hasEmptyOption) {
      alert(
        "Hãy nhập đầy đủ 4 đáp án A, B, C, D."
      );

      return;
    }

    /* UPDATE */

    if (editingQuizId) {
      setQuizzes((old) =>
        old
          .map((quiz) =>
            quiz.id === editingQuizId
              ? {
                  ...quiz,

                  time:
                    questionTime,

                  question:
                    draft.question.trim(),

                  options:
                    draft.options.map(
                      (item) =>
                        item.trim()
                    ),

                  correctIndex:
                    draft.correctIndex,

                  points:
                    draft.points,

                  feedbackCorrect:
                    draft.feedbackCorrect.trim(),

                  feedbackWrong:
                    draft.feedbackWrong.trim(),
                }
              : quiz
          )
          .sort(
            (a, b) =>
              a.time - b.time
          )
      );

      setEditingQuizId(null);

      setShowQuestionEditor(
        false
      );

      setSaveStatus(
        "Có thay đổi chưa lưu"
      );

      return;
    }

    /* CREATE */

    const newQuiz: Quiz = {
      id:
        crypto.randomUUID(),

      time:
        questionTime,

      question:
        draft.question.trim(),

      options:
        draft.options.map(
          (item) =>
            item.trim()
        ),

      correctIndex:
        draft.correctIndex,

      points:
        draft.points,

      feedbackCorrect:
        draft.feedbackCorrect.trim(),

      feedbackWrong:
        draft.feedbackWrong.trim(),
    };

    setQuizzes((old) =>
      [...old, newQuiz].sort(
        (a, b) =>
          a.time - b.time
      )
    );

    setShowQuestionEditor(
      false
    );

    setSaveStatus(
      "Có thay đổi chưa lưu"
    );
  }

  /* =========================================================
     DUPLICATE
  ========================================================= */

  function duplicateQuiz(
    quiz: Quiz
  ) {
    let newTime =
      quiz.time + 0.5;

    if (
      duration > 0 &&
      newTime > duration
    ) {
      newTime =
        quiz.time;
    }

    const copyQuiz: Quiz = {
      ...quiz,

      id:
        crypto.randomUUID(),

      time:
        newTime,

      question:
        `${quiz.question} - Bản sao`,

      options: [
        ...quiz.options,
      ],
    };

    setQuizzes((old) =>
      [
        ...old,
        copyQuiz,
      ].sort(
        (a, b) =>
          a.time - b.time
      )
    );

    setSaveStatus(
      "Có thay đổi chưa lưu"
    );
  }

  /* =========================================================
     DELETE
  ========================================================= */

  function removeQuiz(
    id: string
  ) {
    const confirmed =
      window.confirm(
        "Bạn có chắc muốn xóa câu hỏi này?"
      );

    if (!confirmed) {
      return;
    }

    setQuizzes((old) =>
      old.filter(
        (quiz) =>
          quiz.id !== id
      )
    );

    setAnsweredIds((old) =>
      old.filter(
        (quizId) =>
          quizId !== id
      )
    );

    setSaveStatus(
      "Có thay đổi chưa lưu"
    );
  }

  /* =========================================================
     QUIZ PLAYER
  ========================================================= */

  function answerQuiz(
    index: number
  ) {
    if (!activeQuiz) {
      return;
    }

    if (quizResult) {
      return;
    }

    setSelectedAnswer(
      index
    );

    const correct =
      index ===
      activeQuiz.correctIndex;

    if (correct) {
      setQuizResult(
        "correct"
      );

      setScore(
        (old) =>
          old +
          activeQuiz.points
      );
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  function continueVideo() {
    if (!activeQuiz) {
      return;
    }

    setAnsweredIds(
      (old) => [
        ...old,
        activeQuiz.id,
      ]
    );

    setActiveQuiz(
      null
    );

    setSelectedAnswer(
      null
    );

    setQuizResult(
      null
    );

    previousTimeRef.current =
      currentTime + 0.05;

    setTimeout(() => {
      videoRef.current?.play();
    }, 50);
  }

  /* =========================================================
     JUMP
  ========================================================= */

  function jumpToQuiz(
    quiz: Quiz
  ) {
    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    video.pause();

    video.currentTime =
      quiz.time;

    setCurrentTime(
      quiz.time
    );

    previousTimeRef.current =
      quiz.time;
  }

  /* =========================================================
     SAVE PROJECT
  ========================================================= */

  function saveProject() {
    const projectData: SavedProject = {
      version: 2,

      title:
        projectTitle,

      videoName,

      duration,

      quizzes,

      savedAt:
        new Date().toISOString(),
    };

    try {
      localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(
          projectData
        )
      );

      setSaveStatus(
        `Đã lưu lúc ${new Date().toLocaleTimeString(
          "vi-VN"
        )}`
      );
    } catch (error) {
      console.error(error);

      setSaveStatus(
        "Lưu thất bại"
      );
    }
  }

  /* =========================================================
     RENDER
  ========================================================= */

  return (
    <div className="flex h-screen flex-col overflow-hidden bg-[#061426] text-white">

      {/* TOP BAR */}

      <header className="flex h-16 shrink-0 items-center border-b border-blue-900 bg-[#102d57] px-4">

        <button className="mr-3 rounded-lg p-2 text-slate-300 hover:bg-blue-900">
          <ArrowLeft size={20} />
        </button>

        <button className="flex items-center gap-2 rounded-lg bg-blue-700 px-4 py-2 text-sm font-semibold">
          <FolderOpen size={17} />
          DỰ ÁN ĐÃ TẠO
        </button>

        <div className="ml-5">

          <div className="text-sm font-semibold">
            {projectTitle}
          </div>

          <div className="text-[11px] text-cyan-300">
            {saveStatus}
          </div>

        </div>

        <div className="flex-1" />

        <div className="mr-3 rounded-lg bg-blue-950 px-4 py-2">

          <span className="text-xs text-slate-400">
            Điểm
          </span>

          <strong className="ml-2 text-cyan-300">
            {score}
          </strong>

        </div>

        <div className="flex items-center gap-2">

          <button className="rounded-lg border border-cyan-500 bg-cyan-700/60 px-4 py-2 text-sm font-semibold">
            Thiết kế
          </button>

          <button
            onClick={
              saveProject
            }
            className="flex items-center gap-2 rounded-lg bg-emerald-600 px-4 py-2 text-sm font-bold hover:bg-emerald-500"
          >
            <Save size={17} />
            Lưu dự án
          </button>

          <button className="flex items-center gap-2 rounded-lg bg-blue-800 px-4 py-2 text-sm">
            <Eye size={16} />
            Xem trước
          </button>

          <button className="flex items-center gap-2 rounded-lg bg-gradient-to-r from-blue-500 to-purple-600 px-5 py-2 text-sm font-bold">
            <Download size={17} />
            Xuất file
          </button>

        </div>

      </header>

      {/* WORKSPACE */}

      <div className="flex min-h-0 flex-1">

        {/* LEFT */}

        <aside className="hidden w-[300px] shrink-0 border-r border-blue-900 bg-[#123565] p-4 md:block">

          <div className="mb-5 flex items-center justify-between">

            <div>

              <div className="font-bold">
                DANH SÁCH TRANG
              </div>

              <div className="text-xs text-slate-400">
                1 trang
              </div>

            </div>

            <button className="flex items-center gap-1 rounded-lg bg-blue-700 px-3 py-2 text-xs font-bold">
              <Plus size={14} />
              THÊM
            </button>

          </div>

          <div className="rounded-xl border-2 border-cyan-400 bg-blue-900/60 p-4">

            <div className="flex justify-between">

              <strong>
                1. Trang 1
              </strong>

              <span className="text-xs text-cyan-300">
                {quizzes.length} câu hỏi
              </span>

            </div>

            <div className="mt-3 inline-block rounded-md bg-[#07182c] px-3 py-1 text-xs">
              Trang tương tác
            </div>

          </div>

        </aside>

        {/* CENTER */}

        <main className="flex min-w-0 flex-1 flex-col">

          <section className="flex min-h-0 flex-1 items-center justify-center overflow-auto p-5">

            <div className="w-full max-w-5xl">

              <div className="mb-4 text-center">

                <div className="mb-3 inline-flex items-center gap-2 rounded-full border border-cyan-700 px-4 py-1 text-sm text-cyan-300">

                  <Sparkles size={15} />

                  Trang Video Tương Tác

                </div>

                {!videoUrl && (

                  <h1 className="text-3xl font-bold">
                    Tải lên video
                  </h1>

                )}

              </div>

              {/* VIDEO CANVAS */}

              <div className="relative aspect-video w-full overflow-hidden rounded-2xl border border-blue-500 bg-[#17396e] shadow-2xl">

                {!videoUrl ? (

                  <label className="flex h-full cursor-pointer flex-col items-center justify-center hover:bg-[#1b447f]">

                    <div className="mb-5 rounded-2xl bg-gradient-to-br from-blue-400 to-purple-600 p-5">
                      <Upload size={38} />
                    </div>

                    <div className="text-xl font-bold">
                      Tải lên video
                    </div>

                    <div className="mt-2 text-sm text-slate-400">
                      MP4 / WEBM / MOV
                    </div>

                    <input
                      type="file"
                      accept="video/*"
                      className="hidden"
                      onChange={
                        handleVideoUpload
                      }
                    />

                  </label>

                ) : (

                  <>

                    <video
                      ref={videoRef}
                      src={videoUrl}
                      className="h-full w-full bg-black object-contain"
                      onLoadedMetadata={
                        handleLoadedMetadata
                      }
                      onTimeUpdate={
                        handleTimeUpdate
                      }
                      onPlay={() =>
                        setIsPlaying(true)
                      }
                      onPause={() =>
                        setIsPlaying(false)
                      }
                      onEnded={() =>
                        setIsPlaying(false)
                      }
                    />

                    {/* QUIZ OVERLAY */}

                    {activeQuiz && (

                      <div className="absolute inset-0 flex items-center justify-center bg-black/80 p-8">

                        <div className="w-full max-w-2xl rounded-2xl border border-cyan-500 bg-[#102d57] p-7 shadow-2xl">

                          <div className="text-sm font-bold uppercase tracking-wider text-cyan-300">
                            Câu hỏi tương tác
                          </div>

                          <h2 className="mt-3 text-2xl font-bold">
                            {activeQuiz.question}
                          </h2>

                          <div className="mt-6 grid grid-cols-1 gap-3 md:grid-cols-2">

                            {activeQuiz.options.map(
                              (option, index) => {

                                let style =
                                  "border-blue-700 bg-blue-950 hover:bg-blue-900";

                                if (
                                  quizResult &&
                                  index ===
                                    activeQuiz.correctIndex
                                ) {
                                  style =
                                    "border-green-400 bg-green-700";
                                } else if (
                                  quizResult ===
                                    "wrong" &&
                                  index ===
                                    selectedAnswer
                                ) {
                                  style =
                                    "border-red-400 bg-red-700";
                                }

                                return (

                                  <button
                                    key={index}
                                    disabled={
                                      quizResult !== null
                                    }
                                    onClick={() =>
                                      answerQuiz(index)
                                    }
                                    className={`rounded-xl border p-4 text-left font-semibold ${style}`}
                                  >

                                    <span className="mr-2 text-cyan-300">
                                      {String.fromCharCode(
                                        65 + index
                                      )}
                                      .
                                    </span>

                                    {option}

                                  </button>

                                );
                              }
                            )}

                          </div>

                          {quizResult && (

                            <div
                              className={`mt-6 rounded-xl p-4 ${
                                quizResult ===
                                "correct"
                                  ? "bg-green-950 text-green-300"
                                  : "bg-red-950 text-red-300"
                              }`}
                            >

                              <strong>

                                {quizResult ===
                                "correct"
                                  ? "✓ Chính xác"
                                  : "✕ Chưa chính xác"}

                              </strong>

                              <p className="mt-2 text-sm">

                                {quizResult ===
                                "correct"
                                  ? activeQuiz.feedbackCorrect
                                  : activeQuiz.feedbackWrong}

                              </p>

                              <button
                                onClick={
                                  continueVideo
                                }
                                className="mt-4 rounded-lg bg-cyan-600 px-5 py-2 font-bold text-white"
                              >
                                Tiếp tục video
                              </button>

                            </div>

                          )}

                        </div>

                      </div>

                    )}

                  </>

                )}

              </div>

              {videoUrl && (

                <div className="mt-3 flex justify-between text-xs text-slate-400">

                  <span>
                    {videoName}
                  </span>

                  <label className="cursor-pointer text-cyan-300">

                    Đổi video

                    <input
                      type="file"
                      accept="video/*"
                      className="hidden"
                      onChange={
                        handleVideoUpload
                      }
                    />

                  </label>

                </div>

              )}

            </div>

          </section>

          {/* PLAYER CONTROLS */}

          <div className="flex h-16 shrink-0 items-center gap-4 border-t border-blue-900 bg-[#07182c] px-5">

            <button
              onClick={
                togglePlay
              }
              className="flex h-10 w-10 items-center justify-center rounded-full bg-cyan-500"
            >

              {isPlaying ? (
                <Pause size={18} />
              ) : (
                <Play size={18} />
              )}

            </button>

            <span className="w-28 font-mono text-sm">

              {formatTime(
                currentTime
              )}

              {" / "}

              {formatTime(
                duration
              )}

            </span>

            <input
              type="range"
              min={0}
              max={duration || 10}
              step={0.01}
              value={currentTime}
              onChange={handleSeek}
              className="min-w-0 flex-1 cursor-pointer"
            />

            <button
              onClick={
                openQuestionEditor
              }
              className="flex items-center gap-2 rounded-lg bg-cyan-600 px-4 py-2 text-sm font-bold"
            >
              <Plus size={16} />

              Thêm câu hỏi
            </button>

          </div>

          {/* TIMELINE */}

          <div className="h-40 shrink-0 border-t border-blue-900 bg-[#04101f] p-4">

            <div className="mb-3 flex justify-between">

              <strong>
                Timeline
              </strong>

              <span className="text-xs text-slate-400">
                {quizzes.length} tương tác
              </span>

            </div>

            <div className="relative h-20 rounded-lg border border-blue-950 bg-[#07182c]">

              <div className="absolute left-0 right-0 top-1/2 h-px bg-blue-800" />

              {quizzes.map((quiz) => {

                const left =
                  duration > 0
                    ? (quiz.time /
                        duration) *
                      100
                    : 0;

                return (

                  <button
                    key={quiz.id}
                    style={{
                      left: `${left}%`,
                    }}
                    onClick={() =>
                      jumpToQuiz(quiz)
                    }
                    title={`${quiz.question} - ${formatTime(
                      quiz.time
                    )}`}
                    className="absolute top-5 -translate-x-1/2"
                  >

                    <div className="rounded bg-purple-600 px-2 py-1 text-xs font-bold">
                      Q
                    </div>

                  </button>

                );
              })}

              {/* PLAYHEAD */}

              <div
                className="pointer-events-none absolute bottom-0 top-0 w-[2px] bg-red-500"
                style={{
                  left: `${
                    duration > 0
                      ? (currentTime /
                          duration) *
                        100
                      : 0
                  }%`,
                }}
              />

            </div>

          </div>

        </main>

        {/* RIGHT PANEL */}

        <aside className="hidden w-[330px] shrink-0 overflow-y-auto border-l border-blue-900 bg-[#123565] lg:block">

          <div className="border-b border-blue-900 p-5">

            <div className="flex items-center gap-2 font-bold">
              <Settings size={17} />

              ĐIỀU KIỆN HOÀN THÀNH
            </div>

            <p className="mt-2 text-xs text-slate-400">
              Người học xem video và hoàn thành các câu hỏi tương tác.
            </p>

          </div>

          <div className="border-b border-blue-900 p-5">

            <div className="mb-4 flex items-center gap-2 font-bold">
              <ImageIcon size={17} />

              PHƯƠNG TIỆN
            </div>

            <label className="flex cursor-pointer items-center justify-center gap-2 rounded-lg bg-blue-700 py-3 font-semibold">

              <Upload size={16} />

              {videoUrl
                ? "Thay video"
                : "Tải video lên"}

              <input
                type="file"
                accept="video/*"
                className="hidden"
                onChange={
                  handleVideoUpload
                }
              />

            </label>

          </div>

          {/* QUESTION LIST */}

          <div className="p-5">

            <div className="mb-4 flex items-center justify-between">

              <strong className="text-sm">
                CÂU HỎI
              </strong>

              <span className="rounded-full bg-blue-800 px-2 py-1 text-xs text-cyan-300">
                {quizzes.length}
              </span>

            </div>

            {quizzes.length === 0 ? (

              <button
                onClick={
                  openQuestionEditor
                }
                className="w-full rounded-xl border border-dashed border-blue-600 p-8 text-sm text-slate-400"
              >
                + Thêm câu hỏi
              </button>

            ) : (

              <div className="space-y-3">

                {quizzes.map(
                  (quiz, index) => (

                    <div
                      key={quiz.id}
                      className="rounded-xl border border-blue-800 bg-blue-950/50 p-3"
                    >

                      {/* QUESTION INFORMATION */}

                      <button
                        onClick={() =>
                          jumpToQuiz(quiz)
                        }
                        className="w-full text-left"
                      >

                        <div className="flex justify-between">

                          <strong className="text-sm">
                            Câu {index + 1}
                          </strong>

                          <span className="text-xs font-bold text-cyan-300">
                            {formatTime(
                              quiz.time
                            )}
                          </span>

                        </div>

                        <p className="mt-2 line-clamp-2 text-xs text-slate-300">
                          {quiz.question}
                        </p>

                        <div className="mt-2 text-xs text-purple-300">
                          {quiz.points} điểm
                        </div>

                      </button>

                      {/* ACTIONS */}

                      <div className="mt-3 grid grid-cols-3 gap-2 border-t border-blue-900 pt-3">

                        <button
                          onClick={() =>
                            editQuiz(quiz)
                          }
                          className="flex items-center justify-center gap-1 rounded-lg bg-cyan-900/60 px-2 py-2 text-xs font-semibold text-cyan-300 hover:bg-cyan-800"
                        >
                          <Edit3 size={14} />
                          Sửa
                        </button>

                        <button
                          onClick={() =>
                            duplicateQuiz(quiz)
                          }
                          className="flex items-center justify-center gap-1 rounded-lg bg-purple-900/60 px-2 py-2 text-xs font-semibold text-purple-300 hover:bg-purple-800"
                        >
                          <Copy size={14} />
                          Nhân bản
                        </button>

                        <button
                          onClick={() =>
                            removeQuiz(quiz.id)
                          }
                          className="flex items-center justify-center gap-1 rounded-lg bg-red-950/60 px-2 py-2 text-xs font-semibold text-red-300 hover:bg-red-900"
                        >
                          <Trash2 size={14} />
                          Xóa
                        </button>

                      </div>

                    </div>

                  )
                )}

              </div>

            )}

          </div>

        </aside>

      </div>

      {/* =====================================================
          QUESTION EDITOR
      ===================================================== */}

      {showQuestionEditor && (

        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 p-4">

          <div className="flex max-h-[92vh] w-full max-w-3xl flex-col overflow-hidden rounded-2xl border border-blue-600 bg-[#15376b] shadow-2xl">

            {/* HEADER */}

            <div className="flex shrink-0 items-center justify-between border-b border-blue-700 px-6 py-5">

              <div>

                <h2 className="text-2xl font-bold">

                  {editingQuizId
                    ? "Chỉnh sửa câu hỏi"
                    : "Thêm câu hỏi trắc nghiệm"}

                </h2>

                <p className="mt-1 text-sm text-cyan-300">

                  Xuất hiện tại{" "}

                  <strong>
                    {formatTime(
                      questionTime
                    )}
                  </strong>

                </p>

              </div>

              <button
                onClick={() => {
                  setShowQuestionEditor(
                    false
                  );

                  setEditingQuizId(
                    null
                  );
                }}
                className="rounded-lg p-2 hover:bg-blue-900"
              >
                <X size={24} />
              </button>

            </div>

            {/* BODY */}

            <div className="flex-1 space-y-6 overflow-y-auto px-6 py-6">

              {/* QUESTION */}

              <div>

                <label className="mb-2 block font-bold">
                  Nội dung câu hỏi
                </label>

                <textarea
                  value={draft.question}
                  onChange={(e) =>
                    setDraft({
                      ...draft,

                      question:
                        e.target.value,
                    })
                  }
                  className="min-h-28 w-full rounded-xl border border-blue-500 bg-[#07182c] p-4 text-lg outline-none focus:border-cyan-400"
                  placeholder="Nhập nội dung câu hỏi..."
                />

              </div>

              {/* ANSWERS */}

              <div>

                <label className="mb-4 block font-bold">
                  Đáp án
                </label>

                <div className="space-y-4">

                  {draft.options.map(
                    (option, index) => (

                      <div
                        key={index}
                        className="flex items-center gap-4"
                      >

                        <input
                          type="radio"
                          name="correct-answer"
                          checked={
                            draft.correctIndex ===
                            index
                          }
                          onChange={() =>
                            setDraft({
                              ...draft,

                              correctIndex:
                                index,
                            })
                          }
                          className="h-6 w-6 cursor-pointer accent-cyan-500"
                        />

                        <div className="w-7 font-bold text-cyan-300">
                          {String.fromCharCode(
                            65 + index
                          )}
                        </div>

                        <input
                          value={option}
                          onChange={(e) => {

                            const options = [
                              ...draft.options,
                            ] as [
                              string,
                              string,
                              string,
                              string
                            ];

                            options[index] =
                              e.target.value;

                            setDraft({
                              ...draft,
                              options,
                            });
                          }}
                          className="min-w-0 flex-1 rounded-xl border border-blue-500 bg-[#07182c] px-4 py-4 text-lg outline-none focus:border-cyan-400"
                          placeholder={`Đáp án ${String.fromCharCode(
                            65 + index
                          )}`}
                        />

                      </div>

                    )
                  )}

                </div>

                <p className="mt-3 text-sm text-slate-300">
                  Nút tròn xác định đáp án đúng.
                </p>

              </div>

              {/* QUESTION TIME */}

              <div>

                <label className="mb-2 block font-bold">
                  Thời điểm xuất hiện
                </label>

                <div className="flex items-center gap-3">

                  <input
                    type="number"
                    min={0}
                    max={
                      duration || 999999
                    }
                    step={0.1}
                    value={questionTime}
                    onChange={(e) => {

                      const value =
                        Number(
                          e.target.value
                        );

                      setQuestionTime(
                        Math.max(
                          0,
                          Math.min(
                            duration ||
                              value,
                            value
                          )
                        )
                      );
                    }}
                    className="w-40 rounded-xl border border-blue-500 bg-[#07182c] px-4 py-3 outline-none focus:border-cyan-400"
                  />

                  <span className="text-sm text-slate-300">
                    giây
                  </span>

                </div>

              </div>

              {/* POINTS */}

              <div>

                <label className="mb-2 block font-bold">
                  Điểm
                </label>

                <input
                  type="number"
                  min={1}
                  max={1000}
                  value={draft.points}
                  onChange={(e) =>
                    setDraft({
                      ...draft,

                      points:
                        Math.max(
                          1,
                          Number(
                            e.target.value
                          ) || 1
                        ),
                    })
                  }
                  className="w-40 rounded-xl border border-blue-500 bg-[#07182c] px-4 py-3 outline-none focus:border-cyan-400"
                />

              </div>

              {/* FEEDBACK */}

              <div className="grid gap-5 md:grid-cols-2">

                <div>

                  <label className="mb-2 block font-bold text-green-300">
                    Feedback khi đúng
                  </label>

                  <textarea
                    value={
                      draft.feedbackCorrect
                    }
                    onChange={(e) =>
                      setDraft({
                        ...draft,

                        feedbackCorrect:
                          e.target.value,
                      })
                    }
                    className="min-h-28 w-full rounded-xl border border-green-800 bg-[#07182c] p-4 outline-none"
                  />

                </div>

                <div>

                  <label className="mb-2 block font-bold text-red-300">
                    Feedback khi sai
                  </label>

                  <textarea
                    value={
                      draft.feedbackWrong
                    }
                    onChange={(e) =>
                      setDraft({
                        ...draft,

                        feedbackWrong:
                          e.target.value,
                      })
                    }
                    className="min-h-28 w-full rounded-xl border border-red-900 bg-[#07182c] p-4 outline-none"
                  />

                </div>

              </div>

            </div>

            {/* FOOTER ALWAYS VISIBLE */}

            <div className="flex shrink-0 items-center justify-between border-t border-blue-700 bg-[#0d2850] px-6 py-4">

              <div className="text-sm text-slate-300">

                {editingQuizId
                  ? "Đang chỉnh sửa câu hỏi"
                  : "Câu hỏi mới"}

                {" • "}

                <span className="font-bold text-cyan-300">
                  {formatTime(
                    questionTime
                  )}
                </span>

              </div>

              <div className="flex gap-3">

                <button
                  type="button"
                  onClick={() => {
                    setShowQuestionEditor(
                      false
                    );

                    setEditingQuizId(
                      null
                    );
                  }}
                  className="rounded-xl border border-blue-600 bg-blue-950 px-6 py-3 font-semibold hover:bg-blue-900"
                >
                  Hủy
                </button>

                <button
                  type="button"
                  onClick={
                    saveQuestion
                  }
                  className="flex items-center gap-2 rounded-xl bg-gradient-to-r from-cyan-500 to-purple-600 px-7 py-3 font-bold shadow-lg"
                >
                  <Save size={19} />

                  {editingQuizId
                    ? "Cập nhật câu hỏi"
                    : "Lưu câu hỏi"}

                </button>

              </div>

            </div>

          </div>

        </div>

      )}

    </div>
  );
}
