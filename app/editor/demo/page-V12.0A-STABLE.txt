"use client";

import {
  useEffect,
  useRef,
  useState,
} from "react";

import {
  EditorTopbar,
} from "@/components/editor/EditorTopbar";

import {
  PageNavigator,
} from "@/components/editor/PageNavigator";

import {
  VideoCanvas,
} from "@/components/editor/VideoCanvas";

import {
  PlayerControls,
} from "@/components/editor/PlayerControls";

import {
  Timeline,
} from "@/components/editor/Timeline";

import {
  QuestionPanel,
} from "@/components/editor/QuestionPanel";

import {
  QuestionEditor,
} from "@/components/editor/QuestionEditor";

import {
  useEditorStore,
} from "@/stores/editorStore";

import {
  createEmptyDraft,
} from "@/types/lesson";

import type {
  Quiz,
  QuizDraft,
} from "@/types/lesson";

export default function EditorPage() {
  /* =======================================================
     STORE
  ======================================================= */

  /*
   * Lấy toàn bộ store.
   *
   * Cách này đơn giản và chắc chắn hơn trong giai đoạn
   * chúng ta đang kiểm tra chức năng nhiều trang.
   */
  const editor =
    useEditorStore();

  const {
    project,
    saveStatus,

    hydrate,
    saveProject,

    selectPage,
    addPage,
    renamePage,
    duplicatePage,
    deletePage,

    setPageVideo,
    setPageDuration,

    addQuiz,
    updateQuiz,
    duplicateQuiz,
    deleteQuiz,
  } = editor;

  /* =======================================================
     VIDEO
  ======================================================= */

  const videoRef =
    useRef<HTMLVideoElement>(
      null
    );

  const previousTimeRef =
    useRef(0);

  /* =======================================================
     PLAYER
  ======================================================= */

  const [
    currentTime,
    setCurrentTime,
  ] =
    useState(0);

  const [
    isPlaying,
    setIsPlaying,
  ] =
    useState(false);

  /* =======================================================
     QUIZ PLAYER
  ======================================================= */

  const [
    activeQuiz,
    setActiveQuiz,
  ] =
    useState<Quiz | null>(
      null
    );

  const [
    answeredIds,
    setAnsweredIds,
  ] =
    useState<string[]>([]);

  const [
    score,
    setScore,
  ] =
    useState(0);

  const [
    selectedAnswer,
    setSelectedAnswer,
  ] =
    useState<number | null>(
      null
    );

  const [
    quizResult,
    setQuizResult,
  ] =
    useState<
      "correct" |
      "wrong" |
      null
    >(null);

  /* =======================================================
     QUESTION EDITOR
  ======================================================= */

  const [
    questionEditorOpen,
    setQuestionEditorOpen,
  ] =
    useState(false);

  const [
    editingQuizId,
    setEditingQuizId,
  ] =
    useState<string | null>(
      null
    );

  const [
    questionTime,
    setQuestionTime,
  ] =
    useState(0);

  const [
    draft,
    setDraft,
  ] =
    useState<QuizDraft>(
      createEmptyDraft()
    );

  /* =======================================================
     LOAD PROJECT
  ======================================================= */

  useEffect(() => {
    hydrate();
  }, [hydrate]);

  /* =======================================================
     CURRENT PAGE
  ======================================================= */

  const page =
    project.pages.find(
      (item) =>
        item.id ===
        project.selectedPageId
    ) ??
    project.pages[0];

  /* =======================================================
     RESET WHEN PAGE CHANGES
  ======================================================= */

  useEffect(() => {
    videoRef.current?.pause();

    setCurrentTime(0);

    setIsPlaying(false);

    setActiveQuiz(null);

    setAnsweredIds([]);

    setSelectedAnswer(
      null
    );

    setQuizResult(null);

    setQuestionEditorOpen(
      false
    );

    setEditingQuizId(
      null
    );

    previousTimeRef.current =
      0;
  }, [
    project.selectedPageId,
  ]);

  /* =======================================================
     ADD PAGE
  ======================================================= */

  function handleAddPage() {
    console.log(
      "Bắt đầu thêm trang..."
    );

    console.log(
      "Số trang trước:",
      useEditorStore
        .getState()
        .project
        .pages.length
    );

    addPage();

    console.log(
      "Số trang sau:",
      useEditorStore
        .getState()
        .project
        .pages.length
    );
  }

  /* =======================================================
     SELECT PAGE
  ======================================================= */

  function handleSelectPage(
    pageId: string
  ) {
    videoRef.current?.pause();

    selectPage(
      pageId
    );
  }

  /* =======================================================
     DELETE PAGE
  ======================================================= */

  function handleDeletePage(
    pageId: string
  ) {
    deletePage(
      pageId
    );
  }

  /* =======================================================
     VIDEO UPLOAD
  ======================================================= */

  function handleVideoUpload(
    file: File
  ) {
    if (!page) {
      return;
    }

    const oldUrl =
      page.videoUrl;

    if (
      oldUrl &&
      oldUrl.startsWith(
        "blob:"
      )
    ) {
      URL.revokeObjectURL(
        oldUrl
      );
    }

    const url =
      URL.createObjectURL(
        file
      );

    setPageVideo(
      page.id,
      url,
      file.name
    );

    setCurrentTime(0);

    setIsPlaying(false);

    setAnsweredIds([]);

    setActiveQuiz(null);

    setSelectedAnswer(
      null
    );

    setQuizResult(null);

    previousTimeRef.current =
      0;
  }

  /* =======================================================
     VIDEO METADATA
  ======================================================= */

  function handleLoadedMetadata() {
    if (!page) {
      return;
    }

    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    setPageDuration(
      page.id,
      video.duration
    );
  }

  /* =======================================================
     PLAY / PAUSE
  ======================================================= */

  function togglePlay() {
    const video =
      videoRef.current;

    if (!video) {
      alert(
        "Trang này chưa có video."
      );

      return;
    }

    if (activeQuiz) {
      return;
    }

    if (video.paused) {
      void video.play();
    } else {
      video.pause();
    }
  }

  /* =======================================================
     VIDEO TIME UPDATE
  ======================================================= */

  function handleTimeUpdate() {
    if (!page) {
      return;
    }

    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    const time =
      video.currentTime;

    setCurrentTime(
      time
    );

    if (!activeQuiz) {
      const previous =
        previousTimeRef.current;

      const quiz =
        page.quizzes.find(
          (item) =>
            !answeredIds.includes(
              item.id
            ) &&
            item.time >=
              previous &&
            item.time <=
              time
        );

      if (quiz) {
        video.pause();

        setIsPlaying(
          false
        );

        setActiveQuiz(
          quiz
        );

        setSelectedAnswer(
          null
        );

        setQuizResult(
          null
        );
      }
    }

    previousTimeRef.current =
      time;
  }

  /* =======================================================
     SEEK
  ======================================================= */

  function handleSeek(
    time: number
  ) {
    const video =
      videoRef.current;

    if (
      time <
      currentTime
    ) {
      /*
       * Tua ngược:
       * cho phép câu hỏi xuất hiện lại.
       */
      setAnsweredIds(
        []
      );

      setActiveQuiz(
        null
      );

      setSelectedAnswer(
        null
      );

      setQuizResult(
        null
      );
    }

    setCurrentTime(
      time
    );

    previousTimeRef.current =
      time;

    if (video) {
      video.currentTime =
        time;
    }
  }

  /* =======================================================
     ADD QUESTION
  ======================================================= */

  function openAddQuestion() {
    if (!page) {
      return;
    }

    if (
      !page.videoUrl
    ) {
      alert(
        "Hãy tải video cho trang này trước khi thêm câu hỏi."
      );

      return;
    }

    videoRef.current?.pause();

    setEditingQuizId(
      null
    );

    setQuestionTime(
      currentTime
    );

    setDraft(
      createEmptyDraft()
    );

    setQuestionEditorOpen(
      true
    );
  }

  /* =======================================================
     EDIT QUESTION
  ======================================================= */

  function openEditQuestion(
    quiz: Quiz
  ) {
    videoRef.current?.pause();

    setEditingQuizId(
      quiz.id
    );

    setQuestionTime(
      quiz.time
    );

    const quizType =
      quiz.type ===
      "true-false"
        ? "true-false"
        : "multiple-choice";

    setDraft({
      type:
        quizType,

      question:
        quiz.question,

      options:
        quizType ===
        "true-false"
          ? [
              "Đúng",
              "Sai",
              "",
              "",
            ]
          : [
              quiz.options[0] ??
                "",

              quiz.options[1] ??
                "",

              quiz.options[2] ??
                "",

              quiz.options[3] ??
                "",
            ],

      correctIndex:
        quizType ===
          "true-false"
          ? (
              quiz.correctIndex ===
              1
                ? 1
                : 0
            )
          : quiz.correctIndex,

      points:
        quiz.points,

      feedbackCorrect:
        quiz.feedbackCorrect,

      feedbackWrong:
        quiz.feedbackWrong,
    });

    setQuestionEditorOpen(
      true
    );
  }

  /* =======================================================
     SAVE QUESTION
  ======================================================= */

  function saveQuestion() {
    if (!page) {
      return;
    }

    if (
      !draft.question.trim()
    ) {
      alert(
        "Hãy nhập nội dung câu hỏi."
      );

      return;
    }

    if (
      draft.type ===
      "multiple-choice"
    ) {
      const hasEmptyAnswer =
        draft.options.some(
          (item) =>
            !item.trim()
        );

      if (hasEmptyAnswer) {
        alert(
          "Hãy nhập đủ 4 đáp án A, B, C, D."
        );

        return;
      }
    }

    const quiz: Quiz = {
      id:
        editingQuizId ??
        `quiz-${Date.now()}-${Math.random()
          .toString(36)
          .slice(
            2,
            8
          )}`,

      type:
        draft.type,

      time:
        questionTime,

      question:
        draft.question.trim(),

      options:
        draft.type ===
        "true-false"
          ? [
              "Đúng",
              "Sai",
            ]
          : draft.options.map(
              (item) =>
                item.trim()
            ),

      correctIndex:
        draft.type ===
        "true-false"
          ? (
              draft.correctIndex ===
              1
                ? 1
                : 0
            )
          : draft.correctIndex,

      points:
        draft.points,

      feedbackCorrect:
        draft.feedbackCorrect.trim(),

      feedbackWrong:
        draft.feedbackWrong.trim(),
    };

    if (
      editingQuizId
    ) {
      updateQuiz(
        page.id,
        editingQuizId,
        quiz
      );
    } else {
      addQuiz(
        page.id,
        quiz
      );
    }

    setQuestionEditorOpen(
      false
    );

    setEditingQuizId(
      null
    );
  }

  /* =======================================================
     ANSWER QUESTION
  ======================================================= */

  function answerQuestion(
    index: number
  ) {
    if (
      !activeQuiz ||
      quizResult
    ) {
      return;
    }

    setSelectedAnswer(
      index
    );

    const correct =
      index ===
      activeQuiz.correctIndex;

    if (correct) {
      setQuizResult(
        "correct"
      );

      setScore(
        (oldScore) =>
          oldScore +
          activeQuiz.points
      );
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  /* =======================================================
     CONTINUE VIDEO
  ======================================================= */

  function continueVideo() {
    if (!activeQuiz) {
      return;
    }

    setAnsweredIds(
      (old) => [
        ...old,
        activeQuiz.id,
      ]
    );

    setActiveQuiz(
      null
    );

    setSelectedAnswer(
      null
    );

    setQuizResult(
      null
    );

    previousTimeRef.current =
      currentTime +
      0.05;

    setTimeout(() => {
      void videoRef.current?.play();
    }, 50);
  }

  /* =======================================================
     JUMP TO QUIZ
  ======================================================= */

  function jumpToQuiz(
    quiz: Quiz
  ) {
    const video =
      videoRef.current;

    if (!video) {
      alert(
        "Trang này chưa có video."
      );

      return;
    }

    video.pause();

    setAnsweredIds(
      (old) =>
        old.filter(
          (id) =>
            id !==
            quiz.id
        )
    );

    setActiveQuiz(
      null
    );

    setSelectedAnswer(
      null
    );

    setQuizResult(
      null
    );

    video.currentTime =
      quiz.time;

    setCurrentTime(
      quiz.time
    );

    previousTimeRef.current =
      quiz.time;
  }

  /* =======================================================
     DUPLICATE QUESTION
  ======================================================= */

  function handleDuplicateQuiz(
    quiz: Quiz
  ) {
    if (!page) {
      return;
    }

    duplicateQuiz(
      page.id,
      quiz.id
    );
  }

  /* =======================================================
     DELETE QUESTION
  ======================================================= */

  function handleDeleteQuiz(
    quiz: Quiz
  ) {
    if (!page) {
      return;
    }

    const confirmed =
      window.confirm(
        `Bạn có chắc muốn xóa câu hỏi:\n\n"${quiz.question}"?`
      );

    if (!confirmed) {
      return;
    }

    deleteQuiz(
      page.id,
      quiz.id
    );

    if (
      activeQuiz?.id ===
      quiz.id
    ) {
      setActiveQuiz(
        null
      );

      setSelectedAnswer(
        null
      );

      setQuizResult(
        null
      );
    }
  }

  /* =======================================================
     SAFETY
  ======================================================= */

  if (!page) {
    return (
      <div className="flex h-screen items-center justify-center bg-[#061426] text-white">

        <div className="text-center">

          <div className="text-xl font-bold">
            Không có trang bài giảng
          </div>

          <button
            type="button"
            onClick={
              handleAddPage
            }
            className="mt-5 rounded-lg bg-blue-600 px-6 py-3 font-bold"
          >
            Tạo trang mới
          </button>

        </div>

      </div>
    );
  }

  /* =======================================================
     RENDER
  ======================================================= */

  return (
    <div className="flex h-screen flex-col overflow-hidden bg-[#061426] text-white">

      {/* TOP */}

      <EditorTopbar
        title={
          project.title
        }
        saveStatus={
          saveStatus
        }
        score={
          score
        }
        onSave={
          saveProject
        }
      />

      {/* WORKSPACE */}

      <div className="flex min-h-0 flex-1">

        {/* LEFT */}

        <PageNavigator
          pages={
            project.pages
          }
          selectedPageId={
            project.selectedPageId
          }
          onSelect={
            handleSelectPage
          }
          onAdd={
            handleAddPage
          }
          onRename={
            renamePage
          }
          onDuplicate={
            duplicatePage
          }
          onDelete={
            handleDeletePage
          }
        />

        {/* CENTER */}

        <main className="flex min-w-0 flex-1 flex-col">

          <VideoCanvas
            page={
              page
            }
            videoRef={
              videoRef
            }
            activeQuiz={
              activeQuiz
            }
            selectedAnswer={
              selectedAnswer
            }
            quizResult={
              quizResult
            }
            onVideoUpload={
              handleVideoUpload
            }
            onLoadedMetadata={
              handleLoadedMetadata
            }
            onTimeUpdate={
              handleTimeUpdate
            }
            onPlay={() =>
              setIsPlaying(
                true
              )
            }
            onPause={() =>
              setIsPlaying(
                false
              )
            }
            onAnswer={
              answerQuestion
            }
            onContinue={
              continueVideo
            }
          />

          <PlayerControls
            currentTime={
              currentTime
            }
            duration={
              page.duration
            }
            isPlaying={
              isPlaying
            }
            onTogglePlay={
              togglePlay
            }
            onSeek={
              handleSeek
            }
            onAddQuestion={
              openAddQuestion
            }
          />

          <Timeline
            duration={
              page.duration
            }
            currentTime={
              currentTime
            }
            quizzes={
              page.quizzes
            }
            onQuizClick={
              jumpToQuiz
            }
          />

        </main>

        {/* RIGHT */}

        <QuestionPanel
          page={
            page
          }
          onVideoUpload={
            handleVideoUpload
          }
          onJump={
            jumpToQuiz
          }
          onEdit={
            openEditQuestion
          }
          onDuplicate={
            handleDuplicateQuiz
          }
          onDelete={
            handleDeleteQuiz
          }
        />

      </div>

      {/* MODAL */}

      <QuestionEditor
        open={
          questionEditorOpen
        }
        editing={
          editingQuizId !==
          null
        }
        draft={
          draft
        }
        time={
          questionTime
        }
        duration={
          page.duration
        }
        onDraftChange={
          setDraft
        }
        onTimeChange={
          setQuestionTime
        }
        onClose={() => {
          setQuestionEditorOpen(
            false
          );

          setEditingQuizId(
            null
          );
        }}
        onSave={
          saveQuestion
        }
      />

    </div>
  );
}
