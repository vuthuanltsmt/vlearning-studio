/* =========================================================
   TYPES

type SortMode =
  | "updated-desc"
  | "updated-asc"
  | "title-asc"
  | "title-desc";

/* =========================================================
   DATE
========================================================= */

function formatUpdatedAt(
  value: string
) {
  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return "";
  }

  return date.toLocaleString(
    "vi-VN"
  );
}

/* =========================================================
   SEARCH
========================================================= */

function normalizeText(
  value: string
) {
  return value
    .normalize("NFD")
    .replace(
      /[\u0300-\u036f]/g,
      ""
    )
    .replace(
      /đ/g,
      "d"
    )
    .replace(
      /Đ/g,
      "D"
    )
    .toLowerCase()
    .trim();
}

/* =========================================================
   BLOB -> DATA URL
========================================================= */

function blobToDataUrl(
  blob: Blob
): Promise<string> {
  return new Promise(
    (
      resolve,
      reject
    ) => {
      const reader =
        new FileReader();

      reader.onload =
        () => {
          if (
            typeof reader.result ===
            "string"
          ) {
            resolve(
              reader.result
            );
          } else {
            reject(
              new Error(
                "Không đọc được ảnh."
              )
            );
          }
        };

      reader.onerror =
        () => {
          reject(
            reader.error ??
              new Error(
                "Không đọc được ảnh."
              )
          );
        };

      reader.readAsDataURL(
        blob
      );
    }
  );
}

  const [

    sortMode,

    );

    duplicatingId,

    useState<string | null>(

    thumbnails,

     FILTER + SORT
  ===================================================== */

    useMemo(() => {
      const keyword =
        normalizeText(
          searchQuery
        );

      const filtered =
        keyword
          ? projects.filter(
              (entry) =>
                normalizeText(
                  entry.project.title
                ).includes(
                  keyword
                )
            )
          : [...projects];

      return filtered.sort(
        (a, b) => {
          switch (
            sortMode
          ) {
            case "updated-asc":
              return (
                new Date(
                  a.updatedAt
                ).getTime() -
                new Date(
                  b.updatedAt
                ).getTime()
              );

            case "title-asc":
              return a.project.title.localeCompare(
                b.project.title,
                "vi"
              );

            case "title-desc":
              return b.project.title.localeCompare(
                a.project.title,
                "vi"
              );

            case "updated-desc":
            default:
              return (
                new Date(
                  b.updatedAt
                ).getTime() -
                new Date(
                  a.updatedAt
                ).getTime()
              );
          }
        }
      );
    }, [
      projects,
      searchQuery,
      sortMode,
    ]);

      input.files?.[0];

      return;

        .toLowerCase()
        .endsWith(

        )
    ) {
      alert(
        "Vui lòng chọn file .vlearning.zip hoặc .zip."
      );

      input.value =
        "";

      return;
    }

    setImporting(
      true
    );

    try {
      const result =
        await importProjectBackup(
          file
        );

      logActivity({
        type: "restore_project",
        projectId: result.project.id,
        projectTitle: result.project.title,
        message: `Đã khôi phục dự án "${result.project.title}" từ file backup`,
      });

      refreshProjects();

      if (
        result.warnings.length >
        0
      ) {
        alert(
          [
            "Đã nhập dự án.",
            "",
            `Tên: ${result.project.title}`,
            "",
            "Một số thành phần có cảnh báo:",
            "",
            ...result.warnings.map(
              (warning) =>
                `• ${warning}`
            ),
          ].join("\n")
        );

        return;
      }

      alert(
        [
          "Khôi phục dự án thành công!",
          "",
          `Tên: ${result.project.title}`,
          "",
          "Trang, quiz, video, điều kiện hoàn thành và ảnh đại diện đã được khôi phục.",
        ].join("\n")
      );
    } catch (
      error
    ) {
      console.error(
        "Nhập file backup thất bại:",
        error
      );

      alert(
        error instanceof Error
          ? `Không thể nhập dự án:\n${error.message}`
          : "Không thể nhập dự án."
      );
    } finally {
      input.value =
        "";

      setImporting(
        false
      );
    }
  }

  /* =====================================================
     THUMBNAIL
  ===================================================== */

  async function handleThumbnailChange(
    entry:
      ProjectLibraryEntry,
    event:
      ChangeEvent<HTMLInputElement>
  ) {
    const input =
      event.currentTarget;

    const file =
      input.files?.[0];

    if (!file) {
      return;
    }

    if (
      !file.type.startsWith(
        "image/"
      )
    ) {
      alert(
        "Vui lòng chọn file ảnh."
      );

      input.value =
        "";

      return;
    }

    if (
      file.size >
      5 * 1024 * 1024
    ) {
      alert(
        "Ảnh đại diện tối đa 5 MB."
      );

      input.value =
        "";

      return;
    }

    setThumbnailBusyId(
      entry.project.id
    );

    try {
      await saveProjectThumbnail(
        entry.project.id,
        file
      );

      const dataUrl =
        await blobToDataUrl(
          file
        );

      setThumbnails(
        (current) => ({
          ...current,

          [entry.project.id]:
            dataUrl,
        })
      );

      saveProjectToLibrary(
        entry.project
      );

      refreshProjects();
    } catch (
      error
    ) {
      console.error(
        "Không lưu được ảnh đại diện:",
        error
      );

      alert(
        "Không thể lưu ảnh đại diện."
      );
    } finally {
      input.value =
        "";

      setThumbnailBusyId(
        null
      );
    }
  }

      error

        "Không xóa được ảnh đại diện:",
        error
      );

      alert(
        "Không thể xóa ảnh đại diện."
      );
    } finally {
      setThumbnailBusyId(
        null
      );
    }
  }
