"use client";

import {
  Clock3,
  Copy,
  FileQuestion,
  FolderOpen,
  Layers3,
  Loader2,
  Pencil,
  Plus,
  Search,
  Trash2,
  X,
} from "lucide-react";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  createNewProject,
  deleteProjectFromLibrary,
  duplicateProjectInLibrary,
  getProjectLibrary,
  getProjectQuizCount,
  renameProjectInLibrary,
  setActiveProject,
  syncCurrentProjectToLibrary,
} from "@/lib/projectLibrary";

import type {
  ProjectLibraryEntry,
} from "@/lib/projectLibrary";

/* =========================================================
   DATE
========================================================= */

function formatUpdatedAt(
  value: string
) {
  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return "";
  }

  return date.toLocaleString(
    "vi-VN"
  );
}

/* =========================================================
   SEARCH NORMALIZER
========================================================= */

function normalizeText(
  value: string
) {
  return value
    .normalize("NFD")
    .replace(
      /[\u0300-\u036f]/g,
      ""
    )
    .replace(
      /đ/g,
      "d"
    )
    .replace(
      /Đ/g,
      "D"
    )
    .toLowerCase()
    .trim();
}

/* =========================================================
   PAGE
========================================================= */

export default function ProjectsPage() {
  const [
    projects,
    setProjects,
  ] =
    useState<
      ProjectLibraryEntry[]
    >([]);

  const [
    searchQuery,
    setSearchQuery,
  ] =
    useState("");

  const [
    duplicatingId,
    setDuplicatingId,
  ] =
    useState<string | null>(
      null
    );

  /* =====================================================
     FILTER
  ===================================================== */

  const filteredProjects =
    useMemo(() => {
      const keyword =
        normalizeText(
          searchQuery
        );

      if (!keyword) {
        return projects;
      }

      return projects.filter(
        (entry) =>
          normalizeText(
            entry.project.title
          ).includes(
            keyword
          )
      );
    }, [
      projects,
      searchQuery,
    ]);

  /* =====================================================
     REFRESH
  ===================================================== */

  function refreshProjects() {
    setProjects(
      getProjectLibrary()
    );
  }

  /* =====================================================
     LOAD
  ===================================================== */

  useEffect(() => {
    syncCurrentProjectToLibrary();

    refreshProjects();
  }, []);

  /* =====================================================
     OPEN
  ===================================================== */

  function handleOpen(
    entry:
      ProjectLibraryEntry
  ) {
    setActiveProject(
      entry.project
    );

    window.location.href =
      "/editor/demo";
  }

  /* =====================================================
     CREATE
  ===================================================== */

  function handleCreate() {
    const title =
      window.prompt(
        "Tên bài giảng mới:",
        "Bài giảng mới"
      );

    if (
      title === null
    ) {
      return;
    }

    const cleanTitle =
      title.trim();

    if (!cleanTitle) {
      alert(
        "Tên bài giảng không được để trống."
      );

      return;
    }

    createNewProject(
      cleanTitle
    );

    window.location.href =
      "/editor/demo";
  }

  /* =====================================================
     RENAME
  ===================================================== */

  function handleRename(
    entry:
      ProjectLibraryEntry
  ) {
    const newTitle =
      window.prompt(
        "Nhập tên mới cho bài giảng:",
        entry.project.title
      );

    if (
      newTitle === null
    ) {
      return;
    }

    const cleanTitle =
      newTitle.trim();

    if (!cleanTitle) {
      alert(
        "Tên bài giảng không được để trống."
      );

      return;
    }

    if (
      cleanTitle ===
      entry.project.title
    ) {
      return;
    }

    const renamed =
      renameProjectInLibrary(
        entry.project.id,
        cleanTitle
      );

    if (!renamed) {
      alert(
        "Không thể đổi tên dự án."
      );

      return;
    }

    refreshProjects();
  }

  /* =====================================================
     DUPLICATE
  ===================================================== */

  async function handleDuplicate(
    entry:
      ProjectLibraryEntry
  ) {
    if (
      duplicatingId
    ) {
      return;
    }

    const confirmed =
      window.confirm(
        `Nhân bản dự án "${entry.project.title}"?\n\nTrang, câu hỏi, video và điều kiện hoàn thành sẽ được sao chép sang dự án mới.`
      );

    if (!confirmed) {
      return;
    }

    setDuplicatingId(
      entry.project.id
    );

    try {
      const result =
        await duplicateProjectInLibrary(
          entry.project
        );

      refreshProjects();

      if (
        result.missingVideoPages.length >
        0
      ) {
        alert(
          [
            "Đã nhân bản dự án.",
            "",
            `Tên mới: ${result.project.title}`,
            "",
            "Tuy nhiên không tìm thấy video của:",
            "",
            ...result.missingVideoPages.map(
              (title) =>
                `• ${title}`
            ),
          ].join("\n")
        );

        return;
      }

      alert(
        [
          "Nhân bản dự án thành công!",
          "",
          `Tên mới: ${result.project.title}`,
          "",
          "Trang, câu hỏi, video và điều kiện hoàn thành đã được sao chép.",
        ].join("\n")
      );
    } catch (
      error
    ) {
      console.error(
        "Nhân bản dự án thất bại:",
        error
      );

      alert(
        error instanceof Error
          ? `Không thể nhân bản dự án:\n${error.message}`
          : "Không thể nhân bản dự án."
      );
    } finally {
      setDuplicatingId(
        null
      );
    }
  }

  /* =====================================================
     DELETE
  ===================================================== */

  function handleDelete(
    entry:
      ProjectLibraryEntry
  ) {
    const confirmed =
      window.confirm(
        [
          `Bạn có chắc muốn xóa dự án "${entry.project.title}"?`,
          "",
          "Thao tác này sẽ xóa:",
          "• Dự án khỏi danh sách",
          "• Video của dự án",
          "• Điều kiện hoàn thành",
          "",
          "Không thể hoàn tác.",
        ].join("\n")
      );

    if (!confirmed) {
      return;
    }

    deleteProjectFromLibrary(
      entry.project.id
    );

    refreshProjects();
  }

  /* =====================================================
     UI
  ===================================================== */

  return (
    <main className="min-h-screen bg-[#061426] px-6 py-8 text-white">

      <div className="mx-auto w-full max-w-7xl">

        {/* HEADER */}

        <div className="flex flex-wrap items-center justify-between gap-4">

          <div>

            <div className="text-sm font-bold uppercase tracking-[0.2em] text-cyan-300">
              VLearning Studio
            </div>

            <h1 className="mt-2 text-4xl font-black">
              Dự án của tôi
            </h1>

            <p className="mt-2 text-slate-400">
              Quản lý các bài giảng e-learning đã tạo.
            </p>

          </div>

          <button
            type="button"
            onClick={
              handleCreate
            }
            disabled={
              duplicatingId !==
              null
            }
            className="flex items-center gap-2 rounded-xl bg-gradient-to-r from-cyan-500 to-purple-600 px-6 py-3 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-50"
          >

            <Plus
              size={20}
            />

            Tạo bài giảng mới

          </button>

        </div>

        {/* SEARCH */}

        {projects.length >
          0 && (

          <div className="mt-8 flex flex-wrap items-center justify-between gap-4">

            <div className="relative w-full max-w-xl">

              <Search
                size={19}
                className="pointer-events-none absolute left-4 top-1/2 -translate-y-1/2 text-slate-400"
              />

              <input
                type="text"
                value={
                  searchQuery
                }
                onChange={(
                  event
                ) =>
                  setSearchQuery(
                    event.target.value
                  )
                }
                placeholder="Tìm kiếm bài giảng..."
                className="w-full rounded-xl border border-blue-800 bg-[#102b52] py-3.5 pl-12 pr-12 text-white outline-none transition placeholder:text-slate-500 focus:border-cyan-500"
              />

              {searchQuery && (

                <button
                  type="button"
                  onClick={() =>
                    setSearchQuery(
                      ""
                    )
                  }
                  title="Xóa tìm kiếm"
                  className="absolute right-3 top-1/2 -translate-y-1/2 rounded-lg p-2 text-slate-400 transition hover:bg-blue-900 hover:text-white"
                >
                  <X
                    size={17}
                  />
                </button>

              )}

            </div>

            <div className="text-sm text-slate-400">

              {searchQuery
                ? `${filteredProjects.length} / ${projects.length} dự án`
                : `${projects.length} dự án`}

            </div>

          </div>

        )}

        {/* EMPTY LIBRARY */}

        {projects.length ===
        0 ? (

          <div className="mt-12 rounded-3xl border border-dashed border-blue-700 bg-[#102b52] p-12 text-center">

            <FolderOpen
              size={52}
              className="mx-auto text-cyan-300"
            />

            <h2 className="mt-5 text-2xl font-black">
              Chưa có dự án
            </h2>

            <p className="mt-2 text-slate-400">
              Hãy tạo bài giảng đầu tiên của bạn.
            </p>

            <button
              type="button"
              onClick={
                handleCreate
              }
              className="mt-6 rounded-xl bg-blue-600 px-6 py-3 font-bold transition hover:bg-blue-500"
            >
              + Tạo bài giảng
            </button>

          </div>

        ) : filteredProjects.length ===
          0 ? (

          /* NO SEARCH RESULT */

          <div className="mt-8 rounded-3xl border border-dashed border-blue-800 bg-[#102b52] p-12 text-center">

            <Search
              size={46}
              className="mx-auto text-cyan-300"
            />

            <h2 className="mt-5 text-2xl font-black">
              Không tìm thấy bài giảng
            </h2>

            <p className="mt-2 text-slate-400">
              Không có dự án nào phù hợp với “{searchQuery}”.
            </p>

            <button
              type="button"
              onClick={() =>
                setSearchQuery(
                  ""
                )
              }
              className="mt-6 rounded-xl bg-blue-600 px-6 py-3 font-bold transition hover:bg-blue-500"
            >
              Xóa tìm kiếm
            </button>

          </div>

        ) : (

          /* PROJECT GRID */

          <div className="mt-8 grid gap-5 md:grid-cols-2 xl:grid-cols-3">

            {filteredProjects.map(
              (entry) => {
                const project =
                  entry.project;

                const quizCount =
                  getProjectQuizCount(
                    project
                  );

                const isDuplicating =
                  duplicatingId ===
                  project.id;

                return (
                  <article
                    key={
                      project.id
                    }
                    className="overflow-hidden rounded-2xl border border-blue-800 bg-[#123665] shadow-xl"
                  >

                    {/* CARD HEADER */}

                    <div className="border-b border-blue-800 bg-[#173f73] p-5">

                      <div className="flex items-start justify-between gap-4">

                        <div className="min-w-0">

                          <div className="text-xs font-bold uppercase tracking-wider text-cyan-300">
                            Bài giảng
                          </div>

                          <h2 className="mt-2 line-clamp-2 text-xl font-black">
                            {project.title}
                          </h2>

                        </div>

                        <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-blue-950 text-cyan-300">

                          <FolderOpen
                            size={21}
                          />

                        </div>

                      </div>

                    </div>

                    {/* CONTENT */}

                    <div className="p-5">

                      <div className="grid grid-cols-2 gap-3">

                        <div className="rounded-xl bg-blue-950/50 p-3">

                          <div className="flex items-center gap-2 text-xs text-slate-400">

                            <Layers3
                              size={14}
                            />

                            Số trang

                          </div>

                          <div className="mt-2 text-xl font-black text-cyan-300">
                            {project.pages.length}
                          </div>

                        </div>

                        <div className="rounded-xl bg-blue-950/50 p-3">

                          <div className="flex items-center gap-2 text-xs text-slate-400">

                            <FileQuestion
                              size={14}
                            />

                            Câu hỏi

                          </div>

                          <div className="mt-2 text-xl font-black text-purple-300">
                            {quizCount}
                          </div>

                        </div>

                      </div>

                      <div className="mt-4 flex items-center gap-2 text-xs text-slate-400">

                        <Clock3
                          size={14}
                        />

                        Cập nhật:{" "}

                        {formatUpdatedAt(
                          entry.updatedAt
                        )}

                      </div>

                      {/* PRIMARY ACTIONS */}

                      <div className="mt-5 grid grid-cols-2 gap-2">

                        <button
                          type="button"
                          onClick={() =>
                            handleOpen(
                              entry
                            )
                          }
                          disabled={
                            duplicatingId !==
                            null
                          }
                          className="flex items-center justify-center gap-2 rounded-xl bg-blue-600 px-4 py-3 font-bold transition hover:bg-blue-500 disabled:opacity-50"
                        >

                          <FolderOpen
                            size={18}
                          />

                          Mở dự án

                        </button>

                        <button
                          type="button"
                          onClick={() =>
                            handleDuplicate(
                              entry
                            )
                          }
                          disabled={
                            duplicatingId !==
                            null
                          }
                          className="flex items-center justify-center gap-2 rounded-xl bg-purple-700 px-4 py-3 font-bold transition hover:bg-purple-600 disabled:cursor-wait disabled:opacity-50"
                        >

                          {isDuplicating ? (

                            <Loader2
                              size={18}
                              className="animate-spin"
                            />

                          ) : (

                            <Copy
                              size={18}
                            />

                          )}

                          {isDuplicating
                            ? "Đang sao..."
                            : "Nhân bản"}

                        </button>

                      </div>

                      {/* SECONDARY ACTIONS */}

                      <div className="mt-2 grid grid-cols-2 gap-2">

                        <button
                          type="button"
                          onClick={() =>
                            handleRename(
                              entry
                            )
                          }
                          disabled={
                            duplicatingId !==
                            null
                          }
                          className="flex items-center justify-center gap-2 rounded-xl border border-cyan-800 bg-cyan-950/40 px-4 py-2.5 text-sm font-bold text-cyan-200 transition hover:bg-cyan-900 disabled:opacity-50"
                        >

                          <Pencil
                            size={16}
                          />

                          Đổi tên

                        </button>

                        <button
                          type="button"
                          onClick={() =>
                            handleDelete(
                              entry
                            )
                          }
                          disabled={
                            duplicatingId !==
                            null
                          }
                          className="flex items-center justify-center gap-2 rounded-xl border border-red-800 bg-red-950/50 px-4 py-2.5 text-sm font-bold text-red-300 transition hover:bg-red-900 disabled:opacity-50"
                        >

                          <Trash2
                            size={16}
                          />

                          Xóa dự án

                        </button>

                      </div>

                    </div>

                  </article>
                );
              }
            )}

          </div>

        )}

      </div>

    </main>
  );
}
