function formatUpdatedAt(
  value: string
) {
  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return "";
  }

  return date.toLocaleString(
    "vi-VN"
  );
}

   SEARCH
========================================================= */

function normalizeText(
  value: string
) {
  return value
    .normalize("NFD")
    .replace(
      /[\u0300-\u036f]/g,
      ""
    )
    .replace(
      /đ/g,
      "d"
    )
    .replace(
      /Đ/g,
      "D"
    )
    .toLowerCase()
    .trim();
}

/* =========================================================
   BLOB -> DATA URL
========================================================= */

function blobToDataUrl(
  blob: Blob
): Promise<string> {
  return new Promise(
    (
      resolve,
      reject
    ) => {
      const reader =
        new FileReader();

      reader.onload =
        () => {
          if (
            typeof reader.result ===
            "string"
          ) {
            resolve(
              reader.result
            );
          } else {
            reject(
              new Error(
                "Không đọc được ảnh."
              )
            );
          }
        };

      reader.onerror =
        () => {
          reject(
            reader.error ??
              new Error(
                "Không đọc được ảnh."
              )
          );
        };

      reader.readAsDataURL(
        blob
      );
    }
  );
}

type GameQuestion = {
  id: string;
  prompt: string;
  answers: [

};

type GameSet = {

  title: string;
  description: string;
  questions: GameQuestion[];

};

type GameLink = {
  id: string;
  gameId: string;
  projectId: string;
  pageId: string;
  attachedAt: string;
};

const GAME_LIBRARY_KEY =
  "vlearning-game-library-v1";

const GAME_LINKS_KEY =
  "vlearning-project-game-links-v1";

function makeGameId(
  prefix: string
) {
  try {
    return `${prefix}-${crypto.randomUUID()}`;
  } catch {
    return `${prefix}-${Date.now()}-${Math.random()
      .toString(36)
      .slice(2, 9)}`;
  }
}

function createBlankQuestion(): GameQuestion {
  return {
    id: makeGameId(
      "question"
    ),
    prompt: "",
    answers: [
      "",
      "",
      "",
      "",
    ],
    correct: 0,
  };
}

function createBlankGame(): GameSet {
  return {
    id: makeGameId(
      "game"
    ),
    title:
      "Trò chơi mới",
    description: "",
    questions: [
      createBlankQuestion(),
    ],
    updatedAt:
      new Date().toISOString(),
  };
}

function cloneGame(
  game: GameSet
): GameSet {
  return {
    ...game,
    questions:
      game.questions.map(
        (question) => ({
          ...question,
          answers: [
            question.answers[0],
            question.answers[1],
            question.answers[2],
            question.answers[3],
          ],
        })
      ),
  };
}

function normalizeGameLibrary(
  raw: unknown
): GameSet[] {
  if (
    !Array.isArray(
      raw
    )
  ) {
    return [];
  }

  return raw.flatMap(
    (entry) => {
      if (
        !entry ||
        typeof entry !==
          "object"
      ) {
        return [];
      }

      const source =
        entry as Partial<GameSet>;

      if (
        typeof source.id !==
          "string" ||
        typeof source.title !==
          "string" ||
        !Array.isArray(
          source.questions
        )
      ) {
        return [];
      }

      const questions =
        source.questions.flatMap(
          (question) => {
            if (
              !question ||
              typeof question !==
                "object"
            ) {
              return [];
            }

            const item =
              question as Partial<GameQuestion>;

            if (
              typeof item.id !==
                "string" ||
              typeof item.prompt !==
                "string" ||
              !Array.isArray(
                item.answers
              ) ||
              item.answers.length !==
                4
            ) {
              return [];
            }

            const answers =
              item.answers.map(
                (answer) =>
                  String(
                    answer ?? ""
                  )
              ) as [
                string,
                string,
                string,
                string,
              ];

            return [
              {
                id:
                  item.id,
                prompt:
                  item.prompt,
                answers,
                correct:
                  typeof item.correct ===
                    "number" &&
                  item.correct >=
                    0 &&
                  item.correct <
                    4
                    ? item.correct
                    : 0,
              },
            ];
          }
        );

      if (
        questions.length ===
        0
      ) {
        return [];
      }

      return [
        {
          id:
            source.id,
          title:
            source.title,
          description:
            typeof source.description ===
            "string"
              ? source.description
              : "",
          questions,
          updatedAt:
            typeof source.updatedAt ===
            "string"
              ? source.updatedAt
              : new Date().toISOString(),
        },
      ];
    }
  );
}


function normalizeGameLinks(
  raw: unknown
): GameLink[] {
  if (!Array.isArray(raw)) {
    return [];
  }

  return raw.flatMap(
    (entry) => {
      if (
        !entry ||
        typeof entry !==
          "object"
      ) {
        return [];
      }

      const item =
        entry as Partial<GameLink>;

      if (
        typeof item.id !==
          "string" ||
        typeof item.gameId !==
          "string" ||
        typeof item.projectId !==
          "string" ||
        typeof item.pageId !==
          "string"
      ) {
        return [];
      }

      return [
        {
          id:
            item.id,
          gameId:
            item.gameId,
          projectId:
            item.projectId,
          pageId:
            item.pageId,
          attachedAt:
            typeof item.attachedAt ===
            "string"
              ? item.attachedAt
              : new Date().toISOString(),
        },
      ];
    }
  );
}

    setAvailableProjects,
  ] =

    );

    setAttachProjectId,
  ] =
    useState("");

  const [
    attachPageId,
    setAttachPageId,
  ] =

  ] =

  const [
    current,
    setCurrent,
  ] =
    useState(0);

  const [
    score,
    setScore,
  ] =
    useState(0);

  const [
    selected,
    setSelected,
  ] =
    useState<
      number | null
    >(null);

  const [
    finished,
    setFinished,
  ] =
    useState(false);

  const [
    savedFlash,
    setSavedFlash,
  ] =
    useState(false);

  useEffect(() => {

      const raw =
        localStorage.getItem(

        );

      if (

        return;
      }

      const parsed =
        normalizeGameLibrary(

      }

        setGameLinks(

    setGames(
      next
    );

      );

      );

    const nextProjects =

    setAvailableProjects(
      nextProjects
    );

      showMessage
    ) {
      window.alert(
        nextProjects.length >
        0
          ? `Đã tìm thấy ${nextProjects.length} dự án.`
          : "Chưa tìm thấy dự án đã lưu. Hãy vào Dự án của tôi, mở hoặc tạo một dự án rồi quay lại."
      );
    }
  }

  useEffect(() => {
    reloadProjectsForGame(
      false
    );
  }, []);

  useEffect(() => {
    if (
      incomingProjects.length >
      0
    ) {
      setAvailableProjects(
        incomingProjects
      );
    }
  }, [
    incomingProjects,
  ]);

  function persistGameLinks(
    next: GameLink[]
  ) {
    setGameLinks(
      next
    );

    try {
      localStorage.setItem(
        GAME_LINKS_KEY,
        JSON.stringify(
          next
        )
      );
    } catch {
      window.alert(
        "Không thể lưu liên kết trò chơi với bài giảng."
      );

  }

  useEffect(() => {

      setAttachProjectId(
        ""
      );
      setAttachPageId(
        ""
      );
      return;
    }

    const currentProject =
      availableProjects.find(

          attachProjectId
      ) ??
      availableProjects[0];

    if (

      attachProjectId

      setAttachProjectId(
        currentProject.project.id
      );

    const pageExists =
      currentProject.project.pages.some(
        (pageItem) =>
          pageItem.id ===
          attachPageId
      );

    if (
      !pageExists
    ) {
      setAttachPageId(
        currentProject.project.pages[0]?.id ??
          ""
      );
    }
  }, [

    attachProjectId,
    attachPageId,
  ]);

  function attachCurrentGame() {
    if (
      !validateGame(
        draft
      )
    ) {
      return;
    }

    if (
      !attachProjectId ||
      !attachPageId
    ) {
      window.alert(
        "Hãy chọn dự án và trang muốn gắn trò chơi."
      );
      return;
    }

    const savedGame: GameSet =
      {
        ...cloneGame(
          draft
        ),
        title:
          draft.title.trim(),
        description:
          draft.description.trim(),
        updatedAt:
          new Date().toISOString(),
      };

      games.some(
        (game) =>
          game.id ===
          savedGame.id
      );

    const nextGames =
      exists
        ? games.map(
            (game) =>
              game.id ===
              savedGame.id
                ? savedGame
                : game
          )
        : [
            savedGame,
            ...games,
          ];

    persistGames(
      nextGames
    );

    setDraft(
      cloneGame(
        savedGame
      )
    );
    setActiveGameId(
      savedGame.id
    );

    const duplicate =
      gameLinks.some(
        (link) =>
          link.gameId ===
            savedGame.id &&
          link.projectId ===
            attachProjectId &&
          link.pageId ===
            attachPageId
      );

      duplicate
    ) {
      window.alert(
        "Trò chơi này đã được gắn vào trang đã chọn."
      );
      return;
    }

      ...gameLinks,

          makeGameId(

          ),

          savedGame.id,
        projectId:
          attachProjectId,
        pageId:
          attachPageId,
        attachedAt:
          new Date().toISOString(),
      },
    ]);

    window.alert(
      "Đã gắn trò chơi vào bài giảng."
    );
  }

  function validateGame(
    game: GameSet,
    showMessage = true
  ) {
    if (
      !game.title.trim()
    ) {
      if (
        showMessage
      ) {
        window.alert(
          "Hãy nhập tên trò chơi."
        );
      }
      return false;
    }

    if (
      game.questions.length ===
      0
    ) {
      if (
        showMessage
      ) {
        window.alert(
          "Trò chơi cần ít nhất 1 câu hỏi."
        );
      }
      return false;
    }

    for (
      let index = 0;
      index <
      game.questions.length;
      index += 1
    ) {
      const question =
        game.questions[
          index
        ];

      if (
        !question.prompt.trim()
      ) {
        if (
          showMessage
        ) {
          window.alert(
            `Câu ${index + 1} chưa có nội dung.`
          );
        }
        return false;
      }

      if (
        question.answers.some(
          (answer) =>
            !answer.trim()
        )
      ) {
        if (
          showMessage
        ) {
          window.alert(
            `Câu ${index + 1} cần đủ 4 đáp án.`
          );
        }
        return false;
      }
    }

    return true;
  }

    if (
      !validateGame(
        draft
      )
    ) {
      return;
    }

    const saved: GameSet =
      {
        ...cloneGame(
          draft
        ),
        title:
          draft.title.trim(),
        description:
          draft.description.trim(),
        updatedAt:
          new Date().toISOString(),
      };

    const exists =
      games.some(
        (game) =>
          game.id ===
          saved.id
      );

    const next =
      exists
        ? games.map(
            (game) =>
              game.id ===
              saved.id
                ? saved
                : game
          )
        : [
            saved,
            ...games,
          ];

      next
    );

      )

  }

  function duplicateGame() {

      !validateGame(
        draft
      )
    ) {
      return;
    }

    const copy: GameSet =
      {
        ...cloneGame(
          draft
        ),
        id:
          makeGameId(
            "game"
          ),
        title: `${draft.title.trim()} - Bản sao`,
        updatedAt:
          new Date().toISOString(),
        questions:
          draft.questions.map(
            (question) => ({
              ...question,
              id:
                makeGameId(
                  "question"
                ),
              answers: [
                question.answers[0],
                question.answers[1],
                question.answers[2],
                question.answers[3],
              ],
            })
          ),
      };

    persistGames(
      [
        copy,
        ...games,
      ]
    );
    setDraft(
      cloneGame(
        copy
      )
    );
    setActiveGameId(
      copy.id
    );
  }

  function deleteGame(
    gameId: string
  ) {
    const game =
      games.find(
        (item) =>
          item.id ===
          gameId
      );

    if (
      !game
    ) {
      return;
    }

    if (
      !window.confirm(
        `Xóa trò chơi "${game.title}"?`
      )
    ) {
      return;
    }

    const next =
      games.filter(
        (item) =>
          item.id !==
          gameId
      );

    persistGames(
      next
    );

    persistGameLinks(
      gameLinks.filter(
        (link) =>
          link.gameId !==
          gameId
      )
    );

    if (
      activeGameId ===
      gameId
    ) {
      if (
        next.length >
        0
      ) {
        setActiveGameId(
          next[0].id
        );
        setDraft(
          cloneGame(
            next[0]
          )
        );
      } else {
        setActiveGameId(
          null
        );
        setDraft(
          createBlankGame()
        );
      }
    }
  }

  function addQuestion() {
    setDraft(
      (currentDraft) => ({
        ...currentDraft,
        questions: [
          ...currentDraft.questions,
          createBlankQuestion(),
        ],
      })
    );
  }

  function removeQuestion(
    questionId: string
  ) {
    setDraft(
      (currentDraft) => {
        if (
          currentDraft.questions.length <=
          1
        ) {
          window.alert(
            "Trò chơi cần ít nhất 1 câu hỏi."
          );
          return currentDraft;
        }

        return {
          ...currentDraft,
          questions:
            currentDraft.questions.filter(
              (question) =>
                question.id !==
                questionId
            ),
        };
      }
    );
  }

  function updateQuestionPrompt(
    questionId: string,
    value: string
  ) {
    setDraft(
      (currentDraft) => ({
        ...currentDraft,
        questions:
          currentDraft.questions.map(
            (question) =>
              question.id ===
              questionId
                ? {
                    ...question,
                    prompt:
                      value,
                  }
                : question
          ),
      })
    );
  }

  function updateAnswer(
    questionId: string,
    answerIndex: number,
    value: string
  ) {
    setDraft(
      (currentDraft) => ({
        ...currentDraft,
        questions:
          currentDraft.questions.map(
            (question) => {
              if (
                question.id !==
                questionId
              ) {
                return question;
              }

              const answers =
                [
                  ...question.answers,
                ] as [
                  string,
                  string,
                  string,
                  string,
                ];

              answers[
                answerIndex
              ] =
                value;

              return {
                ...question,
                answers,
              };
            }
          ),
      })
    );
  }

  function setCorrectAnswer(
    questionId: string,
    answerIndex: number
  ) {
    setDraft(
      (currentDraft) => ({
        ...currentDraft,
        questions:
          currentDraft.questions.map(
            (question) =>
              question.id ===
              questionId
                ? {
                    ...question,
                    correct:
                      answerIndex,
                  }
                : question
          ),
      })
    );
  }

  function startPlay() {
    if (
      !validateGame(
        draft
      )
    ) {
      return;
    }

    setCurrent(0);
    setScore(0);
    setSelected(
      null
    );
    setFinished(
      false
    );
    setMode(
      "play"
    );
  }

  function choose(
    index: number
  ) {
    if (
      selected !==
        null ||
      finished
    ) {
      return;
    }

    setSelected(
      index
    );

    if (
      index ===
      draft.questions[
        current
      ].correct
    ) {
      setScore(
        (value) =>
          value + 1
      );
    }
  }

  function nextQuestion() {
    if (
      current >=
      draft.questions.length -
        1
    ) {
      setFinished(
        true
      );
      return;
    }

    setCurrent(
      (value) =>
        value + 1
    );
    setSelected(
      null
    );
  }

  function restartPlay() {
    setCurrent(0);
    setScore(0);
    setSelected(
      null
    );
    setFinished(
      false
    );
  }

    draft.questions[

  const selectedAttachProject =
    availableProjects.find(
      (entry) =>
        entry.project.id ===
        attachProjectId
    ) ??
    null;

  const currentGameLinks =
    gameLinks.filter(
      (link) =>
        link.gameId ===
        draft.id
    );

        (entry) =>

  return (

        onOpenMenu={

        }

      {mode ===

              type="button"
              onClick={() =>
                setMode(
                  "builder"
                )
              }
              className="rounded-xl border border-blue-700 bg-[#092542] px-4 py-2.5 text-xs font-black text-cyan-200 hover:bg-[#0d315a]"
            >
              ← Quay lại chỉnh sửa
            </button>

            <div className="rounded-full bg-cyan-500/10 px-3 py-1.5 text-xs font-black text-cyan-200">

                {Math.round(

              <div className="mt-6 flex flex-wrap justify-center gap-3">

                  Chơi lại

                  type="button"

                    size={17}

                  {

                        ) => (
                          <div
                            key={`${question.id}-${answerIndex}`}
                            className={`rounded-xl border p-3 ${
                              question.correct ===
                              answerIndex
                                ? "border-emerald-500/50 bg-emerald-500/10"
                                : "border-blue-800 bg-[#071d35]"
                            }`}
                          >
                            <label className="flex items-center gap-2 text-[11px] font-black text-slate-400">
                              <input
                                type="radio"
                                name={`correct-${question.id}`}
                                checked={
                                  question.correct ===
                                  answerIndex
                                }
                                onChange={() =>
                                  setCorrectAnswer(
                                    question.id,
                                    answerIndex
                                  )
                                }
                              />
                              ĐÁP ÁN{" "}
                              {String.fromCharCode(
                                65 +
                                  answerIndex
                              )}
                              {question.correct ===
                                answerIndex &&
                                " • ĐÚNG"}
                            </label>

                            <input
                              value={
                                answer
                              }
                              onChange={(
                                event
                              ) =>
                                updateAnswer(
                                  question.id,
                                  answerIndex,
                                  event.target.value
                                )
                              }
                              placeholder={`Nhập đáp án ${String.fromCharCode(
                                65 +
                                  answerIndex
                              )}`}
                              className="mt-2 w-full bg-transparent py-1 text-sm outline-none placeholder:text-slate-600"
                            />
                          </div>
                        )
                      )}
                    </div>
                  </article>
                )
              )}
            </div>

            <button
              type="button"
              onClick={
                addQuestion
              }
              className="mt-5 flex w-full items-center justify-center gap-2 rounded-xl border border-dashed border-cyan-600/60 bg-cyan-950/20 px-4 py-4 text-sm font-black text-cyan-300 hover:bg-cyan-950/40"
            >
              <Plus
                size={17}
              />
              Thêm câu hỏi
            </button>
          </section>
        </div>
      )}
    </div>
  );
}

function ScriptPanel({

Đối tượng: ${audience}

      message: `Đã tạo dự án "${cleanTitle}"`,
    });

      "/editor/demo";

  /* =====================================================
     RENAME
  ===================================================== */

  function handleRename(
    entry:
      ProjectLibraryEntry
  ) {
    const newTitle =
      window.prompt(
        "Nhập tên mới cho bài giảng:",
        entry.project.title
      );

    if (
      newTitle === null
    ) {
      return;
    }

    const cleanTitle =
      newTitle.trim();

    if (!cleanTitle) {
      alert(
        "Tên bài giảng không được để trống."
      );

      return;
    }

    if (
      cleanTitle ===
      entry.project.title
    ) {
      return;
    }

    const renamed =
      renameProjectInLibrary(
        entry.project.id,
        cleanTitle
      );

    if (!renamed) {
      alert(
        "Không thể đổi tên dự án."
      );

      return;
    }

    logActivity({
      type: "rename_project",
      projectId: entry.project.id,
      projectTitle: cleanTitle,
      message: `Đã đổi tên dự án "${entry.project.title}" thành "${cleanTitle}"`,
    });

    refreshProjects();
  }

  async function handleDuplicate(

    ) {

        "Nhân bản dự án thất bại:",

      setDuplicatingId(
        null
      );
    }
  }

  /* =====================================================
     IMPORT BACKUP
  ===================================================== */

  async function handleImportBackup(
    event:
      ChangeEvent<HTMLInputElement>
  ) {
    const input =
      event.currentTarget;

    const file =
      input.files?.[0];

    if (!file) {
      return;
    }

    if (
      !file.name
        .toLowerCase()
        .endsWith(
          ".zip"
        )
    ) {
      alert(
        "Vui lòng chọn file .vlearning.zip hoặc .zip."
      );

      input.value =
        "";

      return;
    }

    setImporting(
      true
    );

    try {
      const result =
        await importProjectBackup(
          file
        );

      logActivity({
        type: "restore_project",
        projectId: result.project.id,
        projectTitle: result.project.title,
        message: `Đã khôi phục dự án "${result.project.title}" từ file backup`,
      });

      refreshProjects();

      if (
        result.warnings.length >
        0
      ) {
        alert(
          [
            "Đã nhập dự án.",
            "",
            `Tên: ${result.project.title}`,
            "",
            "Một số thành phần có cảnh báo:",
            "",
            ...result.warnings.map(
              (warning) =>
                `• ${warning}`
            ),
          ].join("\n")
        );

        return;
      }

      alert(
        [
          "Khôi phục dự án thành công!",
          "",
          `Tên: ${result.project.title}`,
          "",
          "Trang, quiz, video, điều kiện hoàn thành và ảnh đại diện đã được khôi phục.",
        ].join("\n")
      );
    } catch (
      error
    ) {
      console.error(
        "Nhập file backup thất bại:",
        error
      );

      alert(
        error instanceof Error
          ? `Không thể nhập dự án:\n${error.message}`
          : "Không thể nhập dự án."
      );
    } finally {
      input.value =
        "";

      setImporting(
        false
      );
    }
  }

  /* =====================================================
     THUMBNAIL
  ===================================================== */

  async function handleThumbnailChange(
    entry:
      ProjectLibraryEntry,
    event:
      ChangeEvent<HTMLInputElement>
  ) {
    const input =
      event.currentTarget;

    const file =
      input.files?.[0];

    if (!file) {
      return;
    }

    if (
      !file.type.startsWith(
        "image/"
      )
    ) {
      alert(
        "Vui lòng chọn file ảnh."
      );

      input.value =
        "";

      return;
    }

    if (
      file.size >
      5 * 1024 * 1024
    ) {
      alert(
        "Ảnh đại diện tối đa 5 MB."
      );

      input.value =
        "";

      return;
    }

    setThumbnailBusyId(
      entry.project.id
    );

    try {
      await saveProjectThumbnail(
        entry.project.id,
        file
      );

      const dataUrl =
        await blobToDataUrl(
          file
        );

      setThumbnails(
        (current) => ({
          ...current,

          [entry.project.id]:
            dataUrl,
        })
      );

      saveProjectToLibrary(
        entry.project
      );

      refreshProjects();
    } catch (
      error
    ) {
      console.error(
        "Không lưu được ảnh đại diện:",
        error
      );

      alert(
        "Không thể lưu ảnh đại diện."
      );
    } finally {
      input.value =
        "";

      setThumbnailBusyId(
        null
      );
    }
  }

  async function handleDeleteThumbnail(
    entry:
      ProjectLibraryEntry
  ) {
    if (
      globalBusy
    ) {
      return;
    }

    const confirmed =
      window.confirm(
        `Xóa ảnh đại diện của "${entry.project.title}"?`
      );

    if (!confirmed) {
      return;
    }

    setThumbnailBusyId(
      entry.project.id
    );

    try {
      await deleteProjectThumbnail(
        entry.project.id
      );

      setThumbnails(
        (current) => {
          const next = {
            ...current,
          };

          delete next[
            entry.project.id
          ];

          return next;
        }
      );

      saveProjectToLibrary(
        entry.project
      );

      refreshProjects();
    } catch (
      error
    ) {
      console.error(
        "Không xóa được ảnh đại diện:",
        error
      );

      alert(
        "Không thể xóa ảnh đại diện."
      );
    } finally {
      setThumbnailBusyId(
        null
      );
    }
  }

                <h3 className="mt-4 text-2xl font-black">
                  Chưa có bài giảng
                </h3>

                <p className="mt-2 text-slate-400">
                  Tạo bài giảng đầu tiên hoặc nhập dự án đã sao lưu.
                </p>

              </div>

            ) : visibleProjects.length ===
              0 ? (

              <div className="mt-6 rounded-[24px] border border-dashed border-blue-700/60 bg-[#0b2a4c] p-12 text-center">
                <Search
                  size={44}
                  className="mx-auto text-cyan-300"
                />
                <h3 className="mt-4 text-xl font-black">
                  Không tìm thấy bài giảng
                </h3>
              </div>

            ) : (

              <div className="mt-6 grid gap-5 md:grid-cols-2 2xl:grid-cols-3">

                {visibleProjects.map(
                  (entry) => {
                    const project =
                      entry.project;

                    const quizCount =
                      getProjectQuizCount(
                        project
                      );

                    const thumbnail =
                      thumbnails[
                        project.id
                      ];

                    const isDuplicating =
                      duplicatingId ===
                      project.id;

                    const isThumbnailBusy =
                      thumbnailBusyId ===
                      project.id;

                    const isBackupBusy =
                      backupBusyId ===
                      project.id;

                    return (
                      <article
                        key={
                          project.id
                        }
                        className="group overflow-hidden rounded-[20px] border border-blue-800/70 bg-[#0d315a] shadow-[0_14px_45px_rgba(0,0,0,0.2)] transition hover:-translate-y-1 hover:border-cyan-500/50"
                      >

                        <div className="relative h-48 overflow-hidden bg-gradient-to-br from-[#1b4b8a] via-[#20447d] to-[#17335f]">

                          {thumbnail ? (
                            <img
                              src={
                                thumbnail
                              }
                              alt={`Ảnh đại diện ${project.title}`}
                              className="h-full w-full object-cover transition duration-500 group-hover:scale-[1.03]"
                            />
                          ) : (
                            <div className="flex h-full items-center justify-center">
                              <Play
                                size={42}
                                className="text-white/15"
                              />
                            </div>
                          )}

                          <div className="absolute left-4 top-4 rounded-lg border border-cyan-300/30 bg-sky-500/20 px-3 py-1.5 text-[10px] font-black uppercase tracking-wide text-cyan-200 backdrop-blur">
                            {project.pages.length} TRANG
                          </div>

                          <div className="absolute inset-x-0 bottom-0 flex justify-end gap-2 bg-gradient-to-t from-black/75 via-black/20 to-transparent p-3 pt-12">

                            <input
                              id={`thumbnail-${project.id}`}
                              type="file"
                              accept="image/png,image/jpeg,image/webp,image/gif"
                              className="hidden"
                              disabled={
                                globalBusy
                              }
                              onChange={(
                                event
                              ) =>
                                void handleThumbnailChange(
                                  entry,
                                  event
                                )
                              }
                            />

                            <label
                              htmlFor={`thumbnail-${project.id}`}
                              className={`flex cursor-pointer items-center gap-1.5 rounded-lg bg-black/55 px-3 py-2 text-[11px] font-black backdrop-blur transition hover:bg-black/75 ${
                                globalBusy
                                  ? "pointer-events-none opacity-50"
                                  : ""
                              }`}
                            >
                              {isThumbnailBusy ? (
                                <Loader2
                                  size={13}
                                  className="animate-spin"
                                />
                              ) : (
                                <ImagePlus
                                  size={13}
                                />
                              )}
                              {thumbnail
                                ? "Đổi ảnh"
                                : "Thêm ảnh"}
                            </label>

                            {thumbnail && (
                              <button
                                type="button"
                                onClick={() =>
                                  void handleDeleteThumbnail(
                                    entry
                                  )
                                }
                                disabled={
                                  globalBusy
                                }
                                className="flex h-8 w-8 items-center justify-center rounded-lg bg-red-950/75 text-red-200 backdrop-blur hover:bg-red-900 disabled:opacity-50"
                              >
                                <ImageOff
                                  size={14}
                                />
                              </button>
                            )}

                          </div>

                        </div>

                        <div className="p-5">

                          <div className="text-[10px] font-black uppercase tracking-[0.15em] text-cyan-300">
                            Bài giảng
                          </div>

                          <h3 className="mt-2 line-clamp-2 min-h-[52px] text-xl font-black leading-snug">
                            {project.title}
                          </h3>

                          <div className="mt-4 grid grid-cols-2 gap-2">

                            <div className="rounded-xl bg-[#0a294b] p-3">
                              <div className="flex items-center gap-2 text-[10px] font-bold uppercase text-slate-500">
                                <Layers3
                                  size={13}
                                />
                                Số trang
                              </div>
                              <div className="mt-1 text-xl font-black text-cyan-300">
                                {project.pages.length}
                              </div>
                            </div>

                            <div className="rounded-xl bg-[#0a294b] p-3">
                              <div className="flex items-center gap-2 text-[10px] font-bold uppercase text-slate-500">
                                <FileQuestion
                                  size={13}
                                />
                                Câu hỏi
                              </div>
                              <div className="mt-1 text-xl font-black text-fuchsia-300">
                                {quizCount}
                              </div>
                            </div>

                          </div>

                          <div className="mt-4 flex items-center gap-2 text-[11px] text-slate-500">
                            <Clock3
                              size={13}
                            />
                            Cập nhật:{" "}
                            {formatUpdatedAt(
                              entry.updatedAt
                            )}
                          </div>

                          <div className="mt-5 flex items-center gap-2">

                            <button
                              type="button"
                              onClick={() =>
                                handleOpen(
                                  entry
                                )
                              }
                              disabled={
                                globalBusy
                              }
                              className="flex flex-1 items-center justify-center gap-2 rounded-xl bg-blue-600 px-4 py-3 text-sm font-black transition hover:bg-blue-500 disabled:opacity-50"
                            >
                              <FolderOpen
                                size={17}
                              />
                              Mở dự án
                            </button>

                            <button
                              type="button"
                              onClick={() =>
                                void handleDuplicate(
                                  entry
                                )
                              }
                              disabled={
                                globalBusy
                              }
                              title="Nhân bản"
                              className="flex h-11 w-11 items-center justify-center rounded-xl bg-purple-950/60 text-purple-200 transition hover:bg-purple-900 disabled:opacity-50"
                            >
                              {isDuplicating ? (
                                <Loader2
                                  size={17}
                                  className="animate-spin"
                                />
                              ) : (
                                <Copy
                                  size={17}
                                />
                              )}
                            </button>

                            <button
                              type="button"
                              onClick={() =>
                                void handleBackup(
                                  entry
                                )
                              }
                              disabled={
                                globalBusy
                              }
                              title="Sao lưu"
                              className="flex h-11 w-11 items-center justify-center rounded-xl bg-emerald-950/60 text-emerald-200 transition hover:bg-emerald-900 disabled:opacity-50"
                            >
                              {isBackupBusy ? (
                                <Loader2
                                  size={17}
                                  className="animate-spin"
                                />
                              ) : (
                                <Download
                                  size={17}
                                />
                              )}
                            </button>

                          </div>

                          <div className="mt-2 grid grid-cols-2 gap-2">

                            <button
                              type="button"
                              onClick={() =>
                                handleRename(
                                  entry
                                )
                              }
                              disabled={
                                globalBusy
                              }
                              className="flex items-center justify-center gap-2 rounded-xl border border-blue-800 bg-[#12365f] px-3 py-2.5 text-xs font-bold text-slate-300 hover:text-white disabled:opacity-50"
                            >
                              <Pencil
                                size={14}
                              />
                              Đổi tên
                            </button>

                            <button
                              type="button"
                              onClick={() =>
                                handleDelete(
                                  entry
                                )
                              }
                              disabled={
                                globalBusy
                              }
                              className="flex items-center justify-center gap-2 rounded-xl border border-red-900 bg-red-950/25 px-3 py-2.5 text-xs font-bold text-red-300 hover:bg-red-950/50 disabled:opacity-50"
                            >
                              <Trash2
                                size={14}
                              />
                              Xóa
                            </button>

                          </div>

                        </div>

                      </article>
                    );
                  }
                )}

              </div>

            )}

          </section>

          </div>

        </div>

      </div>

    </main>
  );
}
