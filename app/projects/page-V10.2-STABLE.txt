"use client";

import {
  Clock3,
  FileQuestion,
  FolderOpen,
  Layers3,
  Plus,
  Trash2,
} from "lucide-react";

import {
  useEffect,
  useState,
} from "react";

import {
  createNewProject,
  deleteProjectFromLibrary,
  getProjectLibrary,
  getProjectQuizCount,
  setActiveProject,
  syncCurrentProjectToLibrary,
} from "@/lib/projectLibrary";

import type {
  ProjectLibraryEntry,
} from "@/lib/projectLibrary";

/* =========================================================
   DATE
========================================================= */

function formatUpdatedAt(
  value: string
) {
  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return "";
  }

  return date.toLocaleString(
    "vi-VN"
  );
}

/* =========================================================
   PAGE
========================================================= */

export default function ProjectsPage() {
  const [
    projects,
    setProjects,
  ] =
    useState<
      ProjectLibraryEntry[]
    >([]);

  /* =====================================================
     LOAD
  ===================================================== */

  function refreshProjects() {
    setProjects(
      getProjectLibrary()
    );
  }

  useEffect(() => {
    /*
     * Đưa project hiện tại
     * vào thư viện lần đầu.
     */
    syncCurrentProjectToLibrary();

    refreshProjects();
  }, []);

  /* =====================================================
     OPEN
  ===================================================== */

  function handleOpen(
    entry:
      ProjectLibraryEntry
  ) {
    setActiveProject(
      entry.project
    );

    /*
     * Full reload để Zustand
     * hydrate đúng project vừa chọn.
     */
    window.location.href =
      "/editor/demo";
  }

  /* =====================================================
     CREATE
  ===================================================== */

  function handleCreate() {
    const title =
      window.prompt(
        "Tên bài giảng mới:",
        "Bài giảng mới"
      );

    if (title === null) {
      return;
    }

    createNewProject(
      title
    );

    window.location.href =
      "/editor/demo";
  }

  /* =====================================================
     DELETE
  ===================================================== */

  function handleDelete(
    entry:
      ProjectLibraryEntry
  ) {
    const confirmed =
      window.confirm(
        `Xóa dự án "${entry.project.title}" khỏi danh sách?\n\nVideo trong bộ nhớ sẽ chưa bị xóa ở bước này.`
      );

    if (!confirmed) {
      return;
    }

    deleteProjectFromLibrary(
      entry.project.id
    );

    refreshProjects();
  }

  /* =====================================================
     UI
  ===================================================== */

  return (
    <main className="min-h-screen bg-[#061426] px-6 py-8 text-white">

      <div className="mx-auto w-full max-w-7xl">

        {/* HEADER */}

        <div className="flex flex-wrap items-center justify-between gap-4">

          <div>

            <div className="text-sm font-bold uppercase tracking-[0.2em] text-cyan-300">
              VLearning Studio
            </div>

            <h1 className="mt-2 text-4xl font-black">
              Dự án của tôi
            </h1>

            <p className="mt-2 text-slate-400">
              Quản lý các bài giảng e-learning đã tạo.
            </p>

          </div>

          <button
            type="button"
            onClick={
              handleCreate
            }
            className="flex items-center gap-2 rounded-xl bg-gradient-to-r from-cyan-500 to-purple-600 px-6 py-3 font-bold transition hover:brightness-110"
          >

            <Plus
              size={20}
            />

            Tạo bài giảng mới

          </button>

        </div>

        {/* EMPTY */}

        {projects.length === 0 ? (

          <div className="mt-12 rounded-3xl border border-dashed border-blue-700 bg-[#102b52] p-12 text-center">

            <FolderOpen
              size={52}
              className="mx-auto text-cyan-300"
            />

            <h2 className="mt-5 text-2xl font-black">
              Chưa có dự án
            </h2>

            <p className="mt-2 text-slate-400">
              Hãy tạo bài giảng đầu tiên của bạn.
            </p>

            <button
              type="button"
              onClick={
                handleCreate
              }
              className="mt-6 rounded-xl bg-blue-600 px-6 py-3 font-bold transition hover:bg-blue-500"
            >
              + Tạo bài giảng
            </button>

          </div>

        ) : (

          /* PROJECT GRID */

          <div className="mt-10 grid gap-5 md:grid-cols-2 xl:grid-cols-3">

            {projects.map(
              (entry) => {
                const project =
                  entry.project;

                const quizCount =
                  getProjectQuizCount(
                    project
                  );

                return (
                  <article
                    key={
                      project.id
                    }
                    className="overflow-hidden rounded-2xl border border-blue-800 bg-[#123665] shadow-xl"
                  >

                    {/* CARD HEADER */}

                    <div className="border-b border-blue-800 bg-[#173f73] p-5">

                      <div className="flex items-start justify-between gap-4">

                        <div className="min-w-0">

                          <div className="text-xs font-bold uppercase tracking-wider text-cyan-300">
                            Bài giảng
                          </div>

                          <h2 className="mt-2 line-clamp-2 text-xl font-black">
                            {project.title}
                          </h2>

                        </div>

                        <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-blue-950 text-cyan-300">

                          <FolderOpen
                            size={21}
                          />

                        </div>

                      </div>

                    </div>

                    {/* STATS */}

                    <div className="p-5">

                      <div className="grid grid-cols-2 gap-3">

                        <div className="rounded-xl bg-blue-950/50 p-3">

                          <div className="flex items-center gap-2 text-xs text-slate-400">

                            <Layers3
                              size={14}
                            />

                            Số trang

                          </div>

                          <div className="mt-2 text-xl font-black text-cyan-300">
                            {project.pages.length}
                          </div>

                        </div>

                        <div className="rounded-xl bg-blue-950/50 p-3">

                          <div className="flex items-center gap-2 text-xs text-slate-400">

                            <FileQuestion
                              size={14}
                            />

                            Câu hỏi

                          </div>

                          <div className="mt-2 text-xl font-black text-purple-300">
                            {quizCount}
                          </div>

                        </div>

                      </div>

                      <div className="mt-4 flex items-center gap-2 text-xs text-slate-400">

                        <Clock3
                          size={14}
                        />

                        Cập nhật:{" "}

                        {formatUpdatedAt(
                          entry.updatedAt
                        )}

                      </div>

                      {/* ACTIONS */}

                      <div className="mt-5 grid grid-cols-[1fr_auto] gap-2">

                        <button
                          type="button"
                          onClick={() =>
                            handleOpen(
                              entry
                            )
                          }
                          className="flex items-center justify-center gap-2 rounded-xl bg-blue-600 px-4 py-3 font-bold transition hover:bg-blue-500"
                        >

                          <FolderOpen
                            size={18}
                          />

                          Mở dự án

                        </button>

                        <button
                          type="button"
                          onClick={() =>
                            handleDelete(
                              entry
                            )
                          }
                          title="Xóa dự án"
                          className="flex items-center justify-center rounded-xl border border-red-800 bg-red-950/60 px-4 text-red-300 transition hover:bg-red-900"
                        >

                          <Trash2
                            size={18}
                          />

                        </button>

                      </div>

                    </div>

                  </article>
                );
              }
            )}

          </div>

        )}

      </div>

    </main>
  );
}
