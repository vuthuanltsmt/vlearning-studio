"use client";

import {
  ArrowRight,
  BookOpen,
  Clock3,
  FileQuestion,
  FolderOpen,
  Layers3,
  Plus,
  Sparkles,
} from "lucide-react";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  createNewProject,
  getProjectLibrary,
  getProjectQuizCount,
  setActiveProject,
  syncCurrentProjectToLibrary,
} from "@/lib/projectLibrary";

import type {
  ProjectLibraryEntry,
} from "@/lib/projectLibrary";

/* =========================================================
   DATE
========================================================= */

function formatUpdatedAt(
  value: string
) {
  const date =
    new Date(value);

  if (
    Number.isNaN(
      date.getTime()
    )
  ) {
    return "";
  }

  return date.toLocaleString(
    "vi-VN"
  );
}

/* =========================================================
   HOME PAGE
========================================================= */

export default function HomePage() {
  const [
    projects,
    setProjects,
  ] =
    useState<
      ProjectLibraryEntry[]
    >([]);

  /* =====================================================
     LOAD PROJECTS
  ===================================================== */

  useEffect(() => {
    syncCurrentProjectToLibrary();

    setProjects(
      getProjectLibrary()
    );
  }, []);

  /* =====================================================
     STATISTICS
  ===================================================== */

  const statistics =
    useMemo(() => {
      const totalProjects =
        projects.length;

      const totalPages =
        projects.reduce(
          (
            total,
            entry
          ) =>
            total +
            entry.project.pages.length,
          0
        );

      const totalQuizzes =
        projects.reduce(
          (
            total,
            entry
          ) =>
            total +
            getProjectQuizCount(
              entry.project
            ),
          0
        );

      return {
        totalProjects,
        totalPages,
        totalQuizzes,
      };
    }, [
      projects,
    ]);

  /* =====================================================
     RECENT PROJECTS
  ===================================================== */

  const recentProjects =
    useMemo(
      () =>
        projects.slice(
          0,
          6
        ),
      [
        projects,
      ]
    );

  /* =====================================================
     CREATE PROJECT
  ===================================================== */

  function handleCreateProject() {
    const title =
      window.prompt(
        "Tên bài giảng mới:",
        "Bài giảng mới"
      );

    if (
      title === null
    ) {
      return;
    }

    const cleanTitle =
      title.trim();

    if (
      !cleanTitle
    ) {
      alert(
        "Tên bài giảng không được để trống."
      );

      return;
    }

    createNewProject(
      cleanTitle
    );

    window.location.href =
      "/editor/demo";
  }

  /* =====================================================
     OPEN PROJECT
  ===================================================== */

  function handleOpenProject(
    entry:
      ProjectLibraryEntry
  ) {
    setActiveProject(
      entry.project
    );

    window.location.href =
      "/editor/demo";
  }

  /* =====================================================
     OPEN PROJECT LIBRARY
  ===================================================== */

  function handleOpenLibrary() {
    window.location.href =
      "/projects";
  }

  /* =====================================================
     UI
  ===================================================== */

  return (
    <main className="min-h-screen bg-[#061426] text-white">

      {/* =================================================
          HERO
      ================================================= */}

      <section className="border-b border-blue-950 bg-gradient-to-br from-[#0b2344] via-[#102e59] to-[#151b46]">

        <div className="mx-auto flex min-h-[350px] max-w-7xl flex-col justify-center px-6 py-14">

          <div className="flex items-center gap-2 text-sm font-black uppercase tracking-[0.25em] text-cyan-300">

            <Sparkles
              size={18}
            />

            VLearning Studio

          </div>

          <h1 className="mt-5 max-w-3xl text-4xl font-black leading-tight md:text-6xl">

            Xây dựng bài giảng

            <span className="block bg-gradient-to-r from-cyan-300 to-purple-400 bg-clip-text text-transparent">
              e-learning tương tác
            </span>

          </h1>

          <p className="mt-5 max-w-2xl text-base leading-7 text-slate-300 md:text-lg">
            Tạo bài giảng có video, câu hỏi tương tác,
            điều kiện hoàn thành, sao lưu dự án và xuất
            SCORM 1.2.
          </p>

          <div className="mt-8 flex flex-wrap gap-3">

            <button
              type="button"
              onClick={
                handleCreateProject
              }
              className="flex items-center gap-2 rounded-xl bg-gradient-to-r from-cyan-500 to-purple-600 px-6 py-3.5 font-bold transition hover:brightness-110"
            >

              <Plus
                size={20}
              />

              Tạo bài giảng mới

            </button>

            <button
              type="button"
              onClick={
                handleOpenLibrary
              }
              className="flex items-center gap-2 rounded-xl border border-blue-600 bg-blue-950/60 px-6 py-3.5 font-bold transition hover:bg-blue-900"
            >

              <FolderOpen
                size={20}
              />

              Dự án của tôi

            </button>

          </div>

        </div>

      </section>

      {/* =================================================
          CONTENT
      ================================================= */}

      <div className="mx-auto max-w-7xl px-6 py-10">

        {/* =================================================
            STATISTICS
        ================================================= */}

        <section>

          <div className="mb-5">

            <div className="text-xs font-black uppercase tracking-[0.18em] text-cyan-300">
              Tổng quan
            </div>

            <h2 className="mt-2 text-2xl font-black">
              Không gian làm việc
            </h2>

          </div>

          <div className="grid gap-4 md:grid-cols-3">

            {/* PROJECTS */}

            <div className="rounded-2xl border border-blue-800 bg-[#102b52] p-6 shadow-xl">

              <div className="flex items-start justify-between gap-4">

                <div>

                  <div className="text-sm text-slate-400">
                    Dự án
                  </div>

                  <div className="mt-2 text-4xl font-black text-cyan-300">
                    {
                      statistics.totalProjects
                    }
                  </div>

                </div>

                <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-cyan-950 text-cyan-300">

                  <FolderOpen
                    size={24}
                  />

                </div>

              </div>

            </div>

            {/* PAGES */}

            <div className="rounded-2xl border border-blue-800 bg-[#102b52] p-6 shadow-xl">

              <div className="flex items-start justify-between gap-4">

                <div>

                  <div className="text-sm text-slate-400">
                    Tổng số trang
                  </div>

                  <div className="mt-2 text-4xl font-black text-blue-300">
                    {
                      statistics.totalPages
                    }
                  </div>

                </div>

                <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-blue-950 text-blue-300">

                  <Layers3
                    size={24}
                  />

                </div>

              </div>

            </div>

            {/* QUIZZES */}

            <div className="rounded-2xl border border-blue-800 bg-[#102b52] p-6 shadow-xl">

              <div className="flex items-start justify-between gap-4">

                <div>

                  <div className="text-sm text-slate-400">
                    Câu hỏi tương tác
                  </div>

                  <div className="mt-2 text-4xl font-black text-purple-300">
                    {
                      statistics.totalQuizzes
                    }
                  </div>

                </div>

                <div className="flex h-12 w-12 items-center justify-center rounded-xl bg-purple-950 text-purple-300">

                  <FileQuestion
                    size={24}
                  />

                </div>

              </div>

            </div>

          </div>

        </section>

        {/* =================================================
            RECENT PROJECTS
        ================================================= */}

        <section className="mt-12">

          <div className="flex flex-wrap items-end justify-between gap-4">

            <div>

              <div className="text-xs font-black uppercase tracking-[0.18em] text-cyan-300">
                Gần đây
              </div>

              <h2 className="mt-2 text-2xl font-black">
                Dự án gần đây
              </h2>

            </div>

            {projects.length >
              0 && (

              <button
                type="button"
                onClick={
                  handleOpenLibrary
                }
                className="flex items-center gap-2 text-sm font-bold text-cyan-300 transition hover:text-cyan-200"
              >

                Xem tất cả

                <ArrowRight
                  size={17}
                />

              </button>

            )}

          </div>

          {/* NO PROJECT */}

          {recentProjects.length ===
          0 ? (

            <div className="mt-6 rounded-3xl border border-dashed border-blue-700 bg-[#102b52] p-12 text-center">

              <BookOpen
                size={50}
                className="mx-auto text-cyan-300"
              />

              <h3 className="mt-5 text-xl font-black">
                Chưa có bài giảng
              </h3>

              <p className="mt-2 text-slate-400">
                Tạo dự án đầu tiên để bắt đầu xây dựng
                bài giảng tương tác.
              </p>

              <button
                type="button"
                onClick={
                  handleCreateProject
                }
                className="mt-6 inline-flex items-center gap-2 rounded-xl bg-blue-600 px-6 py-3 font-bold transition hover:bg-blue-500"
              >

                <Plus
                  size={18}
                />

                Tạo bài giảng

              </button>

            </div>

          ) : (

            <div className="mt-6 grid gap-5 md:grid-cols-2 xl:grid-cols-3">

              {recentProjects.map(
                (
                  entry
                ) => {
                  const project =
                    entry.project;

                  const quizCount =
                    getProjectQuizCount(
                      project
                    );

                  return (
                    <article
                      key={
                        project.id
                      }
                      className="group overflow-hidden rounded-2xl border border-blue-800 bg-[#123665] shadow-xl transition hover:-translate-y-1 hover:border-cyan-700"
                    >

                      {/* CARD TOP */}

                      <div className="bg-gradient-to-br from-[#173f73] to-[#1b2f66] p-5">

                        <div className="flex items-start justify-between gap-4">

                          <div className="min-w-0">

                            <div className="text-xs font-bold uppercase tracking-wider text-cyan-300">
                              Bài giảng
                            </div>

                            <h3 className="mt-2 line-clamp-2 min-h-[56px] text-xl font-black">
                              {
                                project.title
                              }
                            </h3>

                          </div>

                          <div className="flex h-11 w-11 shrink-0 items-center justify-center rounded-xl bg-blue-950 text-cyan-300">

                            <FolderOpen
                              size={21}
                            />

                          </div>

                        </div>

                      </div>

                      {/* CARD BODY */}

                      <div className="p-5">

                        <div className="grid grid-cols-2 gap-3">

                          <div className="rounded-xl bg-blue-950/50 p-3">

                            <div className="flex items-center gap-2 text-xs text-slate-400">

                              <Layers3
                                size={14}
                              />

                              Trang

                            </div>

                            <div className="mt-2 text-xl font-black text-cyan-300">
                              {
                                project.pages.length
                              }
                            </div>

                          </div>

                          <div className="rounded-xl bg-blue-950/50 p-3">

                            <div className="flex items-center gap-2 text-xs text-slate-400">

                              <FileQuestion
                                size={14}
                              />

                              Câu hỏi

                            </div>

                            <div className="mt-2 text-xl font-black text-purple-300">
                              {
                                quizCount
                              }
                            </div>

                          </div>

                        </div>

                        <div className="mt-4 flex items-center gap-2 text-xs text-slate-400">

                          <Clock3
                            size={14}
                          />

                          Cập nhật:{" "}

                          {
                            formatUpdatedAt(
                              entry.updatedAt
                            )
                          }

                        </div>

                        <button
                          type="button"
                          onClick={() =>
                            handleOpenProject(
                              entry
                            )
                          }
                          className="mt-5 flex w-full items-center justify-center gap-2 rounded-xl bg-blue-600 px-4 py-3 font-bold transition hover:bg-blue-500"
                        >

                          <FolderOpen
                            size={18}
                          />

                          Mở dự án

                        </button>

                      </div>

                    </article>
                  );
                }
              )}

            </div>

          )}

        </section>

        {/* =================================================
            RECENT ACTIVITY
        ================================================= */}

        {recentProjects.length >
          0 && (

          <section className="mt-12">

            <div>

              <div className="text-xs font-black uppercase tracking-[0.18em] text-cyan-300">
                Hoạt động
              </div>

              <h2 className="mt-2 text-2xl font-black">
                Cập nhật gần đây
              </h2>

            </div>

            <div className="mt-6 overflow-hidden rounded-2xl border border-blue-800 bg-[#102b52]">

              {recentProjects
                .slice(
                  0,
                  5
                )
                .map(
                  (
                    entry,
                    index
                  ) => (

                    <button
                      key={
                        entry.project.id
                      }
                      type="button"
                      onClick={() =>
                        handleOpenProject(
                          entry
                        )
                      }
                      className={`flex w-full items-center justify-between gap-4 px-5 py-4 text-left transition hover:bg-blue-900/60 ${
                        index !==
                        Math.min(
                          recentProjects.length,
                          5
                        ) -
                          1
                          ? "border-b border-blue-900"
                          : ""
                      }`}
                    >

                      <div className="flex min-w-0 items-center gap-4">

                        <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-blue-950 text-cyan-300">

                          <BookOpen
                            size={18}
                          />

                        </div>

                        <div className="min-w-0">

                          <div className="truncate font-bold">
                            {
                              entry.project.title
                            }
                          </div>

                          <div className="mt-1 text-xs text-slate-400">
                            Dự án được cập nhật
                          </div>

                        </div>

                      </div>

                      <div className="shrink-0 text-xs text-slate-400">
                        {
                          formatUpdatedAt(
                            entry.updatedAt
                          )
                        }
                      </div>

                    </button>

                  )
                )}

            </div>

          </section>

        )}

      </div>

      {/* =================================================
          FOOTER
      ================================================= */}

      <footer className="mt-14 border-t border-blue-950 bg-[#07182c]">

        <div className="mx-auto flex max-w-7xl flex-wrap items-center justify-between gap-3 px-6 py-6 text-sm text-slate-500">

          <div>
            VLearning Studio
          </div>

          <div>
            E-learning Authoring • SCORM 1.2
          </div>

        </div>

      </footer>

    </main>
  );
}
