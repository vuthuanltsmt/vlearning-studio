/* =========================================================
   HELPERS
========================================================= */

function safeFileName(
  value: string
) {
  const cleaned =
    value
      .normalize("NFD")
      .replace(
        /[\u0300-\u036f]/g,
        ""
      )
      .replace(
        /đ/g,
        "d"
      )
      .replace(
        /Đ/g,
        "D"
      )
      .replace(
        /[^a-zA-Z0-9-_]+/g,
        "-"
      )
      .replace(
        /-+/g,
        "-"
      )
      .replace(
        /^-|-$/g,
        ""
      );

  return (
    cleaned ||
    "elearning-course"
  );
}

function xmlEscape(
  value: string
) {
  return value
    .replace(
      /&/g,
      "&amp;"
    )
    .replace(
      /</g,
      "&lt;"
    )
    .replace(
      />/g,
      "&gt;"
    )
    .replace(
      /"/g,
      "&quot;"
    )
    .replace(
      /'/g,
      "&apos;"
    );
}

function getVideoExtension(
  file: File
) {
  const nameMatch =
    file.name.match(
      /\.[a-zA-Z0-9]{1,8}$/
    );

  if (nameMatch) {
    return nameMatch[0]
      .toLowerCase();
  }

  switch (file.type) {
    case "video/webm":
      return ".webm";

    case "video/ogg":
      return ".ogv";

    case "video/quicktime":
      return ".mov";

    default:
      return ".mp4";
  }
}

function makeCourseDataScript(
  course: ExportCourse
) {
  /*
   * Escape "<" để dữ liệu an toàn
   * khi được JavaScript đọc.
   */
  const json =
    JSON.stringify(
      course
    ).replace(
      /</g,
      "\\u003c"
    );

  return `window.COURSE_DATA = ${json};`;
}

/* =========================================================
   MANIFEST
========================================================= */

function makeManifest(
  course: ExportCourse,
  files: string[]
) {
  const courseTitle =
    xmlEscape(
      course.title
    );

  const identifier =
    `VLEARNING-${safeFileName(
      course.id
    )}`;

  const fileEntries =
    files
      .map(
        (file) =>
          `      <file href="${xmlEscape(
            file
          )}" />`
      )
      .join("\n");

  return `<?xml version="1.0" encoding="UTF-8"?>
<manifest
  identifier="${identifier}"
  version="1.0"
  xmlns="http://www.imsproject.org/xsd/imscp_rootv1p1p2"
  xmlns:adlcp="http://www.adlnet.org/xsd/adlcp_rootv1p2"
  xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
  xsi:schemaLocation="
    http://www.imsproject.org/xsd/imscp_rootv1p1p2
    imscp_rootv1p1p2.xsd
    http://www.adlnet.org/xsd/adlcp_rootv1p2
    adlcp_rootv1p2.xsd
  "
>
  <metadata>
    <schema>ADL SCORM</schema>
    <schemaversion>1.2</schemaversion>
  </metadata>

  <organizations default="ORG-1">
    <organization identifier="ORG-1">
      <title>${courseTitle}</title>

      <item
        identifier="ITEM-1"
        identifierref="RES-1"
      >
        <title>${courseTitle}</title>

        <adlcp:masteryscore>${course.completion.passScorePercent}</adlcp:masteryscore>
      </item>
    </organization>
  </organizations>

  <resources>
    <resource
      identifier="RES-1"
      type="webcontent"
      adlcp:scormtype="sco"
      href="index.html"
    >
${fileEntries}
    </resource>
  </resources>
</manifest>
`;
}

  function quizCorrectIndexes(
    quiz
  ) {
    if (
      quizType(quiz) !==
      "multiple"

      return [

        quiz.correctIndexes

    result.sort(
      function (a, b) {
        return a - b;

    if (result.length > 0) {
      return result;
    }

    return [
      Number.isInteger(
        quiz.correctIndex
      )
        ? quiz.correctIndex
        : 0
    ];
  }

    itemIndex,
    categoryIndex
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "classification"
    ) {
      return;
    }

    var items =
      classificationItems(
        quiz
      );

    if (
      itemIndex < 0 ||
      itemIndex >=
        items.length
    ) {
      return;
    }

    var normalizedCategory =
      categoryIndex === 0 ||
      categoryIndex === 1
        ? categoryIndex
        : -1;

    while (
      state.classificationSelections.length <
      items.length
    ) {
      state.classificationSelections.push(
        -1
      );
    }

    state.classificationSelections[
      itemIndex
    ] =
      normalizedCategory;

    render();
  }

  function submitClassificationAnswer() {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "classification"
    ) {
      return;
    }

    var categories =
      classificationCategories(
        quiz
      );

    var items =
      classificationItems(
        quiz
      );

    if (
      !categories[0] ||
      !categories[1] ||
      items.length < 2 ||
      state.classificationSelections.length !==
        items.length ||
      state.classificationSelections.some(
        function (value) {
          return value < 0;
        }
      )
    ) {
      return;
    }

    var correct =
      state.classificationSelections
        .every(
          function (
            categoryIndex,
            itemIndex
          ) {
            return (
              categoryIndex ===
              items[itemIndex]
                .categoryIndex
            );
          }
        );

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function matchingPairs(
    quiz
  ) {
    var pairs =
      quiz &&
      Array.isArray(
        quiz.matchingPairs
      )
        ? quiz.matchingPairs
        : [];

    return pairs
      .map(
        function (pair) {
          return {
            left: String(
              pair &&
              pair.left != null
                ? pair.left
                : ""
            ).trim(),
            right: String(
              pair &&
              pair.right != null
                ? pair.right
                : ""
            ).trim(),
          };
        }
      )
      .filter(
        function (pair) {
          return (
            pair.left &&
            pair.right
          );
        }
      );
  }

  function changeMatchingSelection(
    leftIndex,
    rightIndex
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "matching"
    ) {
      return;
    }

    var pairs =
      matchingPairs(quiz);

    var next =
      [];

    for (
      var index = 0;
      index < pairs.length;
      index++
    ) {
      next[index] =
        typeof state.matchingSelections[index] ===
        "number"
          ? state.matchingSelections[index]
          : -1;
    }

    next[leftIndex] =
      rightIndex;

    state.matchingSelections =
      next;

    render();
  }

  function submitMatchingAnswer() {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "matching"
    ) {
      return;
    }

    var pairs =
      matchingPairs(quiz);

    if (
      pairs.length < 2 ||
      state.matchingSelections.length !==
        pairs.length ||
      state.matchingSelections.some(
        function (value) {
          return value < 0;
        }
      )
    ) {
      return;
    }

    var correct =
      state.matchingSelections.every(
        function (
          value,
          index
        ) {
          return value === index;
        }
      );

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function normalizeShortAnswerValue(
    value
  ) {
    return String(
      value || ""
    )
      .normalize("NFKC")
      .trim()
      .replace(
        /\s+/g,
        " "
      )
      .toLocaleLowerCase(
        "vi"
      );
  }

  function isShortAnswerCorrect(
    answer,
    acceptedAnswers
  ) {
    var normalizedAnswer =
      normalizeShortAnswerValue(
        answer
      );

    if (!normalizedAnswer) {
      return false;
    }

    var accepted =
      Array.isArray(
        acceptedAnswers
      )
        ? acceptedAnswers
        : [];

    return accepted.some(
      function (
        item
      ) {
        return (
          normalizeShortAnswerValue(
            item
          ) ===
          normalizedAnswer
        );
      }
    );
  }

      if (

    /* ===================================================
       COMPLETE PAGE WITHOUT CALLING render()
    =================================================== */

    function checkPageCompletion() {
      /*
       * Đã hoàn thành rồi.
       */
      if (
        state.completedPageIds.has(
          page.id
        )
      ) {
        updateCompletionStateUi();

        return true;
      }

      /*
       * Quiz bắt buộc nhưng chưa làm hết.
       */
      if (
        !requiredQuizzesDone()
      ) {
        updateCompletionStateUi();

        return false;
      }

      var totalDuration =
        video.duration;

      if (
        !Number.isFinite(
          totalDuration
        ) ||
        totalDuration <= 0
      ) {
        totalDuration =
          state.duration ||
          page.duration;
      }

      var percent =
        watchPercent(
          page.id,
          totalDuration
        );

      /*
       * Chưa xem đủ tỷ lệ yêu cầu.
       */
      if (
        percent <
        course.completion
          .minWatchPercent
      ) {
        updateCompletionStateUi();

        return false;
      }

      /*
       * Đạt toàn bộ điều kiện.
       *
       * QUAN TRỌNG:
       * Không gọi completeCurrentPage()
       * vì function đó render lại player,
       * có thể làm video bị gián đoạn.
       */
      state
        .completedPageIds
        .add(
          page.id
        );

      state.completionNotice =
        "";

      updateCompletionStateUi();

      return true;
    }

    /* ===================================================
       METADATA
    =================================================== */

    video.addEventListener(
      "loadedmetadata",
      function () {
        state.duration =
          video.duration;

        /*
         * render() có thể tạo lại thẻ video
         * sau khi quiz xuất hiện.
         *
         * Khôi phục lại đúng vị trí trước đó.
         */
        if (
          state.currentTime > 0 &&
          state.currentTime <
            video.duration
        ) {
          video.currentTime =
            Math.min(
              state.currentTime,
              Math.max(
                0,
                video.duration -
                  0.05
              )
            );
        }

        updateControls();

        /*
         * Hỗ trợ cả cấu hình
         * minWatchPercent = 0.
         */
        checkPageCompletion();
      }
    );

    /* ===================================================
       PLAY
    =================================================== */

    video.addEventListener(
      "play",
      function () {
        state.started =
          true;

        state.playing =
          true;

        state.lastPlaybackTime =
          video.currentTime;

        updateControls();
      }
    );

    /* ===================================================
       PAUSE
    =================================================== */

    video.addEventListener(
      "pause",
      function () {
        state.playing =
          false;

        state.lastPlaybackTime =
          null;

        updateControls();
      }
    );

    /* ===================================================
       TIME UPDATE
    =================================================== */

    video.addEventListener(
      "timeupdate",
      function () {
        state.currentTime =
          video.currentTime;

        /*
         * Chỉ ghi nhận thời gian xem thật
         * khi video đang phát.
         */
        if (
          state.started &&
          !video.paused
        ) {
          recordWatch(
            page.id,
            video.currentTime,
            video.duration
          );
        }

        updateControls();

        /*
         * Đây là thay đổi chính của V9.4:
         *
         * mỗi lần thời gian xem tăng,
         * kiểm tra ngay điều kiện hoàn thành.
         *
         * Ví dụ yêu cầu 70%:
         * vừa đạt 70% thì Trang tiếp theo
         * được mở ngay.
         */
        checkPageCompletion();

        /*
         * Chưa Play hoặc đang hiện quiz
         * thì không tìm quiz mới.
         */
        if (
          !state.started ||
          state.activeQuiz
        ) {
          return;
        }

        var due =
          sortedQuizzes(
            page
          ).find(
            function (quiz) {
              return (
                !handledQuiz(
                  quiz.id
                ) &&
                quiz.time <=
                  video.currentTime
              );
            }
          );

        if (due) {
          showQuiz(
            due
          );
        }
      }
    );

    /* ===================================================
       VIDEO ENDED
    =================================================== */

    video.addEventListener(
      "ended",
      function () {
        state.playing =
          false;

        /*
         * Ghi nhận nốt đoạn cuối.
         */
        recordWatch(
          page.id,
          video.duration,
          video.duration
        );

        state.lastPlaybackTime =
          null;

        /*
         * Nếu quiz là bắt buộc thì
         * không được bỏ sót quiz.
         */
        if (
          course.completion
            .requireAllQuizzes
        ) {
          var unanswered =
            sortedQuizzes(
              page
            ).find(
              function (
                quiz
              ) {
                return !state
                  .answeredQuizIds
                  .has(
                    quiz.id
                  );
              }
            );

          if (
            unanswered
          ) {
            video.currentTime =
              unanswered.time;

            state.currentTime =
              unanswered.time;

            showQuiz(
              unanswered
            );

            return;
          }
        }

        var percent =
          watchPercent(
            page.id,
            video.duration
          );

        /*
         * Chạy đến cuối nhưng vẫn chưa đủ
         * tỷ lệ xem thật, ví dụ do tua.
         */
        if (
          percent <
          course.completion
            .minWatchPercent
        ) {
          var missing =
            firstUnwatchedSecond(
              page.id,
              video.duration
            );

          state.completionNotice =
            "Bạn mới xem " +
            percent +
            "% video. Cần tối thiểu " +
            course.completion
              .minWatchPercent +
            "% để hoàn thành trang.";

          video.currentTime =
            missing;

          state.currentTime =
            missing;

          updateControls();

          updateCompletionStateUi();

          return;
        }

        /*
         * Nếu đủ điều kiện thì chỉ cập nhật
         * state/UI, không render lại video.
         */
        checkPageCompletion();
      }
    );

    /*
     * Đồng bộ trạng thái ngay lần đầu.
     */
    updateCompletionStateUi();
  }

  function seekVideo(
    value
  ) {
    var page =
      currentPage();

    var video =
      document.getElementById(
        "course-video"
      );

    if (
      !page ||
      !video ||
      state.activeQuiz
    ) {
      return;
    }

    var requested =
      Number(value);

    state.started =
      true;

    state.lastPlaybackTime =
      null;

    if (
      course.completion
        .requireAllQuizzes &&
      requested >
        state.currentTime
    ) {
      var blocking =
        sortedQuizzes(
          page
        ).find(
          function (
            quiz
          ) {
            return (
              !handledQuiz(
                quiz.id
              ) &&
              quiz.time >
                state.currentTime &&
              quiz.time <=
                requested
            );
          }
        );

      if (
        blocking
      ) {
        video.currentTime =
          blocking.time;

        state.currentTime =
          blocking.time;

        showQuiz(
          blocking
        );

        return;
      }
    }

    video.currentTime =
      requested;

    state.currentTime =
      requested;

    updateControls();
  }

        function (

        ) {
          return !state
            .answeredQuizIds
            .has(
              quiz.id
            );
        }
      );

    if (next) {
      showQuiz(
        next
      );

      return;
    }

    completeCurrentPage();
  }

  /* =====================================================

  function renderQuiz() {
    var quiz =
      state.activeQuiz;

    if (!quiz) {
      return "";
    }

    var type =
      quizType(quiz);

    var correctIndexes =
      quizCorrectIndexes(
        quiz
      );

    var answerArea =
      "";

    if (
      type ===
        "classification"
    ) {
      var categories =
        classificationCategories(
          quiz
        );

      var items =
        classificationItems(
          quiz
        );

      var unassigned =
        items
          .map(
            function (
              item,
              itemIndex
            ) {
              var selected =
                typeof state.classificationSelections[itemIndex] ===
                  "number"
                  ? state.classificationSelections[itemIndex]
                  : -1;

              if (selected >= 0) {
                return "";
              }

              var buttons =
                state.quizResult
                  ? ""
                  : categories
                      .map(
                        function (
                          category,
                          categoryIndex
                        ) {
                          return (
                            '<button class="classification-action" data-classify-item="' +
                            itemIndex +
                            '" data-classify-category="' +
                            categoryIndex +
                            '">→ ' +
                            escapeHtml(
                              category ||
                              (
                                "Nhóm " +
                                (
                                  categoryIndex +
                                  1
                                )
                              )
                            ) +
                            "</button>"
                          );
                        }
                      )
                      .join("");

              return (
                '<div class="classification-item" draggable="' +
                (state.quizResult ? "false" : "true") +
                '" data-classify-drag-item="' +
                itemIndex +
                '">' +
                '<div class="classification-item-text">' +
                escapeHtml(
                  item.text
                ) +
                "</div>" +
                (
                  buttons
                    ? '<div class="classification-actions">' +
                      buttons +
                      "</div>"
                    : ""
                ) +
                "</div>"
              );
            }
          )
          .join("");

      var groups =
        categories
          .map(
            function (
              category,
              categoryIndex
            ) {
              var cards =
                items
                  .map(
                    function (
                      item,
                      itemIndex
                    ) {
                      if (
                        state.classificationSelections[itemIndex] !==
                        categoryIndex
                      ) {
                        return "";
                      }

                      var classes =
                        "classification-item";

                      if (
                        state.quizResult
                      ) {
                        classes +=
                          item.categoryIndex ===
                          categoryIndex
                            ? " correct"
                            : " wrong";
                      }

                      var buttons =
                        state.quizResult
                          ? ""
                          : categories
                              .map(
                                function (
                                  targetCategory,
                                  targetIndex
                                ) {
                                  return (
                                    '<button class="classification-action" data-classify-item="' +
                                    itemIndex +
                                    '" data-classify-category="' +
                                    targetIndex +
                                    '"' +
                                    (
                                      targetIndex ===
                                      categoryIndex
                                        ? " disabled"
                                        : ""
                                    ) +
                                    '>→ ' +
                                    escapeHtml(
                                      targetCategory ||
                                      (
                                        "Nhóm " +
                                        (
                                          targetIndex +
                                          1
                                        )
                                      )
                                    ) +
                                    "</button>"
                                  );
                                }
                              )
                              .join("") +
                            '<button class="classification-action" data-classify-item="' +
                            itemIndex +
                            '" data-classify-category="-1">↩ Bỏ chọn</button>';

                      return (
                        '<div class="' +
                        classes +
                        '" draggable="' +
                        (state.quizResult ? "false" : "true") +
                        '" data-classify-drag-item="' +
                        itemIndex +
                        '">' +
                        '<div class="classification-item-text">' +
                        escapeHtml(
                          item.text
                        ) +
                        "</div>" +
                        (
                          buttons
                            ? '<div class="classification-actions">' +
                              buttons +
                              "</div>"
                            : ""
                        ) +
                        "</div>"
                      );
                    }
                  )
                  .join("");

              return (
                '<div class="classification-group" data-classify-drop-category="' +
                categoryIndex +
                '">' +
                '<div class="classification-group-title">' +
                escapeHtml(
                  category ||
                  (
                    "Nhóm " +
                    (
                      categoryIndex +
                      1
                    )
                  )
                ) +
                "</div>" +
                cards +
                "</div>"
              );
            }
          )
          .join("");

      var allClassified =
        items.length >= 2 &&
        state.classificationSelections.length ===
          items.length &&
        state.classificationSelections.every(
          function (value) {
            return value >= 0;
          }
        );

      answerArea =
        '<div class="classification-wrap">' +
        '<div class="classification-note">Kéo từng thẻ vào đúng nhóm. Nếu không muốn kéo, có thể dùng nút nhóm trên từng thẻ.</div>' +
        '<div class="classification-unassigned">' +
        unassigned +
        "</div>" +
        '<div class="classification-groups">' +
        groups +
        "</div>" +
        (
          state.quizResult
            ? ""
            : '<button class="primary-button" id="submit-classification"' +
              (
                allClassified
                  ? ""
                  : " disabled"
              ) +
              '>Xác nhận phân loại</button>'
        ) +
        "</div>";
    } else if (
      type ===
        "ordering"
    ) {
      var orderItems =
        orderingItems(
          quiz
        );

      var orderRows =
        state.orderingSelections
          .map(
            function (
              sourceIndex,
              position
            ) {
              var classes =
                "ordering-row";

              if (
                state.quizResult
              ) {
                classes +=
                  sourceIndex ===
                  position
                    ? " correct"
                    : " wrong";
              }

              return (
                '<div class="' +
                classes +
                '">' +
                '<div class="ordering-position">' +
                (position + 1) +
                "</div>" +
                '<div class="ordering-text">' +
                escapeHtml(
                  orderItems[
                    sourceIndex
                  ] || ""
                ) +
                "</div>" +
                (
                  state.quizResult
                    ? ""
                    : '<div class="ordering-controls">' +
                      '<button class="ordering-move" data-order-position="' +
                      position +
                      '" data-order-direction="-1"' +
                      (
                        position ===
                          0
                          ? " disabled"
                          : ""
                      ) +
                      '>↑</button>' +
                      '<button class="ordering-move" data-order-position="' +
                      position +
                      '" data-order-direction="1"' +
                      (
                        position ===
                          state.orderingSelections.length -
                            1
                          ? " disabled"
                          : ""
                      ) +
                      '>↓</button>' +
                      "</div>"
                ) +
                "</div>"
              );
            }
          )
          .join("");

      answerArea =
        '<div class="ordering-wrap">' +
        orderRows +
        (
          state.quizResult
            ? ""
            : '<button class="primary-button" id="submit-ordering"' +
              (
                state.orderingSelections.length <
                  2
                  ? " disabled"
                  : ""
              ) +
              '>Xác nhận thứ tự</button>'
        ) +
        "</div>";
    } else if (
      type ===
        "matching"
    ) {
      var pairs =
        matchingPairs(quiz);

      var rows =
        pairs.map(
          function (
            pair,
            leftIndex
          ) {
            var selected =
              typeof state.matchingSelections[leftIndex] ===
              "number"
                ? state.matchingSelections[leftIndex]
                : -1;

            var rowClass =
              "matching-row";

            if (state.quizResult) {
              rowClass +=
                selected ===
                leftIndex
                  ? " correct"
                  : " wrong";
            }

            var optionOrder =
              pairs.map(
                function (
                  _,
                  index
                ) {
                  return (
                    pairs.length -
                    1 -
                    index
                  );
                }
              );

            var options =
              '<option value="-1">-- Chọn nội dung phù hợp --</option>' +
              optionOrder
                .map(
                  function (rightIndex) {
                    return (
                      '<option value="' +
                      rightIndex +
                      '"' +
                      (
                        selected ===
                        rightIndex
                          ? " selected"
                          : ""
                      ) +
                      '>' +
                      escapeHtml(
                        pairs[rightIndex].right
                      ) +
                      "</option>"
                    );
                  }
                )
                .join("");

            return (
              '<div class="' +
              rowClass +
              '">' +
              '<div style="font-weight:700">' +
              escapeHtml(
                pair.left
              ) +
              '</div>' +
              '<div class="matching-arrow">↔</div>' +
              '<select class="matching-select" data-match-left="' +
              leftIndex +
              '"' +
              (
                state.quizResult
                  ? " disabled"
                  : ""
              ) +
              '>' +
              options +
              '</select>' +
              '</div>'
            );
          }
        )
        .join("");

      var allSelected =
        pairs.length >= 2 &&
        pairs.every(
          function (
            _,
            index
          ) {
            return (
              typeof state.matchingSelections[index] ===
                "number" &&
              state.matchingSelections[index] >=
                0
            );
          }
        );

      answerArea =
        '<div class="matching-wrap">' +
        rows +
        (
          state.quizResult
            ? ""
            : '<button class="primary-button" id="submit-matching"' +
              (
                allSelected
                  ? ""
                  : " disabled"
              ) +
              '>Xác nhận ghép đôi</button>'
        ) +
        '</div>';
    } else if (
      type ===
        "short-answer"
    ) {
      answerArea =
        '<div class="short-answer-wrap">' +
        '<input class="short-answer-input" id="short-answer-input" type="text" placeholder="Nhập câu trả lời của bạn..." value="' +
        escapeHtml(
          state.shortAnswer ||
          ""
        ) +
        '"' +
        (
          state.quizResult
            ? " disabled"
            : ""
        ) +
        " />" +
        (
          state.quizResult
            ? ""
            : '<button class="primary-button" id="submit-short-answer"' +
              (
                String(
                  state.shortAnswer ||
                  ""
                ).trim()
                  ? ""
                  : " disabled"
              ) +
              '>Xác nhận câu trả lời</button>'
        ) +
        "</div>";
    } else {
      var options =
        quiz.options
          .map(
            function (
              option,
              index
            ) {
              var classes =
                "option";

              var isCorrect =
                correctIndexes
                  .indexOf(index) >=
                0;

              var isSelected =
                type ===
                "multiple"
                  ? state.selectedAnswers
                      .indexOf(index) >=
                    0
                  : index ===
                    state.selectedAnswer;

              if (
                state.quizResult
              ) {
                if (isCorrect) {
                  classes +=
                    " correct";
                } else if (isSelected) {
                  classes +=
                    " wrong";
                } else {
                  classes +=
                    " dim";
                }
              } else if (
                type ===
                  "multiple" &&
                isSelected
              ) {
                classes +=
                  " selected";
              }

              return (
                '<button class="' +
                classes +
                '" data-answer="' +
                index +
                '"' +
                (
                  state.quizResult
                    ? " disabled"
                    : ""
                ) +
                ">" +
                '<span class="option-letter">' +
                (
                  type ===
                    "true-false"
                    ? (
                        index ===
                        0
                          ? "✓"
                          : "✕"
                      )
                    : String.fromCharCode(
                        65 + index
                      )
                ) +
                "</span>" +
                "<span>" +
                escapeHtml(
                  option
                ) +
                "</span>" +
                (
                  type ===
                    "multiple" &&
                  isSelected &&
                  !state.quizResult
                    ? '<span style="margin-left:auto;color:#67e8f9;font-weight:900">✓</span>'
                    : ""
                ) +
                "</button>"
              );
            }
          )
          .join("");

      var submit =
        "";

      if (
        type ===
          "multiple" &&
        !state.quizResult
      ) {
        submit =
          '<button class="primary-button" id="submit-multiple"' +
          (
            state.selectedAnswers.length ===
              0
              ? " disabled"
              : ""
          ) +
          '>Xác nhận đáp án</button>';
      }

      answerArea =
        '<div class="options">' +
        options +
        "</div>" +
        submit;
    }

    var feedback =
      "";

    if (
      state.quizResult
    ) {
      var correct =
        state.quizResult ===
        "correct";

      feedback =
        '<div class="feedback">' +
        '<div class="feedback-title ' +
        (
          correct
            ? "correct"
            : "wrong"
        ) +
        '">' +
        (
          correct
            ? "Chính xác! +" +
              quiz.points +
              " điểm"
            : "Chưa chính xác"
        ) +
        "</div>" +
        "<p>" +
        escapeHtml(
          correct
            ? quiz.feedbackCorrect
            : quiz.feedbackWrong
        ) +
        "</p>" +
        '<button class="primary-button" id="continue-quiz">Tiếp tục</button>' +
        "</div>";
    }

    var skip =
      "";

    if (
      !course.completion
        .requireAllQuizzes &&
      !state.quizResult
    ) {
      skip =
        '<button class="skip-button" id="skip-quiz">Bỏ qua câu hỏi</button>';
    }

    return (
      '<div class="quiz-overlay">' +
      '<div class="quiz-card">' +
      '<div class="quiz-sticky-header">' +
      '<div class="quiz-top">' +
      '<div class="quiz-label">' +
      (
        type ===
          "true-false"
          ? "CÂU HỎI ĐÚNG / SAI"
          : type ===
            "multiple"
            ? "CÂU HỎI NHIỀU ĐÁP ÁN"
            : type ===
              "short-answer"
              ? "CÂU HỎI TRẢ LỜI NGẮN"
              : type ===
                "matching"
                ? "CÂU HỎI GHÉP ĐÔI"
                : type ===
                  "ordering"
                  ? "CÂU HỎI SẮP XẾP THỨ TỰ"
                  : type ===
                    "classification"
                    ? "CÂU HỎI PHÂN LOẠI KÉO THẢ"
                    : "CÂU HỎI TƯƠNG TÁC"
      ) +
      "</div>" +
      '<div class="quiz-points">' +
      quiz.points +
      " điểm</div>" +
      "</div>" +
      "<h2>" +
      escapeHtml(
        quiz.question
      ) +
      "</h2>" +
      "</div>" +
      '<div class="quiz-body">' +
      answerArea +
      feedback +
      skip +
      "</div>" +
      "</div>" +
      "</div>"
    );
  }

  /* =====================================================
     HTML ESCAPE
  ===================================================== */

  function escapeHtml(
    value
  ) {
    return String(
      value == null
        ? ""
        : value
    )
      .replace(
        /&/g,
        "&amp;"
      )
      .replace(
        /</g,
        "&lt;"
      )
      .replace(
        />/g,
        "&gt;"
      )
      .replace(
        /"/g,
        "&quot;"
      )
      .replace(
        /'/g,
        "&#039;"
      );
  }

  /* =====================================================
     COURSE PROGRESS
  ===================================================== */

  function courseProgressPercent(
    completed
  ) {
    return Math.min(
      100,
      Math.round(
        (
          state.pageIndex +
          (completed ? 1 : 0)
        ) /
          Math.max(
            1,
            course.pages.length
          ) *
          100
      )
    );
  }

  function renderCourseProgress(
    completed
  ) {
    if (
      theme.showProgress ===
      false
    ) {
      return "";
    }

    var percent =
      courseProgressPercent(
        completed
      );

    return (
      '<section class="course-progress">' +
      '<div class="course-progress-head">' +
      '<span>Tiến độ bài giảng</span>' +
      '<strong id="course-progress-percent">' +
      percent +
      '%</strong>' +
      '</div>' +
      '<div class="course-progress-track">' +
      '<div class="course-progress-fill" id="course-progress-fill" style="width:' +
      percent +
      '%"></div>' +
      '</div>' +
      '</section>'
    );
  }

  /* =====================================================
     RENDER FINISH
  ===================================================== */

  function renderFinish() {
    var possible =
      totalPossibleScore();

    var percentage =
      possible > 0
        ? Math.round(
            state.score /
              possible *
              100
          )
        : 100;

    var passed =
      percentage >=
      course.completion
        .passScorePercent;

    app.innerHTML =
      '<div class="finish-page">' +
      '<div class="finish-card">' +

      '<div class="finish-status ' +
      (
        passed
          ? "pass"
          : "fail"
      ) +
      '">' +
      (
        passed
          ? "ĐẠT"
          : "CHƯA ĐẠT"
      ) +
      "</div>" +

      "<h1>HOÀN THÀNH BÀI HỌC</h1>" +

      "<p>" +
      escapeHtml(
        displayCourseTitle()
      ) +
      "</p>" +

      '<div class="result-grid">' +

      '<div class="result-box">' +
      '<div class="result-label">Tổng điểm</div>' +
      '<div class="result-value">' +
      state.score +
      " / " +
      possible +
      "</div>" +
      "</div>" +

      '<div class="result-box">' +
      '<div class="result-label">Trả lời đúng</div>' +
      '<div class="result-value">' +
      state.correctQuizIds.size +
      " / " +
      totalQuestions() +
      "</div>" +
      "</div>" +

      '<div class="result-box">' +
      '<div class="result-label">Kết quả</div>' +
      '<div class="result-value">' +
      percentage +
      "%</div>" +
      "</div>" +

      "</div>" +

      '<div class="conditions">' +
      "<strong>Điều kiện hoàn thành</strong>" +

      '<div class="condition-row">' +
      "<span>Điểm tối thiểu</span>" +
      "<strong>" +
      course.completion
        .passScorePercent +
      "%</strong>" +
      "</div>" +

      '<div class="condition-row">' +
      "<span>Xem tối thiểu mỗi video</span>" +
      "<strong>" +
      course.completion
        .minWatchPercent +
      "%</strong>" +
      "</div>" +

      '<div class="condition-row">' +
      "<span>Hoàn thành tất cả câu hỏi</span>" +
      "<strong>" +
      (
        course.completion
          .requireAllQuizzes
          ? "Bắt buộc"
          : "Không bắt buộc"
      ) +
      "</strong>" +
      "</div>" +

      "</div>" +

      '<button class="restart" id="restart">Học lại</button>' +

      "</div>" +
      "</div>";

    document
      .getElementById(
        "restart"
      )
      .addEventListener(
        "click",
        restartCourse
      );
  }

  /* =====================================================
     RENDER MAIN
  ===================================================== */

  function render() {
    if (
      state.finished
    ) {
      renderFinish();

      return;
    }

    var page =
      currentPage();

    if (!page) {
      app.innerHTML =
        '<div class="loading">Không có nội dung bài giảng.</div>';

      return;
    }

    var completed =
      state.completedPageIds
        .has(
          page.id
        );

    var watch =
      watchPercent(
        page.id,
        state.duration ||
          page.duration
      );

    var videoHtml =
      page.videoSrc
        ? (
            '<video id="course-video" preload="metadata" src="' +
            escapeHtml(
              page.videoSrc
            ) +
            '"></video>'
          )
        : (
            '<div class="no-video">Trang này chưa có video</div>'
          );

    var notice =
      state.completionNotice
        ? (
            '<div class="progress-notice">' +
            escapeHtml(
              state.completionNotice
            ) +
            "</div>"
          )
        : completed
          ? (
              '<div class="completed">✓ Trang đã hoàn thành</div>'
            )
          : (
              '<div class="progress-notice">' +
              "Yêu cầu: xem ≥ " +
              course.completion
                .minWatchPercent +
              "%" +
              (
                course.completion
                  .requireAllQuizzes
                  ? " và hoàn thành tất cả câu hỏi"
                  : ""
              ) +
              "</div>"
            );

    app.innerHTML =
      '<div class="player">' +

      '<header class="topbar">' +

      "<div>" +
      '<div class="project-title">' +
      escapeHtml(
        displayCourseTitle()
      ) +
      "</div>" +
      '<div class="preview-label">SCORM 1.2</div>' +
      "</div>" +

      '<div class="top-stats">' +
      '<div class="badge">Điểm <strong>' +
      state.score +
      " / " +
      totalPossibleScore() +
      "</strong></div>" +
      '<div class="badge">Trang <strong>' +
      (
        state.pageIndex +
        1
      ) +
      " / " +
      course.pages.length +
      "</strong></div>" +
      "</div>" +

      "</header>" +

      '<main class="main">' +
      '<div class="content">' +

      '<div class="page-info">' +
      '<div class="page-label">TRANG ' +
      (
        state.pageIndex +
        1
      ) +
      "</div>" +

      '<div class="page-heading">' +
      "<div>" +
      "<h1>" +
      escapeHtml(
        page.title
      ) +
      "</h1>" +
      '<div class="page-meta">' +
      page.quizzes.length +
      " câu hỏi tương tác</div>" +
      "</div>" +

      '<div class="video-name">' +
      escapeHtml(
        page.videoName ||
        ""
      ) +
      "</div>" +
      "</div>" +

      "</div>" +

      '<section class="video-shell">' +
      videoHtml +
      renderQuiz() +
      "</section>" +

      renderCourseProgress(
        completed
      ) +

      '<section class="controls">' +
      '<div class="controls-row">' +

      '<button id="play-button" class="play-button"' +
      (
        !page.videoSrc ||
        state.activeQuiz
          ? " disabled"
          : ""
      ) +
      ">" +
      (
        state.playing
          ? "❚❚"
          : "▶"
      ) +
      "</button>" +

      '<div class="time" id="time-display">' +
      formatTime(
        state.currentTime
      ) +
      " / " +
      formatTime(
        state.duration ||
        page.duration
      ) +
      "</div>" +

      '<input id="seek" class="seek" type="range" min="0" step="0.01" max="' +
      (
        state.duration > 0
          ? state.duration
          : page.duration ||
            1
      ) +
      '" value="' +
      state.currentTime +
      '"' +
      (
        !page.videoSrc ||
        state.activeQuiz
          ? " disabled"
          : ""
      ) +
      " />" +

      "</div>" +
      "</section>" +

      '<section class="navigation' +
      (
        theme.showPageNavigation ===
        false
          ? " is-hidden"
          : ""
      ) +
      '">' +

      '<button id="previous" class="nav-button"' +
      (
        state.pageIndex ===
          0 ||
        state.activeQuiz
          ? " disabled"
          : ""
      ) +
      ">‹ Trang trước</button>" +

      '<div class="progress-info">' +
      "Câu hỏi <strong>" +
      completedQuizCount(
        page
      ) +
      " / " +
      page.quizzes.length +
      "</strong> • Đã xem <strong id=\"watch-percent\">" +
      watch +
      "%</strong>" +
      notice +
      "</div>" +

      '<button id="next" class="nav-button next"' +
      (

/* =========================================================
   BUILD EXPORT COURSE
========================================================= */

async function buildExportCourse(
  project: ProjectDocument,
  zip: JSZip
) {
  const completion =
    loadCompletionSettings();

  const theme =
    loadProjectThemeForProject(
      project.id,
      project.title
    );

  const pages:
    ExportPage[] =
      [];

  const missingVideoPages:
    string[] =
      [];

  const mediaFiles:
    string[] =
      [];

  for (
    let index = 0;
    index <
    project.pages.length;
    index++
  ) {
    const page =
      project.pages[
        index
      ];

    const videoFile =
      await getVideo(
        page.id
      );

    let videoSrc:
      string | null =
      null;

    if (videoFile) {
      const extension =
        getVideoExtension(
          videoFile
        );

      const mediaPath =
        `media/page-${
          index + 1
        }${extension}`;

      /*
       * Video không nén lại.
       *
       * Video vốn đã là dữ liệu nén;
       * STORE giúp export nhanh hơn
       * và đỡ tốn RAM/CPU.
       */
      zip.file(
        mediaPath,
        videoFile,
        {
          binary: true,
          compression:
            "STORE",
        }
      );

      videoSrc =
        mediaPath;

      mediaFiles.push(
        mediaPath
      );
    } else if (
      page.videoName
    ) {
      /*
       * Metadata nói rằng trang từng
       * có video nhưng IndexedDB hiện
       * không còn file.
       */
      missingVideoPages.push(
        page.title
      );
    }

    pages.push({
      id:
        page.id,

      title:
        page.title,

      videoName:
        page.videoName,

      duration:
        page.duration,

      quizzes:
        page.quizzes.map(
          (quiz) => ({
            ...quiz,

            options:
              [...quiz.options],
          })
        ),

      videoSrc,
    });
  }

  const course:
    ExportCourse = {
      id:
        project.id,

      title:
        project.title,

      theme,

      completion: {
        passScorePercent:
          completion.passScorePercent,

        minWatchPercent:
          completion.minWatchPercent,

        requireAllQuizzes:
          completion.requireAllQuizzes,
      },

      pages,
    };

  return {
    course,
    missingVideoPages,
    mediaFiles,
  };
}

/* =========================================================
   PUBLIC EXPORT FUNCTION
========================================================= */

export async function exportCurrentProjectAsScorm(): Promise<ScormExportResult> {
  const project =
    useEditorStore.getState()
      .project;

  if (
    !project ||
    project.pages.length ===
      0
  ) {
    throw new Error(
      "Dự án chưa có nội dung để xuất."
    );
  }

  const zip =
    new JSZip();

  const {
    course,
    missingVideoPages,
    mediaFiles,
  } =
    await buildExportCourse(
      project,
      zip
    );

  const indexHtml =
    makeIndexHtml();

  const stylesCss =
    makeStylesCss();

  const playerJs =
    makePlayerJs();

  const courseDataJs =
    makeCourseDataScript(
      course
    );

  zip.file(
    "index.html",
    indexHtml
  );

  zip.file(
    "styles.css",
    stylesCss
  );

  zip.file(
    "player.js",
    playerJs
  );

  zip.file(
    "course-data.js",
    courseDataJs
  );

  const resourceFiles = [
    "index.html",
    "styles.css",
    "player.js",
    "course-data.js",
    ...mediaFiles,
  ];

  zip.file(
    "imsmanifest.xml",
    makeManifest(
      course,
      resourceFiles
    )
  );

  const blob =
    await zip.generateAsync({
      type:
        "blob",

      compression:
        "DEFLATE",

      compressionOptions: {
        level:
          6,
      },
    });

  const fileName =
    `${safeFileName(
      project.title
    )}-SCORM12.zip`;

  const url =
    URL.createObjectURL(
      blob
    );

  const anchor =
    document.createElement(
      "a"
    );

  anchor.href =
    url;

  anchor.download =
    fileName;

  document.body.appendChild(
    anchor
  );

  anchor.click();

  anchor.remove();

  window.setTimeout(
    () => {
      URL.revokeObjectURL(
        url
      );
    },
    1000
  );

  return {
    fileName,
    missingVideoPages,
  };
}
