/* =========================================================

========================================================= */

function safeFileName(
  value: string
) {
  const cleaned =
    value
      .normalize("NFD")
      .replace(
        /[\u0300-\u036f]/g,
        ""
      )
      .replace(
        /đ/g,
        "d"
      )
      .replace(
        /Đ/g,
        "D"
      )
      .replace(
        /[^a-zA-Z0-9-_]+/g,
        "-"
      )
      .replace(
        /-+/g,
        "-"
      )
      .replace(
        /^-|-$/g,
        ""
      );

  return (
    cleaned ||
    "elearning-course"
  );
}

function xmlEscape(
  value: string
) {
  return value
    .replace(
      /&/g,
      "&amp;"
    )
    .replace(
      /</g,
      "&lt;"
    )
    .replace(
      />/g,
      "&gt;"
    )
    .replace(
      /"/g,
      "&quot;"
    )
    .replace(
      /'/g,
      "&apos;"
    );
}

function getVideoExtension(
  file: File
) {
  const nameMatch =
    file.name.match(
      /\.[a-zA-Z0-9]{1,8}$/
    );

  if (nameMatch) {
    return nameMatch[0]
      .toLowerCase();
  }

  switch (file.type) {
    case "video/webm":
      return ".webm";

    case "video/ogg":
      return ".ogv";

    case "video/quicktime":
      return ".mov";

    default:
      return ".mp4";
  }
}

function makeCourseDataScript(
  course: ExportCourse
) {
  /*
   * Escape "<" để dữ liệu an toàn
   * khi được JavaScript đọc.
   */
  const json =
    JSON.stringify(
      course
    ).replace(
      /</g,
      "\\u003c"
    );

  return `window.COURSE_DATA = ${json};`;
}

/* =========================================================
   MANIFEST
========================================================= */

function makeManifest(
  course: ExportCourse,
  files: string[]
) {
  const courseTitle =
    xmlEscape(
      course.title
    );

  const identifier =
    `VLEARNING-${safeFileName(
      course.id
    )}`;

  const fileEntries =
    files
      .map(
        (file) =>
          `      <file href="${xmlEscape(
            file
          )}" />`
      )
      .join("\n");

  return `<?xml version="1.0" encoding="UTF-8"?>
<manifest
  identifier="${identifier}"
  version="1.0"
  xmlns="http://www.imsproject.org/xsd/imscp_rootv1p1p2"
  xmlns:adlcp="http://www.adlnet.org/xsd/adlcp_rootv1p2"
  xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
  xsi:schemaLocation="
    http://www.imsproject.org/xsd/imscp_rootv1p1p2
    imscp_rootv1p1p2.xsd
    http://www.adlnet.org/xsd/adlcp_rootv1p2
    adlcp_rootv1p2.xsd
  "
>
  <metadata>
    <schema>ADL SCORM</schema>
    <schemaversion>1.2</schemaversion>
  </metadata>

  <organizations default="ORG-1">
    <organization identifier="ORG-1">
      <title>${courseTitle}</title>

      <item
        identifier="ITEM-1"
        identifierref="RES-1"
      >
        <title>${courseTitle}</title>

        <adlcp:masteryscore>${course.completion.passScorePercent}</adlcp:masteryscore>
      </item>
    </organization>
  </organizations>

  <resources>
    <resource
      identifier="RES-1"
      type="webcontent"
      adlcp:scormtype="sco"
      href="index.html"
    >
${fileEntries}
    </resource>
  </resources>
</manifest>
`;
}

  function setToRanges(
    watchedSet
  ) {
    if (
      !watchedSet ||
      watchedSet.size ===
        0

      return "";

    var values =
      Array.from(

                value

              value >= 0

      values.length ===
        0

      return "";

    var ranges =
      [];

    var start =
      values[0];

    var end =
      values[0];

    for (
      var index = 1;
      index <
      values.length;
      index++
    ) {
      var value =
        values[index];

      if (
        value ===
        end + 1
      ) {
        end =
          value;

        continue;
      }

      ranges.push(
        start ===
          end
          ? String(start)
          : start +
            "-" +
            end
      );

      start =
        value;

      end =
        value;
    }

    ranges.push(
      start ===
        end
        ? String(start)
        : start +
          "-" +
          end
    );

    return ranges.join(
      ","
    );
  }

  function rangesToSet(
    value
  ) {
    var result =
      new Set();

    String(
      value || ""
    )
      .split(",")
      .forEach(
        function (
          token

          if (!token) {
            return;

          var parts =
            token.split("-");

          var start =
            Number(parts[0]);

          var end =
            parts.length >
              1
              ? Number(
                  parts[1]
                )
              : start;

          if (
            !Number.isInteger(
              start
            ) ||
            !Number.isInteger(
              end
            ) ||
            start < 0 ||
            end < start
          ) {
            return;
          }

          var safeEnd =
            Math.min(
              end,
              start +
                20000
            );

          for (
            var second = start;
            second <=
            safeEnd;
            second++
          ) {
            result.add(
              second
            );
          }
        }
      );

    return result;
  }

            restoreQuizSet(
              data.k
            );

          state.correctQuizIds =
            restoreQuizSet(
              data.c
            );

          state.completedPageIds =
            new Set();

          if (
            Array.isArray(
              data.d
            )
          ) {
            data.d.forEach(
              function (
                index
              ) {
                if (
                  Number.isInteger(
                    index
                  ) &&
                  course.pages[
                    index
                  ]
                ) {
                  state
                    .completedPageIds
                    .add(
                      course.pages[
                        index
                      ].id
                    );
                }
              }
            );
          }

          state.watched =
            {};

          if (
            data.w &&
            typeof data.w ===
              "object"
          ) {
            Object.keys(
              data.w
            ).forEach(
              function (
                indexText
              ) {
                var index =
                  Number(
                    indexText
                  );

                if (
                  Number.isInteger(
                    index
                  ) &&
                  course.pages[
                    index
                  ]
                ) {
                  state.watched[
                    course.pages[
                      index
                    ].id
                  ] =
                    rangesToSet(
                      data.w[
                        indexText
                      ]
                    );
                }
              }
            );
          }

          state.lastPositionByPage =
            {};

          if (
            data.l &&
            typeof data.l ===
              "object"
          ) {
            Object.keys(
              data.l
            ).forEach(
              function (
                indexText
              ) {
                var index =
                  Number(
                    indexText
                  );

                var value =
                  Number(
                    data.l[
                      indexText
                    ]
                  );

                if (
                  Number.isInteger(
                    index
                  ) &&
                  course.pages[
                    index
                  ] &&
                  Number.isFinite(
                    value
                  ) &&
                  value >= 0
                ) {
                  state
                    .lastPositionByPage[
                    course.pages[
                      index
                    ].id
                  ] =
                    value;
                }
              }
            );
          }

          var restoredPage =
            course.pages[
              state.pageIndex
            ];

          var storedTime =
            Number(
              data.t
            );

          if (
            Number.isFinite(
              storedTime
            ) &&
            storedTime >= 0
          ) {
            state.currentTime =
              storedTime;

            if (
              restoredPage
            ) {
              state
                .lastPositionByPage[
                restoredPage.id
              ] =
                storedTime;
            }
          }

          state.score =
            0;

          course.pages.forEach(
            function (
              page
            ) {
              page.quizzes.forEach(
                function (
                  quiz
                ) {
                  if (
                    state
                      .correctQuizIds
                      .has(
                        quiz.id
                      )
                  ) {
                    state.score +=
                      quiz.points;
                  }
                }
              );
            }
          );

          state.finished =
            data.f ===
            1;

          restored =
            true;
        }
      } catch (error) {
        console.warn(
          "Không đọc được SCORM suspend_data:",
          error
        );
      }
    }

    if (
      !restored &&
      scormStoredLocation
    ) {
      try {
        String(
          scormStoredLocation
        )
          .split(";")
          .forEach(
            function (
              token
            ) {
              var pair =
                token.split("=");

              if (
                pair[0] ===
                  "p"
              ) {
                var pageIndex =
                  Number(
                    pair[1]
                  );

                if (
                  Number.isInteger(
                    pageIndex
                  ) &&
                  pageIndex >=
                    0 &&
                  pageIndex <
                    course.pages.length
                ) {
                  state.pageIndex =
                    pageIndex;
                }
              }

              if (
                pair[0] ===
                  "t"
              ) {
                var time =
                  Number(
                    pair[1]
                  );

                if (
                  Number.isFinite(
                    time
                  ) &&
                  time >= 0
                ) {
                  state.currentTime =
                    time;
                }
              }
            }
          );
      } catch (error) {}
    }

    var activePage =
      currentPage();

    if (
      activePage &&
      state.currentTime >
        0
    ) {
      state
        .lastPositionByPage[
        activePage.id
      ] =
        state.currentTime;
    }

    return restored;
  }

  function scormPersistProgress(
    force
  ) {
    if (
      !scormApi ||
      !scormInitialized ||
      typeof state ===
        "undefined"
    ) {
      return;
    }

    var now =
      Date.now();

    if (
      !force &&
      now -
        scormLastPersistAt <
        1500
    ) {
      return;
    }

    scormLastPersistAt =
      now;

    try {
      var page =
        currentPage();

      var video =
        document.getElementById(
          "course-video"
        );

      if (
        page &&
        video &&
        Number.isFinite(
          video.currentTime
        )
      ) {
        state.currentTime =
          video.currentTime;

        state
          .lastPositionByPage[
          page.id
        ] =
          video.currentTime;
      }

      scormApi.LMSSetValue(
        "cmi.core.lesson_location",
        "p=" +
          state.pageIndex +
          ";t=" +
          (
            Math.round(
              state.currentTime *
              10
            ) /
            10
          )
      );

      var suspendData =
        makeSuspendData();

      if (
        suspendData
      ) {
        scormApi.LMSSetValue(
          "cmi.suspend_data",
          suspendData
        );
      }

      scormApi.LMSSetValue(
        "cmi.core.exit",
        state.finished
          ? ""
          : "suspend"
      );

      scormApi.LMSCommit("");
    } catch (error) {
      console.warn(
        "SCORM persist failed",
        error
      );
    }
  }

  function watchPercent(
    pageId,
    duration
  ) {
    if (
      !Number.isFinite(
        duration
      ) ||
      duration <= 0
    ) {
      return 0;
    }

    var total =
      Math.max(
        1,
        Math.ceil(
          duration
        )
      );

    var watched =
      getWatchSet(
        pageId
      );

    return Math.min(
      100,
      Math.round(
        watched.size /
          total *
          100
      )
    );
  }

  function recordWatch(
    pageId,
    time,
    duration
  ) {
    var previous =
      state.lastPlaybackTime;

    if (
      previous === null
    ) {
      state.lastPlaybackTime =
        time;

      return;
    }

    var delta =
      time -
      previous;

    if (
      delta > 0 &&
      delta <= 2.5
    ) {
      var watched =
        getWatchSet(
          pageId
        );

      var total =
        Math.max(
          1,
          Math.ceil(
            duration
          )
        );

      var start =
        Math.max(
          0,
          Math.floor(
            previous
          )
        );

      var end =
        Math.min(
          total - 1,
          Math.floor(
            time
          )
        );

      for (
        var second = start;
        second <= end;
        second++
      ) {
        watched.add(
          second
        );
      }
    }

    state.lastPlaybackTime =
      time;
  }

  function firstUnwatchedSecond(
    pageId,
    duration
  ) {
    var watched =
      getWatchSet(
        pageId
      );

    var total =
      Math.max(
        1,
        Math.ceil(
          duration
        )
      );

    for (
      var second = 0;
      second < total;
      second++
    ) {
      if (
        !watched.has(
          second
        )
      ) {
        return second;
      }
    }

    return 0;
  }

      "";

    state.matchingSelections =

    state.orderingSelections =
      quizType(quiz) ===
        "ordering"
        ? orderingItems(quiz)

      quizType(quiz) ===

  }

  function quizType(
    quiz
  ) {
    if (
      quiz &&
      quiz.type ===
        "true-false"
    ) {
      return "true-false";
    }

    if (
      quiz &&
      quiz.type ===
        "multiple"
    ) {
      return "multiple";
    }

    if (
      quiz &&
      quiz.type ===
        "short-answer"
    ) {
      return "short-answer";
    }

    if (
      quiz &&
      quiz.type ===
        "matching"
    ) {
      return "matching";
    }

    if (
      quiz &&
      quiz.type ===
        "ordering"
    ) {
      return "ordering";
    }

    if (
      quiz &&
      quiz.type ===
        "classification"
    ) {
      return "classification";
    }

    return "multiple-choice";
  }

  function quizCorrectIndexes(
    quiz
  ) {
    if (
      quizType(quiz) !==
      "multiple"
    ) {
      return [
        quiz.correctIndex
      ];
    }

    var source =
      Array.isArray(
        quiz.correctIndexes
      )
        ? quiz.correctIndexes
        : [];

    var result = [];

    source.forEach(
      function (index) {
        if (
          Number.isInteger(index) &&
          index >= 0 &&
          index <= 3 &&
          result.indexOf(index) ===
            -1
        ) {
          result.push(index);
        }
      }
    );

    result.sort(
      function (a, b) {
        return a - b;
      }
    );

    if (result.length > 0) {
      return result;
    }

    return [
      Number.isInteger(
        quiz.correctIndex
      )
        ? quiz.correctIndex
        : 0
    ];
  }

  function sameIndexes(
    first,
    second
  ) {
    if (
      first.length !==
      second.length
    ) {
      return false;
    }

    for (
      var index = 0;
      index < first.length;
      index++
    ) {
      if (
        first[index] !==
        second[index]
      ) {
        return false;
      }
    }

    return true;
  }

  function answerQuiz(
    index
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult
    ) {
      return;
    }

    if (
      (
        quizType(quiz) ===
          "short-answer" ||
        quizType(quiz) ===
          "matching" ||
        quizType(quiz) ===
          "ordering" ||
        quizType(quiz) ===
          "classification"
      )
    ) {
      return;
    }

    if (
      quizType(quiz) ===
      "multiple"
    ) {
      toggleMultipleAnswer(
        index
      );

      return;
    }

    state.selectedAnswer =
      index;

    var correct =
      index ===
      quiz.correctIndex;

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function toggleMultipleAnswer(
    index
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "multiple"
    ) {
      return;
    }

    var existing =
      state.selectedAnswers
        .indexOf(index);

    if (existing >= 0) {
      state.selectedAnswers
        .splice(
          existing,
          1
        );
    } else {
      state.selectedAnswers
        .push(index);

      state.selectedAnswers
        .sort(
          function (a, b) {
            return a - b;
          }
        );
    }

    render();
  }

  function submitMultipleAnswer() {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "multiple" ||
      state.selectedAnswers.length ===
        0
    ) {
      return;
    }

    var selected =
      state.selectedAnswers
        .slice()
        .sort(
          function (a, b) {
            return a - b;
          }
        );

    var correctIndexes =
      quizCorrectIndexes(
        quiz
      );

    var correct =
      sameIndexes(
        selected,
        correctIndexes
      );

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function moveOrderingItem(
    position,
    direction
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "ordering"
    ) {
      return;
    }

    var target =
      position +
      direction;

    if (
      target < 0 ||
      target >=
        state.orderingSelections.length
    ) {
      return;
    }

    var temp =
      state.orderingSelections[
        position
      ];

    state.orderingSelections[
      position
    ] =
      state.orderingSelections[
        target
      ];

    state.orderingSelections[
      target
    ] =
      temp;

    render();
  }

  function submitOrderingAnswer() {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "ordering"
    ) {
      return;

      orderingItems(
        quiz
      );

    if (
      items.length < 2 ||
      state.orderingSelections.length !==

    ) {
      return;
    }

    var correct =
      state.orderingSelections
        .every(
          function (
            sourceIndex,
            position
          ) {
            return (
              sourceIndex ===
              position
            );
          }
        );

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function classificationCategories(

      quiz &&

        quiz.classificationCategories
      )
        ? quiz.classificationCategories
        : [];

          ? ""
          : source[0]

    ];

  function classificationItems(
    quiz
  ) {
    var items =
      quiz &&
      Array.isArray(
        quiz.classificationItems
      )
        ? quiz.classificationItems

    return items
      .map(

          return {
            text: String(
              item &&
              item.text != null
                ? item.text
                : ""
            ).trim(),
            categoryIndex:
              item &&
              Number(item.categoryIndex) === 1
                ? 1
                : 0
          };

      )
      .filter(
        function (item) {
          return Boolean(
            item.text
          );
        }
      );
  }

  function changeClassificationSelection(
    itemIndex,
    categoryIndex
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "classification"
    ) {
      return;
    }

    var items =
      classificationItems(
        quiz
      );

    if (
      itemIndex < 0 ||
      itemIndex >=
        items.length
    ) {
      return;
    }

    var normalizedCategory =
      categoryIndex === 0 ||
      categoryIndex === 1
        ? categoryIndex
        : -1;

    while (
      state.classificationSelections.length <
      items.length
    ) {
      state.classificationSelections.push(
        -1
      );
    }

    state.classificationSelections[
      itemIndex
    ] =
      normalizedCategory;

    render();
  }

  function submitClassificationAnswer() {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "classification"
    ) {
      return;
    }

    var categories =
      classificationCategories(
        quiz
      );

    var items =
      classificationItems(
        quiz
      );

    if (
      !categories[0] ||
      !categories[1] ||
      items.length < 2 ||
      state.classificationSelections.length !==
        items.length ||
      state.classificationSelections.some(
        function (value) {
          return value < 0;
        }
      )
    ) {
      return;
    }

    var correct =
      state.classificationSelections
        .every(
          function (
            categoryIndex,
            itemIndex
          ) {
            return (
              categoryIndex ===
              items[itemIndex]
                .categoryIndex
            );
          }
        );

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function matchingPairs(
    quiz
  ) {
    var pairs =
      quiz &&
      Array.isArray(
        quiz.matchingPairs
      )
        ? quiz.matchingPairs
        : [];

    return pairs
      .map(
        function (pair) {
          return {
            left: String(
              pair &&
              pair.left != null
                ? pair.left
                : ""
            ).trim(),
            right: String(
              pair &&
              pair.right != null
                ? pair.right
                : ""
            ).trim(),
          };
        }
      )
      .filter(
        function (pair) {
          return (
            pair.left &&
            pair.right
          );
        }
      );
  }

  function changeMatchingSelection(
    leftIndex,
    rightIndex
  ) {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "matching"
    ) {
      return;
    }

    var pairs =
      matchingPairs(quiz);

    var next =
      [];

    for (
      var index = 0;
      index < pairs.length;
      index++
    ) {
      next[index] =
        typeof state.matchingSelections[index] ===
        "number"
          ? state.matchingSelections[index]
          : -1;
    }

    next[leftIndex] =
      rightIndex;

    state.matchingSelections =
      next;

    render();
  }

  function submitMatchingAnswer() {

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "matching"
    ) {
      return;
    }

    var pairs =
      matchingPairs(quiz);

    if (
      pairs.length < 2 ||
      state.matchingSelections.length !==
        pairs.length ||
      state.matchingSelections.some(
        function (value) {
          return value < 0;
        }
      )
    ) {
      return;
    }

    var correct =
      state.matchingSelections.every(
        function (
          value,
          index
        ) {
          return value === index;
        }
      );

    if (correct) {
      state.quizResult =
        "correct";

      if (
        !state.correctQuizIds.has(
          quiz.id
        )
      ) {
        state.correctQuizIds.add(
          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

  function normalizeShortAnswerValue(
    value
  ) {
    return String(
      value || ""
    )
      .normalize("NFKC")
      .trim()
      .replace(
        /\s+/g,
        " "
      )
      .toLocaleLowerCase(
        "vi"
      );
  }

  function isShortAnswerCorrect(
    answer,
    acceptedAnswers
  ) {
    var normalizedAnswer =
      normalizeShortAnswerValue(
        answer
      );

    if (!normalizedAnswer) {
      return false;
    }

    var accepted =
      Array.isArray(
        acceptedAnswers
      )
        ? acceptedAnswers
        : [];

    return accepted.some(
      function (
        item
      ) {
        return (
          normalizeShortAnswerValue(
            item
          ) ===
          normalizedAnswer
        );
      }
    );
  }

  function submitShortAnswer() {
    var quiz =
      state.activeQuiz;

    if (
      !quiz ||
      state.quizResult ||
      quizType(quiz) !==
        "short-answer" ||
      !String(
        state.shortAnswer ||
        ""
      ).trim()
    ) {
      return;
    }

    var correct =
      isShortAnswerCorrect(
        state.shortAnswer,
        quiz.acceptedAnswers
      );

    if (correct) {
      state.quizResult =
        "correct";

      if (

          quiz.id
        );

        state.score +=
          quiz.points;
      }
    } else {
      state.quizResult =
        "wrong";
    }

    render();
  }

    /* ===================================================

    /* ===================================================
       CHECK WHETHER ALL REQUIRED QUIZZES ARE DONE
    =================================================== */

    function requiredQuizzesDone() {
      if (
        !course.completion
          .requireAllQuizzes
      ) {
        return true;
      }

      return page.quizzes.every(
        function (quiz) {
          return state
            .answeredQuizIds
            .has(
              quiz.id
            );
        }
      );
    }

    /* ===================================================
       COMPLETE PAGE WITHOUT CALLING render()
    =================================================== */

    function checkPageCompletion() {
      /*
       * Đã hoàn thành rồi.
       */
      if (
        state.completedPageIds.has(
          page.id
        )
      ) {
        updateCompletionStateUi();

        return true;
      }

      /*
       * Quiz bắt buộc nhưng chưa làm hết.
       */
      if (
        !requiredQuizzesDone()
      ) {
        updateCompletionStateUi();

        return false;
      }

      var totalDuration =
        video.duration;

      if (
        !Number.isFinite(
          totalDuration
        ) ||
        totalDuration <= 0
      ) {
        totalDuration =
          state.duration ||
          page.duration;
      }

      var percent =
        watchPercent(
          page.id,
          totalDuration
        );

      /*
       * Chưa xem đủ tỷ lệ yêu cầu.
       */
      if (
        percent <
        course.completion
          .minWatchPercent
      ) {
        updateCompletionStateUi();

        return false;
      }

      /*
       * Đạt toàn bộ điều kiện.
       *
       * QUAN TRỌNG:
       * Không gọi completeCurrentPage()
       * vì function đó render lại player,
       * có thể làm video bị gián đoạn.
       */
      state
        .completedPageIds
        .add(
          page.id
        );

      state.completionNotice =
        "";

      scormPersistProgress(
        true
      );

      updateCompletionStateUi();

      return true;
    }

    /* ===================================================
       METADATA
    =================================================== */

    video.addEventListener(
      "loadedmetadata",
      function () {
        state.duration =
          video.duration;

        var savedPosition =
          Number(
            state
              .lastPositionByPage[
              page.id
            ]
          );

        if (
          !Number.isFinite(
            savedPosition
          )
        ) {
          savedPosition =
            state.currentTime;
        }

        if (
          savedPosition > 0 &&
          savedPosition <
            video.duration
        ) {
          video.currentTime =
            Math.min(
              savedPosition,
              Math.max(
                0,
                video.duration -
                  0.05
              )
            );

          state.currentTime =
            video.currentTime;
        }

        updateControls();

        /*
         * Hỗ trợ cả cấu hình
         * minWatchPercent = 0.
         */
        checkPageCompletion();
      }
    );

    /* ===================================================
       PLAY
    =================================================== */

    video.addEventListener(
      "play",
      function () {
        state.started =
          true;

        state.playing =
          true;

        state.lastPlaybackTime =
          video.currentTime;

        updateControls();
      }
    );

    /* ===================================================
       PAUSE
    =================================================== */

    video.addEventListener(
      "pause",
      function () {
        state.playing =
          false;

        state.lastPlaybackTime =
          null;

        state
          .lastPositionByPage[
          page.id
        ] =
          video.currentTime;

        scormPersistProgress(
          true
        );

        updateControls();
      }
    );

    /* ===================================================
       TIME UPDATE
    =================================================== */

    video.addEventListener(
      "timeupdate",
      function () {
        state.currentTime =
          video.currentTime;

        state
          .lastPositionByPage[
          page.id
        ] =
          video.currentTime;

        scormPersistProgress(
          false
        );

        /*
         * Chỉ ghi nhận thời gian xem thật
         * khi video đang phát.
         */
        if (
          state.started &&
          !video.paused
        ) {
          recordWatch(
            page.id,
            video.currentTime,
            video.duration
          );
        }

        updateControls();

        /*
         * Đây là thay đổi chính của V9.4:
         *
         * mỗi lần thời gian xem tăng,
         * kiểm tra ngay điều kiện hoàn thành.
         *
         * Ví dụ yêu cầu 70%:
         * vừa đạt 70% thì Trang tiếp theo
         * được mở ngay.
         */
        checkPageCompletion();

        /*
         * Chưa Play hoặc đang hiện quiz
         * thì không tìm quiz mới.
         */
        if (
          !state.started ||
          state.activeQuiz
        ) {
          return;
        }

        var due =
          sortedQuizzes(
            page
          ).find(
            function (quiz) {
              return (
                !handledQuiz(
                  quiz.id
                ) &&
                quiz.time <=
                  video.currentTime
              );
            }
          );

        if (due) {
          showQuiz(
            due
          );
        }
      }
    );

    /* ===================================================
       VIDEO ENDED
    =================================================== */

    video.addEventListener(
      "ended",
      function () {
        state.playing =
          false;

        /*
         * Ghi nhận nốt đoạn cuối.
         */
        recordWatch(
          page.id,
          video.duration,
          video.duration
        );

        state.lastPlaybackTime =
          null;

        state
          .lastPositionByPage[
          page.id
        ] =
          video.duration;

        scormPersistProgress(
          true
        );

        /*
         * Nếu quiz là bắt buộc thì
         * không được bỏ sót quiz.
         */
        if (
          course.completion
            .requireAllQuizzes
        ) {
          var unanswered =
            sortedQuizzes(
              page
            ).find(
              function (
                quiz
              ) {
                return !state
                  .answeredQuizIds
                  .has(
                    quiz.id
                  );
              }
            );

          if (
            unanswered
          ) {
            video.currentTime =
              unanswered.time;

            state.currentTime =
              unanswered.time;

            showQuiz(
              unanswered
            );

            return;
          }
        }

        var percent =
          watchPercent(
            page.id,
            video.duration
          );

        /*
         * Chạy đến cuối nhưng vẫn chưa đủ
         * tỷ lệ xem thật, ví dụ do tua.
         */
        if (
          percent <
          course.completion
            .minWatchPercent
        ) {
          var missing =
            firstUnwatchedSecond(
              page.id,
              video.duration
            );

          state.completionNotice =
            "Bạn mới xem " +
            percent +
            "% video. Cần tối thiểu " +
            course.completion
              .minWatchPercent +
            "% để hoàn thành trang.";

          video.currentTime =
            missing;

          state.currentTime =
            missing;

          updateControls();

          updateCompletionStateUi();

          return;
        }

        /*
         * Nếu đủ điều kiện thì chỉ cập nhật
         * state/UI, không render lại video.
         */
        checkPageCompletion();
      }
    );

    /*
     * Đồng bộ trạng thái ngay lần đầu.
     */
    updateCompletionStateUi();
  }

  function seekVideo(
    value
  ) {
    var page =
      currentPage();

    var video =
      document.getElementById(
        "course-video"
      );

    if (
      !page ||
      !video ||
      state.activeQuiz
    ) {
      return;
    }

    var requested =
      Number(value);

    state.started =
      true;

    state.lastPlaybackTime =
      null;

    if (
      course.completion
        .requireAllQuizzes &&
      requested >
        state.currentTime
    ) {
      var blocking =
        sortedQuizzes(
          page
        ).find(
          function (
            quiz
          ) {
            return (
              !handledQuiz(
                quiz.id
              ) &&
              quiz.time >
                state.currentTime &&
              quiz.time <=
                requested
            );
          }
        );

      if (
        blocking
      ) {
        video.currentTime =
          blocking.time;

        state.currentTime =
          blocking.time;

        showQuiz(
          blocking
        );

        return;
      }
    }

    video.currentTime =
      requested;

    state.currentTime =
      requested;

    state
      .lastPositionByPage[
      page.id
    ] =
      requested;

    scormPersistProgress(
      true
    );

    updateControls();
  }

  /* =====================================================
     NO VIDEO PAGE
  ===================================================== */

  function handleNoVideoPage() {
    var page =
      currentPage();

    if (!page) {
      return;
    }

    if (
      page.quizzes.length ===
      0
    ) {
      completeCurrentPage();

      return;
    }

    if (
      !course.completion
        .requireAllQuizzes
    ) {
      completeCurrentPage();

      return;
    }

    var next =
      page.quizzes.find(
        function (
          quiz
        ) {
          return !state
            .answeredQuizIds
            .has(
              quiz.id
            );
        }
      );

    if (next) {
      showQuiz(
        next
      );

      return;
    }

    completeCurrentPage();
  }

  /* =====================================================
     V12.1E LESSON OUTLINE
  ===================================================== */

  function isPageUnlocked(
    targetIndex
  ) {
    var targetPage =
      course.pages[
        targetIndex
      ];

    if (!targetPage) {
      return false;
    }

    if (
      targetIndex ===
      state.pageIndex
    ) {
      return true;
    }

    if (
      state.completedPageIds
        .has(
          targetPage.id
        )
    ) {
      return true;
    }

    if (
      targetIndex <
      state.pageIndex
    ) {
      return false;
    }

    for (
      var index = 0;
      index <
      targetIndex;
      index++
    ) {
      if (
        !state.completedPageIds
          .has(
            course.pages[
              index
            ].id
          )
      ) {
        return false;
      }
    }

    return true;
  }

  function goToOutlinePage(
    targetIndex
  ) {
    if (
      state.activeQuiz ||
      targetIndex ===
        state.pageIndex ||
      !isPageUnlocked(
        targetIndex
      )
    ) {
      return;
    }

    scormPersistProgress(
      true
    );

    var video =
      document.getElementById(
        "course-video"
      );

    if (
      video &&
      typeof video.pause ===
        "function"
    ) {
      video.pause();
    }

    state.pageIndex =
      targetIndex;

    resetPageState();

    scormPersistProgress(
      true
    );

    render();
  }

  function renderLessonOutline() {
    var completedCount =
      0;

    course.pages.forEach(
      function (
        outlinePage
      ) {
        if (
          state.completedPageIds
            .has(
              outlinePage.id
            )
        ) {
          completedCount++;
        }
      }
    );

    var items =
      course.pages.map(
        function (
          outlinePage,
          outlineIndex
        ) {
          var isCurrent =
            outlineIndex ===
            state.pageIndex;

          var isCompleted =
            state.completedPageIds
              .has(
                outlinePage.id
              );

          var unlocked =
            isPageUnlocked(
              outlineIndex
            );

          var disabled =
            !unlocked ||
            Boolean(
              state.activeQuiz
            );

          var stateText =
            isCurrent
              ? "Đang học"
              : isCompleted
                ? "Đã hoàn thành"
                : unlocked
                  ? "Có thể mở"
                  : "Chưa mở khóa";

          var numberHtml =
            isCompleted
              ? "✓"
              : unlocked
                ? String(
                    outlineIndex +
                    1
                  )
                : "🔒";

          return (
            '<button type="button" class="lesson-outline-item' +
            (
              isCurrent
                ? " current"
                : ""
            ) +
            (
              isCompleted
                ? " completed"
                : ""
            ) +
            (
              !unlocked
                ? " locked"
                : ""
            ) +
            '" data-outline-page="' +
            outlineIndex +
            '"' +
            (
              disabled
                ? " disabled"
                : ""
            ) +
            '>' +

            '<div class="lesson-outline-row">' +

            '<div class="lesson-outline-number">' +
            numberHtml +
            "</div>" +

            '<div class="lesson-outline-copy">' +

            '<div class="lesson-outline-page-title">' +
            escapeHtml(
              outlinePage.title
            ) +
            "</div>" +

            '<div class="lesson-outline-page-meta">' +
            outlinePage.quizzes.length +
            " câu hỏi • " +
            stateText +
            "</div>" +

            "</div>" +
            "</div>" +
            "</button>"
          );
        }
      )
      .join("");

    return (
      '<aside class="lesson-outline" id="lesson-outline">' +

      '<div class="lesson-outline-head">' +
      '<div class="lesson-outline-title">' +
      '<span class="lesson-outline-title-icon">☷</span>' +
      "Mục lục bài học" +
      "</div>" +
      '<div class="lesson-outline-summary">' +
      completedCount +
      " / " +
      course.pages.length +
      " nội dung đã hoàn thành" +
      "</div>" +
      "</div>" +

      '<div class="lesson-outline-list">' +
      items +
      "</div>" +

      '<div class="lesson-outline-foot">' +
      "Hoàn thành nội dung hiện tại để mở khóa nội dung tiếp theo." +
      "</div>" +

      "</aside>"
    );
  }

  function previousPage() {
    if (
      state.pageIndex <= 0 ||
      state.activeQuiz
    ) {
      return;
    }

    scormPersistProgress(
      true
    );

    state.pageIndex--;

    resetPageState();

    scormPersistProgress(
      true
    );

    render();
  }

  function nextPage() {
    var page =
      currentPage();

    if (
      !page ||
      !state
        .completedPageIds
        .has(
          page.id
        ) ||
      state.activeQuiz
    ) {
      return;
    }

    if (
      state.pageIndex <
      course.pages.length -
        1
    ) {
      scormPersistProgress(
        true
      );

      state.pageIndex++;

      resetPageState();

      scormPersistProgress(
        true
      );

      render();

      return;
    }

    finishCourse();
  }

  function resetPageState() {
    var page =
      currentPage();

    var savedPosition =
      page
        ? Number(
            state
              .lastPositionByPage[
              page.id
            ]
          )
        : 0;

    state.currentTime =
      Number.isFinite(
        savedPosition
      )
        ? Math.max(
            0,
            savedPosition
          )
        : 0;

    state.duration =
      0;

    state.playing =
      false;

    state.started =
      false;

    state.activeQuiz =
      null;

    state.selectedAnswer =
      null;

    state.selectedAnswers =
      [];

    state.shortAnswer =
      "";

    state.matchingSelections =
      [];

    state.orderingSelections =
      [];

    state.classificationSelections =
      [];

    state.quizResult =
      null;

    state.lastPlaybackTime =
      null;

    state.completionNotice =
      "";
  }

  /* =====================================================
     FINISH
  ===================================================== */

  function finishCourse() {
    state.finished =
      true;

    var possible =
      totalPossibleScore();

    var percentage =
      possible > 0
        ? Math.round(
            state.score /
              possible *
              100
          )
        : 100;

    var passed =
      percentage >=
      course.completion
        .passScorePercent;

    scormReport(
      percentage,
      passed
    );

    scormPersistProgress(
      true
    );

    render();
  }

    state.currentTime =

    state.duration =
      0;

    state.playing =
      false;

    state.started =
      false;

      null;

    state.selectedAnswer =
      null;

    state.selectedAnswers =
      [];

    state.shortAnswer =
      "";

    state.matchingSelections =
      [];

    state.orderingSelections =
      [];

    state.classificationSelections =
      [];

    state.quizResult =
      null;

    state.answeredQuizIds =
      new Set();

    state.skippedQuizIds =
      new Set();

    state.correctQuizIds =
      new Set();

    state.completedPageIds =
      new Set();

    state.watched =
      {};

    state.lastPositionByPage =
      {};

    state.lastPlaybackTime =
      null;

    state.score =
      0;

    state.completionNotice =
      "";

    state.finished =
      false;

    state.introOpen =
      true;

    state.showCertificate =
      false;

    state.certificateLearnerName =

    try {

      ) {

  /* =====================================================
     UPDATE SMALL UI PARTS
  ===================================================== */

  function updateControls() {
    var page =
      currentPage();

    if (!page) {
      return;
    }

    var play =
      document.getElementById(
        "play-button"
      );

    if (play) {
      play.textContent =
        state.playing
          ? "❚❚"
          : "▶";
    }

    var time =
      document.getElementById(
        "time-display"
      );

    if (time) {
      time.textContent =
        formatTime(
          state.currentTime
        ) +
        " / " +
        formatTime(
          state.duration
        );
    }

    var seek =
      document.getElementById(
        "seek"
      );

    if (seek) {
      seek.max =
        String(
          state.duration > 0
            ? state.duration
            : 1
        );

      seek.value =
        String(
          Math.min(
            state.currentTime,
            state.duration > 0
              ? state.duration

        classificationItems(

      .replace(
        /</g,
        "&lt;"
      )
      .replace(
        />/g,
        "&gt;"
      )
      .replace(
        /"/g,
        "&quot;"
      )
      .replace(

        "&#039;"
      );
  }

  function formatReportDuration(
    seconds
  ) {
    var safeSeconds =
      Math.max(
        0,
        Math.floor(
          Number.isFinite(
            seconds
          )
            ? seconds
            : 0
        )
      );

    var hours =
      Math.floor(
        safeSeconds /
          3600
      );

    var minutes =
      Math.floor(
        (
          safeSeconds %
          3600
        ) /
          60
      );

    var remainingSeconds =
      safeSeconds %
      60;

    if (
      hours > 0
    ) {
      return (
        hours +
        " giờ " +
        minutes +
        " phút"
      );
    }

    if (
      minutes > 0
    ) {
      return (
        minutes +
        " phút " +
        remainingSeconds +
        " giây"
      );
    }

    return (
      remainingSeconds +
      " giây"
    );
  }

  function pageQuizStats(
    page
  ) {
    var answered =
      0;

    var correct =
      0;

    var skipped =
      0;

    var earnedScore =
      0;

    var possibleScore =
      0;

    page.quizzes.forEach(
      function (
        quiz
      ) {
        possibleScore +=
          quiz.points;

        if (
          state
            .answeredQuizIds
            .has(
              quiz.id
            )
        ) {
          answered++;
        }

        if (
          state
            .correctQuizIds
            .has(
              quiz.id
            )
        ) {
          correct++;

          earnedScore +=
            quiz.points;
        }

        if (
          state
            .skippedQuizIds
            .has(
              quiz.id
            )
        ) {
          skipped++;
        }
      }
    );

    return {
      answered:
        answered,

      correct:
        correct,

      wrong:
        Math.max(
          0,
          answered -
            correct
        ),

      skipped:
        skipped,

      earnedScore:
        earnedScore,

      possibleScore:
        possibleScore
    };
  }

            page.duration
          ),
          formatReportDuration(
            watchedSeconds
          )
        ]);
      }
    );

      "\ufeff" +
      rows
        .map(
          function (

            return row
              .map(
                csvCell
              )

          }
        )
        .join("\r\n");

      new Blob(
        [
          csv
        ],
        {
          type:
            "text/csv;charset=utf-8;"
        }
      );

      URL.createObjectURL(
        blob
      );

    var link =
      document.createElement(
        "a"
      );

    link.href =
      url;

    link.download =
      safeReportFileName(
        displayCourseTitle()
      ) +
      "-Bao-cao-ket-qua-SCORM.csv";

    document.body.appendChild(
      link
    );

    link.click();

    link.remove();

    window.setTimeout(
      function () {
        URL.revokeObjectURL(
          url
        );
      },
      1000
    );
  }

  function certificateEscapeHtml(value) {
    return String(value == null ? "" : value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  }

      Math.max(
        0,
        answeredCount -

  function formatCourseDuration(
    seconds
  ) {
    var safeSeconds =
      Math.max(
        0,
        Math.floor(
          Number.isFinite(
            seconds
          )
            ? seconds
            : 0
        )
      );

    if (
      safeSeconds <=
      0
    ) {
      return "Chưa xác định";
    }

    var hours =
      Math.floor(
        safeSeconds /
          3600
      );

    var minutes =
      Math.ceil(
        (
          safeSeconds %
          3600
        ) /
          60
      );

    if (
      hours >
      0
    ) {
      return (
        hours +
        " giờ " +
        minutes +
        " phút"
      );
    }

    return (
      Math.max(
        1,
        minutes
      ) +
      " phút"
    );
  }

      .querySelectorAll(
        "[data-classify-drag-item]"
      )
      .forEach(
        function (element) {
          element.addEventListener(
            "dragstart",
            function (event) {
              if (
                state.quizResult ||
                !event.dataTransfer
              ) {
                return;
              }

              event.dataTransfer.setData(
                "text/plain",
                String(
                  element.getAttribute(
                    "data-classify-drag-item"
                  )
                )
              );
            }
          );
        }
      );

    document
      .querySelectorAll(
        "[data-classify-drop-category]"
      )
      .forEach(
        function (element) {
          element.addEventListener(
            "dragover",
            function (event) {
              if (
                !state.quizResult
              ) {
                event.preventDefault();
              }
            }
          );

          element.addEventListener(
            "drop",
            function (event) {
              if (
                state.quizResult ||
                !event.dataTransfer
              ) {
                return;
              }

              event.preventDefault();

              changeClassificationSelection(
                Number(
                  event.dataTransfer.getData(
                    "text/plain"
                  )
                ),
                Number(
                  element.getAttribute(
                    "data-classify-drop-category"
                  )
                )
              );
            }
          );
        }
      );

    var submitClassificationButton =
      document.getElementById(
        "submit-classification"
      );

    if (
      submitClassificationButton
    ) {
      submitClassificationButton.addEventListener(
        "click",
        submitClassificationAnswer
      );
    }

    document
      .querySelectorAll(
        "[data-match-left]"
      )
      .forEach(
        function (element) {
          element.addEventListener(
            "change",
            function () {
              changeMatchingSelection(
                Number(
                  element.getAttribute(
                    "data-match-left"
                  )
                ),
                Number(
                  element.value
                )
              );
            }
          );
        }
      );

    var submitMatchingButton =
      document.getElementById(
        "submit-matching"
      );

    if (
      submitMatchingButton
    ) {
      submitMatchingButton.addEventListener(
        "click",
        submitMatchingAnswer
      );
    }

    var shortAnswerInput =
      document.getElementById(
        "short-answer-input"
      );

    if (
      shortAnswerInput
    ) {
      shortAnswerInput.addEventListener(
        "input",
        function (
          event
        ) {
          state.shortAnswer =
            event.target.value;

          var submitShortButton =
            document.getElementById(
              "submit-short-answer"
            );

          if (
            submitShortButton
          ) {
            submitShortButton.disabled =
              !String(
                state.shortAnswer ||
                ""
              ).trim();
          }
        }
      );

      shortAnswerInput.addEventListener(
        "keydown",
        function (
          event
        ) {
          if (
            event.key ===
              "Enter" &&
            String(
              state.shortAnswer ||
              ""
            ).trim()
          ) {
            submitShortAnswer();
          }
        }
      );
    }

    var submitShortButton =
      document.getElementById(
        "submit-short-answer"
      );

    if (
      submitShortButton
    ) {
      submitShortButton.addEventListener(
        "click",
        submitShortAnswer
      );
    }

    var continueButton =
      document.getElementById(
        "continue-quiz"
      );

    if (
      continueButton
    ) {
      continueButton.addEventListener(
        "click",
        continueAfterQuiz
      );
    }

    var skipButton =
      document.getElementById(
        "skip-quiz"
      );

    if (skipButton) {
      skipButton.addEventListener(
        "click",
        skipQuiz
      );
    }

    if (
      page.videoSrc
    ) {
      bindVideoEvents();
    } else {
      window.setTimeout(
        handleNoVideoPage,
        0
      );
    }
  }

  document.addEventListener(
    "fullscreenchange",
    updateFullscreenButton
  );

  restoreScormProgress();

  render();
})();
`;
}

/* =========================================================
   BUILD EXPORT COURSE
========================================================= */

async function buildExportCourse(
  project: ProjectDocument,
  zip: JSZip
) {
  const completion =
    loadCompletionSettings();

  const theme =
    loadProjectThemeForProject(
      project.id,
      project.title
    );

  const pages:
    ExportPage[] =
      [];

  const missingVideoPages:
    string[] =
      [];

  const mediaFiles:
    string[] =
      [];

  for (
    let index = 0;
    index <
    project.pages.length;
    index++
  ) {
    const page =
      project.pages[
        index
      ];

    const videoFile =
      await getVideo(
        page.id
      );

    let videoSrc:
      string | null =
      null;

    if (videoFile) {
      const extension =
        getVideoExtension(
          videoFile
        );

      const mediaPath =
        `media/page-${
          index + 1
        }${extension}`;

      /*
       * Video không nén lại.
       *
       * Video vốn đã là dữ liệu nén;
       * STORE giúp export nhanh hơn
       * và đỡ tốn RAM/CPU.
       */
      zip.file(
        mediaPath,
        videoFile,
        {
          binary: true,
          compression:
            "STORE",
        }
      );

      videoSrc =
        mediaPath;

      mediaFiles.push(
        mediaPath
      );
    } else if (
      page.videoName
    ) {
      /*
       * Metadata nói rằng trang từng
       * có video nhưng IndexedDB hiện
       * không còn file.
       */
      missingVideoPages.push(
        page.title
      );
    }

    pages.push({
      id:
        page.id,

      title:
        page.title,

      videoName:
        page.videoName,

      duration:
        page.duration,

      quizzes:
        page.quizzes.map(
          (quiz) => ({
            ...quiz,

            options:
              [...quiz.options],
          })
        ),

      videoSrc,
    });
  }

  const course:
    ExportCourse = {
      id:
        project.id,

      title:
        project.title,

      theme,

      completion: {
        passScorePercent:
          completion.passScorePercent,

        minWatchPercent:
          completion.minWatchPercent,

        requireAllQuizzes:
          completion.requireAllQuizzes,
      },

      pages,
    };

  return {
    course,
    missingVideoPages,
    mediaFiles,
  };
}

/* =========================================================
   PUBLIC EXPORT FUNCTION
========================================================= */

export async function exportCurrentProjectAsScorm(): Promise<ScormExportResult> {
  const project =
    useEditorStore.getState()
      .project;

  if (
    !project ||
    project.pages.length ===
      0
  ) {
    throw new Error(
      "Dự án chưa có nội dung để xuất."
    );
  }

  const zip =
    new JSZip();

  const {
    course,
    missingVideoPages,
    mediaFiles,
  } =
    await buildExportCourse(
      project,
      zip
    );

  const indexHtml =
    makeIndexHtml();

  const stylesCss =
    makeStylesCss();

  const playerJs =
    makePlayerJs();

  const courseDataJs =
    makeCourseDataScript(
      course
    );

  zip.file(
    "index.html",
    indexHtml
  );

  zip.file(
    "styles.css",
    stylesCss
  );

  zip.file(
    "player.js",
    playerJs
  );

  zip.file(
    "course-data.js",
    courseDataJs
  );

  const resourceFiles = [
    "index.html",
    "styles.css",
    "player.js",
    "course-data.js",
    ...mediaFiles,
  ];

  zip.file(
    "imsmanifest.xml",
    makeManifest(
      course,
      resourceFiles
    )
  );

  const blob =
    await zip.generateAsync({
      type:
        "blob",

      compression:
        "DEFLATE",

      compressionOptions: {
        level:
          6,
      },
    });

  const fileName =
    `${safeFileName(
      project.title
    )}-SCORM12.zip`;

  const url =
    URL.createObjectURL(
      blob
    );

  const anchor =
    document.createElement(
      "a"
    );

  anchor.href =
    url;

  anchor.download =
    fileName;

  document.body.appendChild(
    anchor
  );

  anchor.click();

  anchor.remove();

  window.setTimeout(
    () => {
      URL.revokeObjectURL(
        url
      );
    },
    1000
  );

  return {
    fileName,
    missingVideoPages,
  };
}
