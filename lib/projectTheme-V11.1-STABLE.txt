import type {
  ProjectTheme,
  ThemeButtonStyle,
  ThemeFont,
  ThemeMode,
} from "@/types/theme";

const STORAGE_KEY =
  "vlearning-studio-project-theme-v1";

const CHANGE_EVENT =
  "vlearning-project-theme-changed";

type ThemeMap =
  Record<string, ProjectTheme>;

const HEX_COLOR =
  /^#[0-9a-fA-F]{6}$/;

const VALID_MODES: ThemeMode[] = [
  "dark",
  "light",
];

const VALID_FONTS: ThemeFont[] = [
  "system",
  "sans",
  "serif",
];

const VALID_BUTTON_STYLES: ThemeButtonStyle[] = [
  "rounded",
  "pill",
  "square",
];

/* =========================================================
   DEFAULT
========================================================= */

export function createDefaultProjectTheme(
  projectTitle = ""
): ProjectTheme {
  return {
    version: 1,
    courseTitle:
      projectTitle,
    mode: "dark",
    primaryColor:
      "#2563eb",
    accentColor:
      "#8b5cf6",
    backgroundColor:
      "#07182c",
    surfaceColor:
      "#102b52",
    textColor:
      "#ffffff",
    fontFamily:
      "system",
    buttonStyle:
      "rounded",
    playerRadius: 16,
    showProgress: true,
    showPageNavigation: true,
  };
}

/* =========================================================
   HELPERS
========================================================= */

function readThemeMap(): ThemeMap {
  if (
    typeof window ===
    "undefined"
  ) {
    return {};
  }

  try {
    const raw =
      window.localStorage.getItem(
        STORAGE_KEY
      );

    if (!raw) {
      return {};
    }

    const parsed =
      JSON.parse(raw);

    if (
      !parsed ||
      typeof parsed !==
        "object" ||
      Array.isArray(parsed)
    ) {
      return {};
    }

    return parsed as ThemeMap;
  } catch (
    error
  ) {
    console.error(
      "Không thể đọc thiết kế dự án:",
      error
    );

    return {};
  }
}

function writeThemeMap(
  map: ThemeMap
) {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  window.localStorage.setItem(
    STORAGE_KEY,
    JSON.stringify(map)
  );
}

function normalizeColor(
  value: unknown,
  fallback: string
) {
  return typeof value ===
      "string" &&
    HEX_COLOR.test(value)
    ? value.toLowerCase()
    : fallback;
}

function normalizeProjectTheme(
  value: Partial<ProjectTheme> | null | undefined,
  projectTitle = ""
): ProjectTheme {
  const defaults =
    createDefaultProjectTheme(
      projectTitle
    );

  const mode =
    VALID_MODES.includes(
      value?.mode as ThemeMode
    )
      ? (value?.mode as ThemeMode)
      : defaults.mode;

  const fontFamily =
    VALID_FONTS.includes(
      value?.fontFamily as ThemeFont
    )
      ? (value?.fontFamily as ThemeFont)
      : defaults.fontFamily;

  const buttonStyle =
    VALID_BUTTON_STYLES.includes(
      value?.buttonStyle as ThemeButtonStyle
    )
      ? (value?.buttonStyle as ThemeButtonStyle)
      : defaults.buttonStyle;

  const rawRadius =
    Number(value?.playerRadius);

  const playerRadius =
    Number.isFinite(rawRadius)
      ? Math.min(
          32,
          Math.max(
            0,
            Math.round(rawRadius)
          )
        )
      : defaults.playerRadius;

  return {
    version: 1,
    courseTitle:
      typeof value?.courseTitle ===
        "string"
        ? value.courseTitle
        : defaults.courseTitle,
    mode,
    primaryColor:
      normalizeColor(
        value?.primaryColor,
        defaults.primaryColor
      ),
    accentColor:
      normalizeColor(
        value?.accentColor,
        defaults.accentColor
      ),
    backgroundColor:
      normalizeColor(
        value?.backgroundColor,
        defaults.backgroundColor
      ),
    surfaceColor:
      normalizeColor(
        value?.surfaceColor,
        defaults.surfaceColor
      ),
    textColor:
      normalizeColor(
        value?.textColor,
        defaults.textColor
      ),
    fontFamily,
    buttonStyle,
    playerRadius,
    showProgress:
      typeof value?.showProgress ===
        "boolean"
        ? value.showProgress
        : defaults.showProgress,
    showPageNavigation:
      typeof value?.showPageNavigation ===
        "boolean"
        ? value.showPageNavigation
        : defaults.showPageNavigation,
  };
}

function notifyThemeChanged(
  projectId: string
) {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  window.dispatchEvent(
    new CustomEvent(
      CHANGE_EVENT,
      {
        detail: {
          projectId,
        },
      }
    )
  );
}

/* =========================================================
   PUBLIC API
========================================================= */

export function loadProjectThemeForProject(
  projectId: string,
  projectTitle = ""
): ProjectTheme {
  const map =
    readThemeMap();

  return normalizeProjectTheme(
    map[projectId],
    projectTitle
  );
}

export function saveProjectThemeForProject(
  projectId: string,
  theme: ProjectTheme,
  projectTitle = ""
): ProjectTheme {
  const normalized =
    normalizeProjectTheme(
      theme,
      projectTitle
    );

  const map =
    readThemeMap();

  map[projectId] =
    normalized;

  writeThemeMap(map);
  notifyThemeChanged(projectId);

  return normalized;
}

export function resetProjectThemeForProject(
  projectId: string,
  projectTitle = ""
): ProjectTheme {
  const map =
    readThemeMap();

  delete map[projectId];

  writeThemeMap(map);
  notifyThemeChanged(projectId);

  return createDefaultProjectTheme(
    projectTitle
  );
}

export function copyProjectTheme(
  sourceProjectId: string,
  targetProjectId: string,
  targetProjectTitle = ""
) {
  const map =
    readThemeMap();

  const source =
    map[sourceProjectId];

  if (!source) {
    return false;
  }

  map[targetProjectId] =
    normalizeProjectTheme(
      {
        ...source,
        courseTitle:
          targetProjectTitle ||
          source.courseTitle,
      },
      targetProjectTitle
    );

  writeThemeMap(map);
  notifyThemeChanged(
    targetProjectId
  );

  return true;
}

export function deleteProjectTheme(
  projectId: string
) {
  const map =
    readThemeMap();

  if (!map[projectId]) {
    return;
  }

  delete map[projectId];

  writeThemeMap(map);
  notifyThemeChanged(projectId);
}

export function getProjectThemeChangeEventName() {
  return CHANGE_EVENT;
}
