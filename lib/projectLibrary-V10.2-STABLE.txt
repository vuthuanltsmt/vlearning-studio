import type {
  ProjectDocument,
} from "@/types/lesson";

export type ProjectLibraryEntry = {
  project: ProjectDocument;
  updatedAt: string;
};

const LIBRARY_STORAGE_KEY =
  "vlearning-studio-project-library-v1";

const ACTIVE_PROJECT_STORAGE_KEY =
  "vlearning-studio-project-v4";

/* =========================================================
   ID
========================================================= */

function createId() {
  if (
    typeof crypto !== "undefined" &&
    typeof crypto.randomUUID ===
      "function"
  ) {
    return crypto.randomUUID();
  }

  return `${Date.now()}-${Math.random()
    .toString(36)
    .slice(2)}`;
}

/* =========================================================
   REMOVE RUNTIME VIDEO URLS
========================================================= */

function cleanProject(
  project: ProjectDocument
): ProjectDocument {
  return {
    ...project,

    pages:
      project.pages.map(
        (page) => ({
          ...page,
          videoUrl:
            undefined,

          quizzes:
            page.quizzes.map(
              (quiz) => ({
                ...quiz,
                options:
                  [...quiz.options],
              })
            ),
        })
      ),
  };
}

/* =========================================================
   READ LIBRARY
========================================================= */

export function getProjectLibrary(): ProjectLibraryEntry[] {
  if (
    typeof window ===
    "undefined"
  ) {
    return [];
  }

  try {
    const raw =
      window.localStorage.getItem(
        LIBRARY_STORAGE_KEY
      );

    if (!raw) {
      return [];
    }

    const parsed =
      JSON.parse(raw);

    if (
      !Array.isArray(parsed)
    ) {
      return [];
    }

    return parsed.sort(
      (
        a:
          ProjectLibraryEntry,
        b:
          ProjectLibraryEntry
      ) =>
        new Date(
          b.updatedAt
        ).getTime() -
        new Date(
          a.updatedAt
        ).getTime()
    );
  } catch (
    error
  ) {
    console.error(
      "Không đọc được thư viện dự án:",
      error
    );

    return [];
  }
}

/* =========================================================
   WRITE LIBRARY
========================================================= */

function writeLibrary(
  entries:
    ProjectLibraryEntry[]
) {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  window.localStorage.setItem(
    LIBRARY_STORAGE_KEY,
    JSON.stringify(
      entries
    )
  );
}

/* =========================================================
   SAVE PROJECT TO LIBRARY
========================================================= */

export function saveProjectToLibrary(
  project: ProjectDocument
) {
  const clean =
    cleanProject(
      project
    );

  const entries =
    getProjectLibrary();

  const existingIndex =
    entries.findIndex(
      (entry) =>
        entry.project.id ===
        clean.id
    );

  const entry:
    ProjectLibraryEntry = {
      project:
        clean,

      updatedAt:
        new Date().toISOString(),
    };

  if (
    existingIndex >= 0
  ) {
    entries[
      existingIndex
    ] =
      entry;
  } else {
    entries.unshift(
      entry
    );
  }

  writeLibrary(
    entries
  );
}

/* =========================================================
   SYNC CURRENT PROJECT
========================================================= */

export function syncCurrentProjectToLibrary() {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  try {
    const raw =
      window.localStorage.getItem(
        ACTIVE_PROJECT_STORAGE_KEY
      );

    if (!raw) {
      return;
    }

    const project:
      ProjectDocument =
        JSON.parse(raw);

    if (
      !project?.id ||
      !Array.isArray(
        project.pages
      )
    ) {
      return;
    }

    saveProjectToLibrary(
      project
    );
  } catch (
    error
  ) {
    console.error(
      "Không đồng bộ được dự án hiện tại:",
      error
    );
  }
}

/* =========================================================
   OPEN PROJECT
========================================================= */

export function setActiveProject(
  project: ProjectDocument
) {
  if (
    typeof window ===
    "undefined"
  ) {
    return;
  }

  const clean =
    cleanProject(
      project
    );

  window.localStorage.setItem(
    ACTIVE_PROJECT_STORAGE_KEY,
    JSON.stringify(
      clean
    )
  );

  saveProjectToLibrary(
    clean
  );
}

/* =========================================================
   CREATE NEW PROJECT
========================================================= */

export function createNewProject(
  title: string
): ProjectDocument {
  const pageId =
    createId();

  const project:
    ProjectDocument = {
      id:
        createId(),

      title:
        title.trim() ||
        "Bài giảng mới",

      selectedPageId:
        pageId,

      pages: [
        {
          id:
            pageId,

          title:
            "Trang 1",

          videoName:
            "",

          duration:
            10,

          quizzes:
            [],
        },
      ],
    };

  setActiveProject(
    project
  );

  return project;
}

/* =========================================================
   DELETE PROJECT
========================================================= */

export function deleteProjectFromLibrary(
  projectId: string
) {
  const entries =
    getProjectLibrary().filter(
      (entry) =>
        entry.project.id !==
        projectId
    );

  writeLibrary(
    entries
  );
}

/* =========================================================
   STATS
========================================================= */

export function getProjectQuizCount(
  project: ProjectDocument
) {
  return project.pages.reduce(
    (
      total,
      page
    ) =>
      total +
      page.quizzes.length,
    0
  );
}
