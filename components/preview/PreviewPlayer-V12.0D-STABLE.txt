function getThemeFontFamily(
  fontFamily: ProjectTheme["fontFamily"]
) {
  switch (fontFamily) {
    case "serif":
      return 'Georgia, "Times New Roman", serif';

    case "sans":
      return 'Arial, Helvetica, sans-serif';

    case "system":
    default:
      return 'system-ui, -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif';
  }
}

function getThemeButtonRadius(
  buttonStyle: ProjectTheme["buttonStyle"]
) {
  switch (buttonStyle) {
    case "pill":
      return 999;

    case "square":
      return 4;

    case "rounded":
    default:
      return 12;
  }
}

function colorWithAlpha(
  hex: string,
  alpha: number
) {
  const value =
    hex.replace("#", "");

  if (
    !/^[0-9a-fA-F]{6}$/.test(
      value
    )
  ) {
    return hex;
  }

  const red =
    parseInt(
      value.slice(0, 2),
      16
    );

  const green =
    parseInt(
      value.slice(2, 4),
      16
    );

  const blue =
    parseInt(
      value.slice(4, 6),
      16
    );

  return `rgba(${red}, ${green}, ${blue}, ${alpha})`;
}

  /* =====================================================
     WATCH PERCENT
  ===================================================== */

  function getWatchPercent(
    pageId: string,
    totalDuration: number
  ) {
    if (
      !Number.isFinite(
        totalDuration
      ) ||
      totalDuration <= 0
    ) {
      return 0;
    }

    const watched =
      watchedSecondsRef
        .current[
        pageId
      ];

    if (!watched) {
      return 0;
    }

    const totalSeconds =
      Math.max(
        1,
        Math.ceil(
          totalDuration
        )
      );

    return Math.min(
      100,
      Math.round(
        (watched.size /
          totalSeconds) *
          100
      )
    );
  }

  /* =====================================================
     RECORD WATCH PROGRESS
  ===================================================== */

  function recordWatchProgress(
    pageId: string,
    time: number,
    totalDuration: number
  ) {
    const previous =
      lastPlaybackTimeRef.current;

    if (
      previous === null
    ) {
      lastPlaybackTimeRef.current =
        time;

      return;
    }

    const delta =
      time -
      previous;

    /*
     * Playback bình thường.
     *
     * Nếu delta lớn, nhiều khả năng
     * người học vừa tua nên phần
     * bị bỏ qua không được tính.
     */
    if (
      delta > 0 &&
      delta <= 2.5
    ) {
      if (
        !watchedSecondsRef
          .current[
          pageId
        ]
      ) {
        watchedSecondsRef
          .current[
          pageId
        ] =
          new Set<number>();
      }

      const watchedSet =
        watchedSecondsRef
          .current[
          pageId
        ];

      const totalSeconds =
        Math.max(
          1,
          Math.ceil(
            totalDuration
          )
        );

      const startSecond =
        Math.max(
          0,
          Math.floor(
            previous
          )
        );

      const endSecond =
        Math.min(
          totalSeconds - 1,
          Math.floor(
            time
          )
        );

      for (
        let second =
          startSecond;
        second <=
        endSecond;
        second++
      ) {
        watchedSet.add(
          second
        );
      }

      setCurrentWatchPercent(
        getWatchPercent(
          pageId,
          totalDuration
        )
      );
    }

    lastPlaybackTimeRef.current =
      time;
  }

  /* =====================================================
     FIRST UNWATCHED SECOND
  ===================================================== */

  function getFirstUnwatchedSecond(
    pageId: string,
    totalDuration: number
  ) {
    const watched =
      watchedSecondsRef
        .current[
        pageId
      ];

    const totalSeconds =
      Math.max(
        1,
        Math.ceil(
          totalDuration
        )
      );

    if (!watched) {
      return 0;
    }

    for (
      let second = 0;
      second <
      totalSeconds;
      second++
    ) {
      if (
        !watched.has(
          second
        )
      ) {
        return second;
      }
    }

    return 0;
  }

  /* =====================================================
     LOAD VIDEO
  ===================================================== */

  useEffect(() => {
    if (
      !hydrated ||
      !page
    ) {
      return;
    }

    const pageId =
      page.id;

    let cancelled =
      false;

    let objectUrl:
      string | null =
      null;

    setVideoResolvedPageId(
      null
    );

    setLoadingVideo(
      true
    );

    setVideoUrl(
      null
    );

    async function loadVideo() {
      try {
        const file =
          await getVideo(
            pageId
          );

        if (
          cancelled
        ) {
          return;
        }

        if (!file) {
          setVideoUrl(
            null
          );

          URL.createObjectURL(
            file
          );

        setVideoUrl(
          objectUrl
        );

        error
      ) {
        console.error(
          "Không tải được video Preview:",

        setVideoUrl(
          null
        );

        if (
          !cancelled
        ) {
          setLoadingVideo(
            false
          );

          setVideoResolvedPageId(
            pageId
          );
        }
      }

    void loadVideo();

    return () => {
      cancelled =
        true;

      if (objectUrl) {
        URL.revokeObjectURL(
          objectUrl
        );
      }
    };

    hydrated,
    page?.id,
  ]);

  /* =====================================================
     RESET PAGE PLAYER
  ===================================================== */

    hasStartedVideoRef.current =
      false;

      null;

      page?.duration ??
        0
    );

      false
    );

    setActiveQuiz(
      null
    );

    setSelectedAnswer(
      null
    );

    setSelectedAnswers([]);

    setShortAnswer("");

    setMatchingSelections([]);

    setQuizResult(
      null
    );

    setCompletionNotice(
      ""
    );

    if (page) {
      setCurrentWatchPercent(
        getWatchPercent(
          page.id,
          page.duration
        )

      setCurrentWatchPercent(
        0
      );

    page?.id,
  ]);

     PAGE PROGRESS
  ===================================================== */

      ? page.quizzes.filter(

            answeredQuizIds.includes(
              quiz.id
            )

      : 0;

  const pageCompleted =
    page
      ? completedPageIds.includes(
          page.id
        )
      : false;

  /* =====================================================
     SHOW QUIZ
  ===================================================== */

  function showQuiz(
    quiz: Quiz
  ) {
    videoRef.current?.pause();

    lastPlaybackTimeRef.current =
      null;

    setIsPlaying(
      false
    );

    setActiveQuiz(
      quiz
    );

    setSelectedAnswer(
      null
    );

    setSelectedAnswers([]);

    setShortAnswer("");

    setMatchingSelections([]);

    setQuizResult(
      null
    );
  }

  /* =====================================================
     PLAY / PAUSE
  ===================================================== */

  function togglePlay() {
    const video =
      videoRef.current;

    if (
      !video ||
      activeQuiz
    ) {
      return;
    }

    if (video.paused) {
      hasStartedVideoRef.current =
        true;

      lastPlaybackTimeRef.current =
        video.currentTime;

      setCompletionNotice(
        ""
      );

      void video.play();
    } else {
      video.pause();
    }
  }

  /* =====================================================
     METADATA
  ===================================================== */

  function handleLoadedMetadata() {
    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    setDuration(
      video.duration
    );

    if (page) {
      setCurrentWatchPercent(
        getWatchPercent(
          page.id,
          video.duration
        )
      );
    }
  }

  /* =====================================================
     TIME UPDATE
  ===================================================== */

  function handleTimeUpdate() {
    if (!page) {
      return;
    }

    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    const time =
      video.currentTime;

    setCurrentTime(
      time
    );

    if (
      hasStartedVideoRef
        .current &&
      !video.paused
    ) {
      recordWatchProgress(
        page.id,
        time,
        video.duration
      );
    }

    /*
     * Chưa Play thì không bật quiz.
     */
    if (
      !hasStartedVideoRef
        .current
    ) {
      return;
    }

    if (activeQuiz) {
      return;
    }

    const dueQuiz =
      [...page.quizzes]
        .sort(
          (a, b) =>
            a.time -
            b.time
        )
        .find(
          (quiz) =>
            !isQuizHandled(
              quiz.id
            ) &&
            quiz.time <=
              time
        );

    if (dueQuiz) {
      showQuiz(
        dueQuiz
      );
    }
  }

  /* =====================================================
     SEEK
  ===================================================== */

  function handleSeek(
    requestedTime: number
  ) {
    if (
      !page ||
      activeQuiz
    ) {
      return;
    }

    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    hasStartedVideoRef.current =
      true;

    lastPlaybackTimeRef.current =
      null;

    /*
     * Nếu quiz bắt buộc:
     * không cho tua vượt quiz.
     *
     * Nếu quiz không bắt buộc:
     * người học có thể tua bình thường.
     */
    if (
      completionSettings
        .requireAllQuizzes &&
      requestedTime >
        currentTime
    ) {
      const blockingQuiz =
        [...page.quizzes]
          .sort(
            (a, b) =>
              a.time -
              b.time
          )
          .find(
            (quiz) =>
              !isQuizHandled(
                quiz.id
              ) &&
              quiz.time >
                currentTime &&
              quiz.time <=
                requestedTime
          );

      if (blockingQuiz) {
        video.currentTime =
          blockingQuiz.time;

        setCurrentTime(
          blockingQuiz.time
        );

        showQuiz(
          blockingQuiz
        );

        return;
      }
    }

    video.currentTime =
      requestedTime;

    setCurrentTime(
      requestedTime
    );
  }

  /* =====================================================
     ANSWER QUIZ
  ===================================================== */

  function answerQuiz(
    answerIndex: number
  ) {
    if (
      !activeQuiz ||
      quizResult
    ) {
      return;
    }

    if (
      getQuizType(
        activeQuiz
      ) ===
      "short-answer" ||
      getQuizType(
        activeQuiz
      ) ===
      "matching"
    ) {
      return;
    }

    if (
      getQuizType(
        activeQuiz
      ) ===
      "multiple"
    ) {
      toggleMultipleAnswer(
        answerIndex
      );

      return;
    }

    setSelectedAnswer(
      answerIndex
    );

    const correct =
      answerIndex ===
      activeQuiz.correctIndex;

    if (correct) {
      setQuizResult(
        "correct"
      );

      if (
        !correctQuizIds.includes(
          activeQuiz.id
        )
      ) {
        setCorrectQuizIds(
          (old) => [
            ...old,
            activeQuiz.id,
          ]
        );

        setScore(
          (old) =>
            old +
            activeQuiz.points
        );
      }
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  function toggleMultipleAnswer(
    answerIndex: number
  ) {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "multiple"
    ) {
      return;
    }

    setSelectedAnswers(
      (old) =>
        old.includes(
          answerIndex
        )
          ? old.filter(
              (item) =>
                item !==
                answerIndex
            )
          : [
              ...old,
              answerIndex,
            ].sort(
              (a, b) =>
                a - b
            )
    );
  }

  function submitMultipleAnswer() {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "multiple" ||
      selectedAnswers.length ===
        0
    ) {
      return;
    }

    const correctIndexes =
      getQuizCorrectIndexes(
        activeQuiz
      );

    const selected =
      [...selectedAnswers].sort(
        (a, b) => a - b
      );

    const correct =
      selected.length ===
        correctIndexes.length &&
      selected.every(
        (value, index) =>
          value ===
          correctIndexes[index]
      );

    if (correct) {
      setQuizResult(
        "correct"
      );

      if (
        !correctQuizIds.includes(
          activeQuiz.id
        )
      ) {
        setCorrectQuizIds(
          (old) => [
            ...old,
            activeQuiz.id,
          ]
        );

        setScore(
          (old) =>
            old +
            activeQuiz.points
        );
      }
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  function submitShortAnswer() {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "short-answer" ||
      !shortAnswer.trim()
    ) {
      return;
    }

    const correct =
      isShortAnswerCorrect(
        shortAnswer,
        activeQuiz.acceptedAnswers
      );

    if (correct) {
      setQuizResult(
        "correct"
      );

      if (
        !correctQuizIds.includes(
          activeQuiz.id
        )
      ) {
        setCorrectQuizIds(
          (old) => [
            ...old,
            activeQuiz.id,
          ]
        );

        setScore(
          (old) =>
            old +
            activeQuiz.points
        );
      }
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  function changeMatchingSelection(
    leftIndex: number,
    rightIndex: number
  ) {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "matching"
    ) {
      return;
    }

    const pairCount =
      (
        activeQuiz.matchingPairs ??
        []
      ).filter(
        (pair) =>
          pair.left?.trim() &&
          pair.right?.trim()
      ).length;

    setMatchingSelections(
      (old) => {
        const next =
          Array.from(
            { length: pairCount },
            (_, index) =>
              old[index] ?? -1
          );

        next[leftIndex] =
          rightIndex;

        return next;
      }
    );
  }

  function submitMatchingAnswer() {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "matching"
    ) {
      return;
    }

    const pairCount =
      (
        activeQuiz.matchingPairs ??
        []
      ).filter(
        (pair) =>
          pair.left?.trim() &&
          pair.right?.trim()
      ).length;

    if (
      pairCount < 2 ||
      matchingSelections.length !==
        pairCount ||
      matchingSelections.some(
        (value) =>
          value < 0
      )
    ) {
      return;
    }

    const correct =
      isMatchingAnswerCorrect(
        matchingSelections,
        activeQuiz.matchingPairs
      );

    if (correct) {
      setQuizResult(
        "correct"
      );

      if (
        !correctQuizIds.includes(
          activeQuiz.id
        )
      ) {
        setCorrectQuizIds(
          (old) => [
            ...old,
            activeQuiz.id,
          ]
        );

        setScore(
          (old) =>
            old +
            activeQuiz.points
        );
      }
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  }

  /* =====================================================
     VIDEO ENDED
  ===================================================== */

  function handleVideoEnded() {
    if (!page) {
      return;
    }

    const video =
      videoRef.current;

    setIsPlaying(
      false
    );

    if (video) {
      recordWatchProgress(
        page.id,
        video.duration,
        video.duration
      );
    }

    lastPlaybackTimeRef.current =
      null;

    /*
     * Chỉ bắt câu hỏi chưa làm
     * khi giáo viên bật chế độ
     * bắt buộc hoàn thành quiz.
     */
    if (
      completionSettings
        .requireAllQuizzes
    ) {
      const unanswered =
        [...page.quizzes]
          .sort(
            (a, b) =>
              a.time -
              b.time
          )
          .find(
            (quiz) =>
              !answeredQuizIds.includes(
                quiz.id
              )
          );

      if (unanswered) {
        if (video) {
          video.currentTime =
            unanswered.time;
        }

        setCurrentTime(
          unanswered.time
        );

        showQuiz(
          unanswered
        );

        return;
      }
    }

    /*
     * Kiểm tra tỷ lệ xem video.
     */
    const watchPercent =
      getWatchPercent(
        page.id,
        video?.duration ??
          duration
      );

    setCurrentWatchPercent(
      watchPercent
    );

    if (
      watchPercent <
      completionSettings
        .minWatchPercent
    ) {
      const missingSecond =
        getFirstUnwatchedSecond(
          page.id,
          video?.duration ??
            duration
        );

      setCompletionNotice(
        `Bạn mới xem ${watchPercent}% video. Cần tối thiểu ${completionSettings.minWatchPercent}% để hoàn thành trang.`
      );

      if (video) {
        video.currentTime =
          missingSecond;
      }

      setCurrentTime(
        missingSecond
      );

      return;
    }

    /*
     * Hoàn thành trang.
     */
    setCompletionNotice(
      ""
    );

    setCompletedPageIds(
      (old) =>
        old.includes(
          page.id
        )
          ? old
          : [
              ...old,
              page.id,
            ]
    );
  }

                                ? "CÂU HỎI GHÉP ĐÔI"
                                : "CÂU HỎI TƯƠNG TÁC"}

                      </div>

                      <div
                        className="shrink-0 px-3 py-1 text-sm font-bold"
                        style={{
                          backgroundColor:
                            colorWithAlpha(
                              projectTheme.accentColor,
                              0.18
                            ),
                          color:
                            projectTheme.accentColor,
                          borderRadius:
                            `${themeButtonRadius}px`,
                        }}
                      >

                        {activeQuiz.points} điểm

                      </div>

                    </div>

                    <h2 className="mt-3 break-words text-xl font-black sm:text-2xl">

                      {activeQuiz.question}

                    </h2>

                  </div>

                  <div className="px-4 pb-5 pt-1 sm:px-6 sm:pb-6">

                  {getQuizType(
                    activeQuiz
                  ) ===
                  "matching" ? (

                    <div className="mt-4 space-y-3">

                      {(activeQuiz.matchingPairs ?? [])
                        .filter(
                          (pair) =>
                            pair.left?.trim() &&
                            pair.right?.trim()
                        )
                        .map(
                          (pair, leftIndex, pairs) => {
                            const optionOrder =
                              pairs.map(
                                (_, index) =>
                                  pairs.length -
                                  1 -
                                  index
                              );

                            const selected =
                              matchingSelections[
                                leftIndex
                              ] ?? -1;

                            const isCorrect =
                              selected ===
                              leftIndex;

                            return (
                              <div
                                key={leftIndex}
                                className={`grid gap-2 border p-3 md:grid-cols-[minmax(0,1fr)_36px_minmax(0,1fr)] md:items-center ${
                                  quizResult
                                    ? isCorrect
                                      ? "border-green-500 bg-green-950/30"
                                      : "border-red-500 bg-red-950/30"
                                    : "border-blue-700 bg-blue-950/40"
                                }`}
                                style={{
                                  borderRadius:
                                    `${themeButtonRadius}px`,
                                }}
                              >
                                <div className="font-semibold">
                                  {pair.left}
                                </div>

                                <div
                                  className="text-center font-black"
                                  style={{
                                    color:
                                      projectTheme.accentColor,
                                  }}
                                >
                                  ↔
                                </div>

                                <select
                                  value={selected}
                                  disabled={
                                    quizResult !==
                                    null
                                  }
                                  onChange={(e) =>
                                    changeMatchingSelection(
                                      leftIndex,
                                      Number(
                                        e.target.value
                                      )
                                    )
                                  }
                                  className="min-w-0 border px-3 py-3 outline-none disabled:opacity-80"
                                  style={{
                                    backgroundColor:
                                      colorWithAlpha(
                                        projectTheme.backgroundColor,
                                        0.75
                                      ),
                                    borderColor:
                                      themeBorderColor,
                                    color:
                                      projectTheme.textColor,
                                    borderRadius:
                                      `${themeButtonRadius}px`,
                                  }}
                                >
                                  <option value={-1}>
                                    -- Chọn nội dung phù hợp --
                                  </option>

                                  {optionOrder.map(
                                    (rightIndex) => (
                                      <option
                                        key={rightIndex}
                                        value={rightIndex}
                                      >
                                        {pairs[rightIndex].right}
                                      </option>
                                    )
                                  )}
                                </select>
                              </div>
                            );
                          }
                        )}

                      {!quizResult && (
                        <button
                          type="button"
                          disabled={
                            !(activeQuiz.matchingPairs ?? [])
                              .filter(
                                (pair) =>
                                  pair.left?.trim() &&
                                  pair.right?.trim()
                              )
                              .every(
                                (_, index) =>
                                  (matchingSelections[index] ?? -1) >=
                                  0
                              )
                          }
                          onClick={
                            submitMatchingAnswer
                          }
                          className="w-full px-5 py-3 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                          style={{
                            backgroundColor:
                              projectTheme.primaryColor,
                            color: "#ffffff",
                            borderRadius:
                              `${themeButtonRadius}px`,
                          }}
                        >
                          Xác nhận ghép đôi
                        </button>
                      )}

                    </div>

                  ) : getQuizType(
                    activeQuiz
                  ) ===
                  "short-answer" ? (

                    <div className="mt-6 space-y-4">

                      <input
                        type="text"
                        value={
                          shortAnswer
                        }
                        disabled={
                          quizResult !==
                          null
                        }
                        onChange={(e) =>
                          setShortAnswer(
                            e.target.value
                          )
                        }
                        onKeyDown={(e) => {
                          if (
                            e.key ===
                            "Enter" &&
                            shortAnswer.trim() &&
                            !quizResult
                          ) {
                            submitShortAnswer();
                          }
                        }}
                        className="w-full border px-4 py-4 text-lg outline-none transition disabled:opacity-70"
                        style={{
                          backgroundColor:
                            colorWithAlpha(
                              projectTheme.backgroundColor,
                              0.7
                            ),
                          borderColor:
                            themeBorderColor,
                          color:
                            projectTheme.textColor,
                          borderRadius:
                            `${themeButtonRadius}px`,
                        }}
                        placeholder="Nhập câu trả lời của bạn..."
                      />

                      {!quizResult && (
                        <button
                          type="button"
                          disabled={
                            !shortAnswer.trim()
                          }
                          onClick={
                            submitShortAnswer
                          }
                          className="w-full px-5 py-3 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                          style={{
                            backgroundColor:
                              projectTheme.primaryColor,
                            color: "#ffffff",
                            borderRadius:
                              `${themeButtonRadius}px`,
                          }}
                        >
                          Xác nhận câu trả lời
                        </button>
                      )}

                    </div>

                  ) : (

                    <>

                      <div className="mt-6 grid gap-3 md:grid-cols-2">

                        {activeQuiz.options.map(
                          (
                            option,
                            index
                          ) => {
                            const quizType =
                              getQuizType(
                                activeQuiz
                              );

                            const correctIndexes =
                              getQuizCorrectIndexes(
                                activeQuiz
                              );

                            const isCorrect =
                              correctIndexes.includes(
                                index
                              );

                            const isSelected =
                              quizType ===
                              "multiple"
                                ? selectedAnswers.includes(
                                    index
                                  )
                                : index ===
                                  selectedAnswer;

                            let classes =
                              "border-blue-600 bg-blue-950/70 hover:bg-blue-900";

                            if (quizResult) {
                              if (isCorrect) {
                                classes =
                                  "border-green-400 bg-green-900/70";
                              } else if (isSelected) {
                                classes =
                                  "border-red-400 bg-red-900/70";
                              } else {
                                classes =
                                  "border-blue-900 bg-blue-950/50 opacity-60";
                              }
                            } else if (
                              quizType ===
                                "multiple" &&
                              isSelected
                            ) {
                              classes =
                                "border-cyan-400 bg-cyan-950/70 ring-1 ring-cyan-400";
                            }

                            return (
                              <button
                                key={
                                  index
                                }
                                type="button"
                                disabled={
                                  quizResult !==
                                  null
                                }
                                onClick={() => {
                                  if (
                                    getQuizType(
                                      activeQuiz
                                    ) ===
                                    "multiple"
                                  ) {
                                    toggleMultipleAnswer(
                                      index
                                    );

                                    return;
                                  }

                                  answerQuiz(
                                    index
                                  );
                                }}
                                className={`flex min-h-20 items-center gap-4 rounded-xl border-2 p-4 text-left font-semibold transition ${classes}`}
                              >

                                <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-black/30 font-black">

                                  {getQuizType(
                                    activeQuiz
                                  ) ===
                                  "true-false"
                                    ? (
                                        index ===
                                        0
                                          ? "✓"
                                          : "✕"
                                      )
                                    : String.fromCharCode(
                                        65 +
                                          index
                                      )}

                                </span>

                                <span>
                                  {option}
                                </span>

                                {getQuizType(
                                  activeQuiz
                                ) ===
                                  "multiple" &&
                                selectedAnswers.includes(
                                  index
                                ) &&
                                !quizResult && (
                                  <span className="ml-auto text-cyan-300">
                                    ✓
                                  </span>
                                )}

                              </button>
                            );
                          }
                        )}

                      </div>

                      {getQuizType(
                        activeQuiz
                      ) ===
                        "multiple" &&
                      !quizResult && (

                        <button
                          type="button"
                          disabled={
                            selectedAnswers.length ===
                            0
                          }
                          onClick={
                            submitMultipleAnswer
                          }
                          className="mt-5 w-full px-5 py-3 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                          style={{
                            backgroundColor:
                              projectTheme.primaryColor,
                            color: "#ffffff",
                            borderRadius:
                              `${themeButtonRadius}px`,
                          }}
                        >
                          Xác nhận đáp án
                        </button>

                      )}

                    </>

                  )}

                  {quizResult && (

                    <div className="mt-6 rounded-xl bg-black/20 p-4">

                      <div
                        className={`flex items-center gap-2 text-lg font-bold ${
                          quizResult ===
                          "correct"
                            ? "text-green-300"
                            : "text-red-300"
                        }`}
                      >

                        {quizResult ===
                        "correct" ? (

                          <CheckCircle2
                            size={22}
                          />

                        ) : (

                          <XCircle
                            size={22}
                          />

                        )}

                        {quizResult ===
                        "correct"
                          ? `Chính xác! +${activeQuiz.points} điểm`
                          : "Chưa chính xác"}

                      </div>

                      <p className="mt-2 text-slate-200">

                        {quizResult ===
                        "correct"
                          ? activeQuiz.feedbackCorrect
                          : activeQuiz.feedbackWrong}

                      </p>

                      <button
                        type="button"
                        onClick={
                          continueAfterQuiz
                        }
                        className="mt-5 w-full px-5 py-3 font-bold transition hover:brightness-110"
                        style={{
                          backgroundColor:
                            projectTheme.primaryColor,
                          color: "#ffffff",
                          borderRadius:
                            `${themeButtonRadius}px`,
                        }}
                      >

                        Tiếp tục

                      </button>

                    </div>

                  )}

                  {!completionSettings.requireAllQuizzes &&
                    !quizResult && (

                      <button
                        type="button"
                        onClick={
                          skipActiveQuiz
                        }
                        className="mt-4 flex w-full items-center justify-center gap-2 border px-5 py-3 font-bold transition hover:brightness-110"
                        style={{
                          borderColor:
                            themeBorderColor,
                          backgroundColor:
                            colorWithAlpha(
                              projectTheme.backgroundColor,
                              0.55
                            ),
                          color:
                            projectTheme.textColor,
                          borderRadius:
                            `${themeButtonRadius}px`,
                        }}
                      >

                        <SkipForward
                          size={18}
                        />

                        Bỏ qua câu hỏi

                      </button>

                    )}

                  </div>

                </div>

              </div>

            )}

          </section>

          {projectTheme.showProgress && (
            <section
              className="mt-3 shrink-0 border px-4 py-3"
              style={{
                backgroundColor:
                  projectTheme.surfaceColor,
                borderColor:
                  themeBorderColor,
                borderRadius:
                  `${Math.max(10, projectTheme.playerRadius)}px`,
              }}
            >
              <div className="mb-2 flex items-center justify-between gap-4 text-xs">
                <span
                  style={{
                    color:
                      themeMutedText,
                  }}
                >
                  Tiến độ bài giảng
                </span>

                <strong
                  style={{
                    color:
                      projectTheme.accentColor,
                  }}
                >
                  {Math.round(
                    ((pageIndex +
                      (pageCompleted ? 1 : 0)) /
                      Math.max(
                        1,
                        project.pages.length
                      )) *
                      100
                  )}%
                </strong>
              </div>

              <div
                className="h-2 overflow-hidden rounded-full"
                style={{
                  backgroundColor:
                    colorWithAlpha(
                      projectTheme.backgroundColor,
                      0.5
                    ),
                }}
              >
                <div
                  className="h-full rounded-full transition-all"
                  style={{
                    width: `${Math.min(
                      100,
                      Math.round(
                        ((pageIndex +
                          (pageCompleted ? 1 : 0)) /
                          Math.max(
                            1,
                            project.pages.length
                          )) *
                          100
                      )
                    )}%`,
                    backgroundColor:
                      projectTheme.accentColor,
                  }}
                />
              </div>
            </section>
          )}

          {/* PLAYER */}

          <section
            className="mt-3 shrink-0 border px-4 py-3"
            style={{
              backgroundColor:
                projectTheme.surfaceColor,
              borderColor:
                themeBorderColor,
              borderRadius:
                `${Math.max(10, projectTheme.playerRadius)}px`,
            }}
          >

            <div className="flex items-center gap-4">

              <button
                type="button"
                onClick={
                  togglePlay
                }
                disabled={
                  !videoUrl ||
                  activeQuiz !==
                    null
                }
                className="flex h-11 w-11 shrink-0 items-center justify-center text-white transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                style={{
                  backgroundColor:
                    projectTheme.primaryColor,
                  borderRadius:
                    `${projectTheme.buttonStyle === "square" ? 8 : 999}px`,
                }}
              >

                {isPlaying ? (

                  <Pause
                    size={20}
                    fill="currentColor"
                  />

                ) : (

                  <Play
                    size={20}
                    fill="currentColor"
                  />

                )}

              </button>

              <div className="w-32 shrink-0 font-mono text-sm">

                {formatTime(
                  currentTime
                )}

                {" / "}

                {formatTime(
                  duration
                )}

              </div>

              <input
                type="range"
                min={0}
                max={
                  duration > 0
                    ? duration
                    : 1
                }
                step={0.01}
                value={
                  Math.min(
                    currentTime,
                    duration > 0
                      ? duration
                      : 1
                  )
                }
                disabled={
                  !videoUrl ||
                  activeQuiz !==
                    null
                }
                onChange={(event) =>
                  handleSeek(
                    Number(
                      event.target.value
                    )
                  )
                }
                className="min-w-0 flex-1 cursor-pointer disabled:cursor-not-allowed"
                style={{
                  accentColor:
                    projectTheme.accentColor,
                }}
              />

            </div>

          </section>

          {/* NAVIGATION */}

          {projectTheme.showPageNavigation && (
          <section className="mt-3 shrink-0">

            <div className="flex items-center justify-between gap-4">

              <button
                type="button"
                onClick={
                  previousPage
                }
                disabled={
                  pageIndex ===
                    0 ||
                  activeQuiz !==
                    null
                }
                className="flex items-center gap-2 border px-5 py-2.5 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                style={{
                  borderColor:
                    themeBorderColor,
                  backgroundColor:
                    colorWithAlpha(
                      projectTheme.surfaceColor,
                      0.9
                    ),
                  color:
                    projectTheme.textColor,
                  borderRadius:
                    `${themeButtonRadius}px`,
                }}
              >

                <ChevronLeft
                  size={18}
                />

                Trang trước

              </button>

              <div className="text-center">

                <div className="text-sm text-slate-400">

                  Câu hỏi{" "}

                  <strong className="text-cyan-300">
                    {completedQuizCount}
                  </strong>

                  {" / "}

                  {page.quizzes.length}

                  {" • Đã xem "}

                  <strong
                    className={
                      currentWatchPercent >=
                      completionSettings
                        .minWatchPercent
                        ? "text-green-300"
                        : "text-yellow-300"
                    }
                  >

                    {currentWatchPercent}%

                  </strong>

                </div>

                {completionNotice ? (

                  <div className="mt-1 text-xs font-semibold text-yellow-300">

                    {completionNotice}

                  </div>

                ) : pageCompleted ? (

                  <div className="mt-1 text-xs font-bold text-green-300">

                    ✓ Trang đã hoàn thành

                  </div>

                ) : (

                  <div className="mt-1 text-xs text-slate-500">

                    Yêu cầu: xem ≥{" "}
                    {completionSettings.minWatchPercent}%

                    {completionSettings.requireAllQuizzes
                      ? " và hoàn thành tất cả câu hỏi"
                      : ""}

                  </div>

                )}

              </div>

              <button
                type="button"
                onClick={
                  nextPage
                }
                disabled={
                  !pageCompleted ||
                  activeQuiz !==
                    null
                }
                className="flex items-center gap-2 px-5 py-2.5 font-bold text-white transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                style={{
                  background: `linear-gradient(90deg, ${projectTheme.primaryColor}, ${projectTheme.accentColor})`,
                  borderRadius:
                    `${themeButtonRadius}px`,
                }}
              >

                {pageIndex ===
                project.pages.length -
                  1
                  ? "Hoàn thành"
                  : "Trang tiếp theo"}

                <ChevronRight
                  size={18}
                />

              </button>

            </div>

          </section>
          )}

        </div>

      </main>

    </div>
  );
}
