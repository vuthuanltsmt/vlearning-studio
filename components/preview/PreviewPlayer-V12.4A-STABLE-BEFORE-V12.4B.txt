function getThemeFontFamily(
  fontFamily: ProjectTheme["fontFamily"]
) {
  switch (fontFamily) {
    case "serif":
      return 'Georgia, "Times New Roman", serif';

    case "sans":
      return 'Arial, Helvetica, sans-serif';

    case "system":
    default:
      return 'system-ui, -apple-system, BlinkMacSystemFont, "Segoe UI", sans-serif';
  }
}

function getThemeButtonRadius(
  buttonStyle: ProjectTheme["buttonStyle"]
) {
  switch (buttonStyle) {
    case "pill":
      return 999;

    case "square":
      return 4;

    case "rounded":
    default:
      return 12;
  }
}

function colorWithAlpha(
  hex: string,
  alpha: number
) {
  const value =
    hex.replace("#", "");

  if (
    !/^[0-9a-fA-F]{6}$/.test(
      value
    )
  ) {
    return hex;
  }

  const red =
    parseInt(
      value.slice(0, 2),
      16
    );

  const green =
    parseInt(
      value.slice(2, 4),
      16
    );

  const blue =
    parseInt(
      value.slice(4, 6),
      16
    );

  return `rgba(${red}, ${green}, ${blue}, ${alpha})`;
}

function formatReportDuration(
  seconds: number
) {
  const safeSeconds =
    Math.max(
      0,
      Math.floor(
        Number.isFinite(
          seconds
        )
          ? seconds
          : 0
      )
    );

  const hours =
    Math.floor(
      safeSeconds /
        3600
    );

  const minutes =
    Math.floor(
      (safeSeconds %
        3600) /
        60
    );

  const remainingSeconds =
    safeSeconds %
    60;

  if (hours > 0) {
    return `${hours} giờ ${minutes} phút`;
  }

  if (minutes > 0) {
    return `${minutes} phút ${remainingSeconds} giây`;
  }

  return `${remainingSeconds} giây`;
}

type LearnerProgressSnapshot = {

function getLearnerProgressKey(
  projectId: string
) {
  return `${LEARNER_PROGRESS_KEY_PREFIX}:${projectId}`;
}

  /* =====================================================
     WATCH PERCENT
  ===================================================== */

  function getWatchPercent(
    pageId: string,
    totalDuration: number
  ) {
    if (
      !Number.isFinite(
        totalDuration
      ) ||
      totalDuration <= 0
    ) {
      return 0;
    }

    const watched =
      watchedSecondsRef
        .current[
        pageId
      ];

    if (!watched) {
      return 0;
    }

    const totalSeconds =
      Math.max(
        1,
        Math.ceil(
          totalDuration
        )
      );

    return Math.min(
      100,
      Math.round(
        (watched.size /
          totalSeconds) *
          100
      )
    );
  }

  /* =====================================================
     RECORD WATCH PROGRESS
  ===================================================== */

  function recordWatchProgress(
    pageId: string,
    time: number,
    totalDuration: number
  ) {
    const previous =
      lastPlaybackTimeRef.current;

    if (
      previous === null
    ) {
      lastPlaybackTimeRef.current =
        time;

      return;
    }

    const delta =
      time -
      previous;

    /*
     * Playback bình thường.
     *
     * Nếu delta lớn, nhiều khả năng
     * người học vừa tua nên phần
     * bị bỏ qua không được tính.
     */
    if (
      delta > 0 &&
      delta <= 2.5
    ) {
      if (
        !watchedSecondsRef
          .current[
          pageId
        ]
      ) {
        watchedSecondsRef
          .current[
          pageId
        ] =
          new Set<number>();
      }

      const watchedSet =
        watchedSecondsRef
          .current[
          pageId
        ];

      const totalSeconds =
        Math.max(
          1,
          Math.ceil(
            totalDuration
          )
        );

      const startSecond =
        Math.max(
          0,
          Math.floor(
            previous
          )
        );

      const endSecond =
        Math.min(
          totalSeconds - 1,
          Math.floor(
            time
          )
        );

      for (
        let second =
          startSecond;
        second <=
        endSecond;
        second++
      ) {
        watchedSet.add(
          second
        );
      }

      setCurrentWatchPercent(
        getWatchPercent(
          pageId,
          totalDuration
        )
      );
    }

    lastPlaybackTimeRef.current =
      time;
  }

  /* =====================================================
     FIRST UNWATCHED SECOND
  ===================================================== */

  function getFirstUnwatchedSecond(
    pageId: string,
    totalDuration: number
  ) {
    const watched =
      watchedSecondsRef
        .current[
        pageId
      ];

    const totalSeconds =
      Math.max(
        1,
        Math.ceil(
          totalDuration
        )
      );

    if (!watched) {
      return 0;
    }

    for (
      let second = 0;
      second <
      totalSeconds;
      second++
    ) {
      if (
        !watched.has(
          second
        )
      ) {
        return second;
      }
    }

    return 0;
  }

  /* =====================================================
     TIME UPDATE
  ===================================================== */

  function handleTimeUpdate() {
    if (!page) {
      return;
    }

    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    const time =
      video.currentTime;

    setCurrentTime(
      time
    );

    lastPositionByPageRef
      .current[
      page.id
    ] =
      time;

    const now =
      Date.now();

    if (
      progressRestoredRef.current &&
      now -
        lastProgressSaveAtRef.current >=
        1200
    ) {
      lastProgressSaveAtRef.current =
        now;

      saveLearnerProgress(
        page.id,
        time
      );
    }

    if (
      hasStartedVideoRef
        .current &&
      !video.paused
    ) {
      recordWatchProgress(
        page.id,
        time,
        video.duration
      );
    }

    /*
     * Chưa Play thì không bật quiz.
     */
    if (
      !hasStartedVideoRef
        .current
    ) {
      return;
    }

    if (activeQuiz) {
      return;
    }

    const dueQuiz =
      [...page.quizzes]
        .sort(
          (a, b) =>
            a.time -
            b.time
        )
        .find(
          (quiz) =>
            !isQuizHandled(
              quiz.id
            ) &&
            quiz.time <=
              time
        );

    if (dueQuiz) {
      showQuiz(
        dueQuiz
      );
    }
  }

  /* =====================================================
     SEEK
  ===================================================== */

  function handleSeek(
    requestedTime: number
  ) {
    if (
      !page ||
      activeQuiz
    ) {
      return;
    }

    const video =
      videoRef.current;

    if (!video) {
      return;
    }

    hasStartedVideoRef.current =
      true;

    lastPlaybackTimeRef.current =
      null;

    /*
     * Nếu quiz bắt buộc:
     * không cho tua vượt quiz.
     *
     * Nếu quiz không bắt buộc:
     * người học có thể tua bình thường.
     */
    if (
      completionSettings
        .requireAllQuizzes &&
      requestedTime >
        currentTime
    ) {
      const blockingQuiz =
        [...page.quizzes]
          .sort(
            (a, b) =>
              a.time -
              b.time
          )
          .find(
            (quiz) =>
              !isQuizHandled(
                quiz.id
              ) &&
              quiz.time >
                currentTime &&
              quiz.time <=
                requestedTime
          );

      if (blockingQuiz) {
        video.currentTime =
          blockingQuiz.time;

        setCurrentTime(
          blockingQuiz.time
        );

        showQuiz(
          blockingQuiz
        );

        return;
      }
    }

    video.currentTime =
      requestedTime;

    setCurrentTime(
      requestedTime
    );
  }

  /* =====================================================
     ANSWER QUIZ
  ===================================================== */

  function answerQuiz(
    answerIndex: number
  ) {
    if (
      !activeQuiz ||
      quizResult
    ) {
      return;
    }

    if (
      getQuizType(
        activeQuiz
      ) ===
      "short-answer" ||
      getQuizType(
        activeQuiz
      ) ===
      "matching" ||
      getQuizType(
        activeQuiz
      ) ===
      "ordering" ||
      getQuizType(
        activeQuiz
      ) ===
      "classification"
    ) {
      return;
    }

    if (
      getQuizType(
        activeQuiz
      ) ===
      "multiple"
    ) {
      toggleMultipleAnswer(
        answerIndex
      );

      return;
    }

    setSelectedAnswer(
      answerIndex
    );

    const correct =
      answerIndex ===
      activeQuiz.correctIndex;

    if (correct) {
      setQuizResult(
        "correct"
      );

      if (
        !correctQuizIds.includes(
          activeQuiz.id
        )
      ) {
        setCorrectQuizIds(
          (old) => [
            ...old,
            activeQuiz.id,
          ]
        );

        setScore(
          (old) =>
            old +
            activeQuiz.points
        );
      }
    } else {
      setQuizResult(
        "wrong"
      );
    }
  }

  function toggleMultipleAnswer(
    answerIndex: number
  ) {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "multiple"
    ) {
      return;
    }

    setSelectedAnswers(
      (old) =>
        old.includes(
          answerIndex
        )
          ? old.filter(
              (item) =>
                item !==
                answerIndex
            )
          : [
              ...old,
              answerIndex,
            ].sort(
              (a, b) =>
                a - b
            )
    );
  }

  function submitMultipleAnswer() {

  function changeMatchingSelection(
    leftIndex: number,
    rightIndex: number
  ) {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "matching"
    ) {
      return;
    }

    const pairCount =
      (
        activeQuiz.matchingPairs ??
        []
      ).filter(
        (pair) =>
          pair.left?.trim() &&
          pair.right?.trim()
      ).length;

    setMatchingSelections(
      (old) => {
        const next =
          Array.from(
            { length: pairCount },
            (_, index) =>
              old[index] ?? -1
          );

        next[leftIndex] =
          rightIndex;

        return next;
      }
    );
  }

  function submitMatchingAnswer() {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "matching"
    ) {

        activeQuiz.matchingPairs ??

      )

      isMatchingAnswerCorrect(
        matchingSelections,
        activeQuiz.matchingPairs

    if (correct) {
      setQuizResult(
        "correct"

      if (

          activeQuiz.id

  }

  function moveOrderingItem(
    position: number,
    direction:
      | -1
      | 1
  ) {
    if (
      !activeQuiz ||
      quizResult ||
      getQuizType(
        activeQuiz
      ) !==
        "ordering"
    ) {
      return;
    }

    setOrderingSelections(
      (old) => {
        const target =
          position +
          direction;

        if (
          target < 0 ||
          target >=
            old.length
        ) {
          return old;
        }

        const next =
          [
            ...old,
          ];

        [
          next[position],
          next[target],
        ] = [
          next[target],
          next[position],
        ];

        return next;
      }
    );
  }

        return next;

     * bắt buộc hoàn thành quiz.

    if (
      completionSettings
        .requireAllQuizzes
    ) {
      const unanswered =
        [...page.quizzes]
          .sort(
            (a, b) =>
              a.time -
              b.time
          )
          .find(
            (quiz) =>
              !answeredQuizIds.includes(
                quiz.id
              )
          );

      if (unanswered) {
        if (video) {
          video.currentTime =
            unanswered.time;
        }

        setCurrentTime(
          unanswered.time
        );

        showQuiz(
          unanswered
        );

        return;
      }
    }

    /*
     * Kiểm tra tỷ lệ xem video.
     */
    const watchPercent =
      getWatchPercent(
        page.id,
        video?.duration ??
          duration
      );

    setCurrentWatchPercent(
      watchPercent
    );

    if (
      watchPercent <
      completionSettings
        .minWatchPercent
    ) {
      const missingSecond =
        getFirstUnwatchedSecond(
          page.id,
          video?.duration ??
            duration
        );

      setCompletionNotice(
        `Bạn mới xem ${watchPercent}% video. Cần tối thiểu ${completionSettings.minWatchPercent}% để hoàn thành trang.`
      );

      if (video) {
        video.currentTime =
          missingSecond;
      }

      setCurrentTime(
        missingSecond
      );

      return;
    }

    /*
     * Hoàn thành trang.
     */
    setCompletionNotice(
      ""
    );

    setCompletedPageIds(
      (old) =>
        old.includes(
          page.id
        )
          ? old
          : [
              ...old,
              page.id,
            ]
    );
  }

  /* =====================================================
     PAGE WITHOUT VIDEO
  ===================================================== */

  useEffect(() => {
    if (!page) {
      return;
    }

    if (
      videoResolvedPageId !==
      page.id
    ) {
      return;
    }

    if (
      loadingVideo ||
      videoUrl

      return;

            : [

      return;

     * coi là hoàn thành.

    if (
      !completionSettings
        .requireAllQuizzes
    ) {
      setCompletedPageIds(
        (old) =>
          old.includes(
            page.id
          )
            ? old
            : [
                ...old,
                page.id,
              ]
      );

      return;
    }

    if (activeQuiz) {
      return;
    }

    const nextQuiz =
      page.quizzes.find(
        (quiz) =>
          !answeredQuizIds.includes(
            quiz.id
          )

    setMatchingSelections([]);

    setOrderingSelections([]);

      );

      return;
    }

    /*

     */
    setCompletedPageIds(
      (old) =>
        old.includes(
          page.id
        )
          ? old
          : [
              ...old,
              page.id,
            ]
    );
  }, [

    page?.quizzes,
    videoResolvedPageId,
    loadingVideo,
    videoUrl,

    answeredQuizIds,
    completionSettings
      .requireAllQuizzes,

  function isPageUnlocked(
    targetIndex: number
  ) {
    const targetPage =
      project.pages[
        targetIndex
      ];

    if (!targetPage) {
      return false;
    }

    if (
      targetIndex ===
      pageIndex
    ) {
      return true;
    }

    if (
      completedPageIds.includes(
        targetPage.id
      )
    ) {
      return true;
    }

    if (
      targetIndex <
      pageIndex
    ) {
      return false;
    }

    return project.pages
      .slice(
        0,
        targetIndex
      )
      .every((item) =>
        completedPageIds.includes(
          item.id
        )
      );
  }

                          type="button"
                          disabled={
                            !shortAnswer.trim()
                          }
                          onClick={
                            submitShortAnswer
                          }
                          className="w-full px-5 py-3 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                          style={{
                            backgroundColor:
                              projectTheme.primaryColor,
                            color: "#ffffff",
                            borderRadius:
                              `${themeButtonRadius}px`,
                          }}
                        >
                          Xác nhận câu trả lời
                        </button>
                      )}

                    </div>

                  ) : (

                    <>

                      <div className="mt-6 grid gap-3 md:grid-cols-2">

                        {activeQuiz.options.map(
                          (
                            option,
                            index
                          ) => {
                            const quizType =
                              getQuizType(
                                activeQuiz
                              );

                            const correctIndexes =
                              getQuizCorrectIndexes(
                                activeQuiz
                              );

                            const isCorrect =
                              correctIndexes.includes(
                                index
                              );

                            const isSelected =
                              quizType ===
                              "multiple"
                                ? selectedAnswers.includes(
                                    index
                                  )
                                : index ===
                                  selectedAnswer;

                            let classes =
                              "border-blue-600 bg-blue-950/70 hover:bg-blue-900";

                            if (quizResult) {
                              if (isCorrect) {
                                classes =
                                  "border-green-400 bg-green-900/70";
                              } else if (isSelected) {
                                classes =
                                  "border-red-400 bg-red-900/70";
                              } else {
                                classes =
                                  "border-blue-900 bg-blue-950/50 opacity-60";
                              }
                            } else if (
                              quizType ===
                                "multiple" &&
                              isSelected
                            ) {
                              classes =
                                "border-cyan-400 bg-cyan-950/70 ring-1 ring-cyan-400";
                            }

                            return (
                              <button
                                key={
                                  index
                                }
                                type="button"
                                disabled={
                                  quizResult !==
                                  null
                                }
                                onClick={() => {
                                  if (
                                    getQuizType(
                                      activeQuiz
                                    ) ===
                                    "multiple"
                                  ) {
                                    toggleMultipleAnswer(
                                      index
                                    );

                                    return;
                                  }

                                  answerQuiz(
                                    index
                                  );
                                }}
                                className={`flex min-h-20 items-center gap-4 rounded-xl border-2 p-4 text-left font-semibold transition ${classes}`}
                              >

                                <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-full bg-black/30 font-black">

                                  {getQuizType(
                                    activeQuiz
                                  ) ===
                                  "true-false"
                                    ? (
                                        index ===
                                        0
                                          ? "✓"
                                          : "✕"
                                      )
                                    : String.fromCharCode(
                                        65 +
                                          index
                                      )}

                                </span>

                                <span>
                                  {option}
                                </span>

                                {getQuizType(
                                  activeQuiz
                                ) ===
                                  "multiple" &&
                                selectedAnswers.includes(
                                  index
                                ) &&
                                !quizResult && (
                                  <span className="ml-auto text-cyan-300">
                                    ✓
                                  </span>
                                )}

                              </button>
                            );
                          }
                        )}

                      </div>

                      {getQuizType(
                        activeQuiz
                      ) ===
                        "multiple" &&
                      !quizResult && (

                        <button
                          type="button"
                          disabled={
                            selectedAnswers.length ===
                            0
                          }
                          onClick={
                            submitMultipleAnswer
                          }
                          className="mt-5 w-full px-5 py-3 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                          style={{
                            backgroundColor:
                              projectTheme.primaryColor,
                            color: "#ffffff",
                            borderRadius:
                              `${themeButtonRadius}px`,
                          }}
                        >
                          Xác nhận đáp án
                        </button>

                      )}

                    </>

                  )}

                  {quizResult && (

                    <div className="mt-6 rounded-xl bg-black/20 p-4">

                      <div
                        className={`flex items-center gap-2 text-lg font-bold ${
                          quizResult ===
                          "correct"
                            ? "text-green-300"
                            : "text-red-300"
                        }`}
                      >

                        {quizResult ===
                        "correct" ? (

                          <CheckCircle2
                            size={22}
                          />

                        ) : (

                          <XCircle
                            size={22}
                          />

                        )}

                        {quizResult ===
                        "correct"
                          ? `Chính xác! +${activeQuiz.points} điểm`
                          : "Chưa chính xác"}

                      </div>

                      <p className="mt-2 text-slate-200">

                        {quizResult ===
                        "correct"
                          ? activeQuiz.feedbackCorrect
                          : activeQuiz.feedbackWrong}

                      </p>

                      <button
                        type="button"
                        onClick={
                          continueAfterQuiz
                        }
                        className="mt-5 w-full px-5 py-3 font-bold transition hover:brightness-110"
                        style={{
                          backgroundColor:
                            projectTheme.primaryColor,
                          color: "#ffffff",
                          borderRadius:
                            `${themeButtonRadius}px`,
                        }}
                      >

                        Tiếp tục

                      </button>

                    </div>

                  )}

                  {!completionSettings.requireAllQuizzes &&
                    !quizResult && (

                      <button
                        type="button"
                        onClick={
                          skipActiveQuiz
                        }
                        className="mt-4 flex w-full items-center justify-center gap-2 border px-5 py-3 font-bold transition hover:brightness-110"
                        style={{
                          borderColor:
                            themeBorderColor,
                          backgroundColor:
                            colorWithAlpha(
                              projectTheme.backgroundColor,
                              0.55
                            ),
                          color:
                            projectTheme.textColor,
                          borderRadius:
                            `${themeButtonRadius}px`,
                        }}
                      >

                        <SkipForward
                          size={18}
                        />

                        Bỏ qua câu hỏi

                      </button>

                    )}

                  </div>

                </div>

              </div>

            )}

          </section>

          {projectTheme.showProgress && (
            <section
              className="mt-3 shrink-0 border px-4 py-3"
              style={{
                backgroundColor:
                  projectTheme.surfaceColor,
                borderColor:
                  themeBorderColor,
                borderRadius:
                  `${Math.max(10, projectTheme.playerRadius)}px`,
              }}
            >
              <div className="mb-2 flex items-center justify-between gap-4 text-xs">
                <span
                  style={{
                    color:
                      themeMutedText,
                  }}
                >
                  Tiến độ bài giảng
                </span>

                <strong
                  style={{
                    color:
                      projectTheme.accentColor,
                  }}
                >
                  {courseProgressPercent}%
                </strong>
              </div>

              <div
                className="h-2 overflow-hidden rounded-full"
                style={{
                  backgroundColor:
                    colorWithAlpha(
                      projectTheme.backgroundColor,
                      0.5
                    ),
                }}
              >
                <div
                  className="h-full rounded-full transition-all"
                  style={{
                    width:
                      `${courseProgressPercent}%`,
                    backgroundColor:
                      projectTheme.accentColor,
                  }}
                />
              </div>
            </section>
          )}

          {/* PLAYER */}

          <section
            className="mt-3 shrink-0 border px-4 py-3"
            style={{
              backgroundColor:
                projectTheme.surfaceColor,
              borderColor:
                themeBorderColor,
              borderRadius:
                `${Math.max(10, projectTheme.playerRadius)}px`,
            }}
          >

            <div className="flex items-center gap-4">

              <button
                type="button"
                onClick={
                  togglePlay
                }
                disabled={
                  !videoUrl ||
                  activeQuiz !==
                    null
                }
                className="flex h-11 w-11 shrink-0 items-center justify-center text-white transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                style={{
                  backgroundColor:
                    projectTheme.primaryColor,
                  borderRadius:
                    `${projectTheme.buttonStyle === "square" ? 8 : 999}px`,
                }}
              >

                {isPlaying ? (

                  <Pause
                    size={20}
                    fill="currentColor"
                  />

                ) : (

                  <Play
                    size={20}
                    fill="currentColor"
                  />

                )}

              </button>

              <div className="w-32 shrink-0 font-mono text-sm">

                {formatTime(
                  currentTime
                )}

                {" / "}

                {formatTime(
                  duration
                )}

              </div>

              <input
                type="range"
                min={0}
                max={
                  duration > 0
                    ? duration
                    : 1
                }
                step={0.01}
                value={
                  Math.min(
                    currentTime,
                    duration > 0
                      ? duration
                      : 1
                  )
                }
                disabled={
                  !videoUrl ||
                  activeQuiz !==
                    null
                }
                onChange={(event) =>
                  handleSeek(
                    Number(
                      event.target.value
                    )
                  )
                }
                className="min-w-0 flex-1 cursor-pointer disabled:cursor-not-allowed"
                style={{
                  accentColor:
                    projectTheme.accentColor,
                }}
              />

            </div>

          </section>

          {/* NAVIGATION */}

          {projectTheme.showPageNavigation && (
          <section className="mt-3 shrink-0">

            <div className="flex items-center justify-between gap-4">

              <button
                type="button"
                onClick={
                  previousPage
                }
                disabled={
                  pageIndex ===
                    0 ||
                  activeQuiz !==
                    null
                }
                className="flex items-center gap-2 border px-5 py-2.5 font-bold transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                style={{
                  borderColor:
                    themeBorderColor,
                  backgroundColor:
                    colorWithAlpha(
                      projectTheme.surfaceColor,
                      0.9
                    ),
                  color:
                    projectTheme.textColor,
                  borderRadius:
                    `${themeButtonRadius}px`,
                }}
              >

                <ChevronLeft
                  size={18}
                />

                Trang trước

              </button>

              <div className="text-center">

                <div className="text-sm text-slate-400">

                  Câu hỏi{" "}

                  <strong className="text-cyan-300">
                    {completedQuizCount}
                  </strong>

                  {" / "}

                  {page.quizzes.length}

                  {" • Đã xem "}

                  <strong
                    className={
                      currentWatchPercent >=
                      completionSettings
                        .minWatchPercent
                        ? "text-green-300"
                        : "text-yellow-300"
                    }
                  >

                    {currentWatchPercent}%

                  </strong>

                </div>

                {completionNotice ? (

                  <div className="mt-1 text-xs font-semibold text-yellow-300">

                    {completionNotice}

                  </div>

                ) : pageCompleted ? (

                  <div className="mt-1 text-xs font-bold text-green-300">

                    ✓ Trang đã hoàn thành

                  </div>

                ) : (

                  <div className="mt-1 text-xs text-slate-500">

                    Yêu cầu: xem ≥{" "}
                    {completionSettings.minWatchPercent}%

                    {completionSettings.requireAllQuizzes
                      ? " và hoàn thành tất cả câu hỏi"
                      : ""}

                  </div>

                )}

              </div>

              <button
                type="button"
                onClick={
                  nextPage
                }
                disabled={
                  !pageCompleted ||
                  activeQuiz !==
                    null
                }
                className="flex items-center gap-2 px-5 py-2.5 font-bold text-white transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40"
                style={{
                  background: `linear-gradient(90deg, ${projectTheme.primaryColor}, ${projectTheme.accentColor})`,
                  borderRadius:
                    `${themeButtonRadius}px`,
                }}
              >

                {pageIndex ===
                project.pages.length -
                  1
                  ? "Hoàn thành"
                  : "Trang tiếp theo"}

                <ChevronRight
                  size={18}
                />

              </button>

            </div>

          </section>
          )}

          {/* V12.2A FIX: FINAL COMPLETION CTA
              Luôn hiện ở trang cuối ngay cả khi giáo viên
              đã tắt showPageNavigation trong theme. */}
          {!projectTheme.showPageNavigation &&
            pageIndex ===
              project.pages.length -
                1 && (
            <section className="mt-3 shrink-0">

              <div
                className="flex flex-col items-center gap-2 border p-4 sm:flex-row sm:justify-between"
                style={{
                  borderColor:
                    themeBorderColor,
                  backgroundColor:
                    colorWithAlpha(
                      projectTheme.surfaceColor,
                      0.9
                    ),
                  borderRadius:
                    `${projectTheme.playerRadius}px`,
                }}
              >

                <div className="text-center sm:text-left">

                  {pageCompleted ? (
                    <div className="text-sm font-bold text-green-300">
                      ✓ Nội dung cuối đã hoàn thành
                    </div>
                  ) : (
                    <>
                      <div className="text-sm font-bold text-yellow-300">
                        Chưa đủ điều kiện hoàn thành
                      </div>

                      <div
                        className="mt-1 text-xs"
                        style={{
                          color:
                            themeMutedText,
                        }}
                      >
                        Cần xem ≥{" "}
                        {completionSettings.minWatchPercent}%
                        {completionSettings.requireAllQuizzes
                          ? " và hoàn thành tất cả câu hỏi"
                          : ""}
                      </div>
                    </>
                  )}

                </div>

                <button
                  type="button"
                  onClick={
                    nextPage
                  }
                  disabled={
                    !pageCompleted ||
                    activeQuiz !==
                      null
                  }
                  className="flex w-full items-center justify-center gap-2 px-6 py-3 font-black text-white transition hover:brightness-110 disabled:cursor-not-allowed disabled:opacity-40 sm:w-auto"
                  style={{
                    background:
                      `linear-gradient(90deg, ${projectTheme.primaryColor}, ${projectTheme.accentColor})`,
                    borderRadius:
                      `${themeButtonRadius}px`,
                  }}
                >
                  Hoàn thành bài học

                  <ChevronRight
                    size={18}
                  />
                </button>

              </div>

            </section>
          )}

          </div>

        </div>

      </main>

    </div>
  );
}
