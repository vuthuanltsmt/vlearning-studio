"use client";

import {
  Check,
  Palette,
  RotateCcw,
  X,
} from "lucide-react";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  createDefaultProjectTheme,
  loadProjectThemeForProject,
  resetProjectThemeForProject,
  saveProjectThemeForProject,
} from "@/lib/projectTheme";

import type {
  ProjectTheme,
  ThemeButtonStyle,
  ThemeFont,
  ThemeMode,
} from "@/types/theme";

type Props = {
  open: boolean;
  projectId: string;
  projectTitle: string;
  onClose: () => void;
};

const HEX_COLOR =
  /^#[0-9a-fA-F]{6}$/;

/* =========================================================
   HELPERS
========================================================= */

function getFontFamily(
  font: ThemeFont
) {
  switch (font) {
    case "serif":
      return "Georgia, 'Times New Roman', serif";

    case "sans":
      return "Arial, Helvetica, sans-serif";

    case "system":
    default:
      return "system-ui, -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif";
  }
}

function getButtonRadius(
  style: ThemeButtonStyle
) {
  switch (style) {
    case "pill":
      return 999;

    case "square":
      return 4;

    case "rounded":
    default:
      return 12;
  }
}

function safeColor(
  value: string,
  fallback: string
) {
  return HEX_COLOR.test(value)
    ? value
    : fallback;
}

/* =========================================================
   COMPONENT
========================================================= */

export function DesignModal({
  open,
  projectId,
  projectTitle,
  onClose,
}: Props) {
  const [
    draft,
    setDraft,
  ] =
    useState<ProjectTheme>(
      createDefaultProjectTheme(
        projectTitle
      )
    );

  const [
    savedMessage,
    setSavedMessage,
  ] =
    useState("");

  /* =====================================================
     LOAD
  ===================================================== */

  useEffect(() => {
    if (!open) {
      return;
    }

    setDraft(
      loadProjectThemeForProject(
        projectId,
        projectTitle
      )
    );

    setSavedMessage("");
  }, [
    open,
    projectId,
    projectTitle,
  ]);

  /* =====================================================
     ESC
  ===================================================== */

  useEffect(() => {
    if (!open) {
      return;
    }

    function handleKeyDown(
      event: KeyboardEvent
    ) {
      if (
        event.key ===
        "Escape"
      ) {
        onClose();
      }
    }

    window.addEventListener(
      "keydown",
      handleKeyDown
    );

    return () => {
      window.removeEventListener(
        "keydown",
        handleKeyDown
      );
    };
  }, [
    open,
    onClose,
  ]);

  /* =====================================================
     PREVIEW STYLE
  ===================================================== */

  const previewStyle =
    useMemo(
      () => ({
        fontFamily:
          getFontFamily(
            draft.fontFamily
          ),
        backgroundColor:
          safeColor(
            draft.backgroundColor,
            "#07182c"
          ),
        color:
          safeColor(
            draft.textColor,
            "#ffffff"
          ),
      }),
      [
        draft.backgroundColor,
        draft.fontFamily,
        draft.textColor,
      ]
    );

  const surfaceColor =
    safeColor(
      draft.surfaceColor,
      "#102b52"
    );

  const primaryColor =
    safeColor(
      draft.primaryColor,
      "#2563eb"
    );

  const accentColor =
    safeColor(
      draft.accentColor,
      "#8b5cf6"
    );

  /* =====================================================
     ACTIONS
  ===================================================== */

  function updateDraft<K extends keyof ProjectTheme>(
    key: K,
    value: ProjectTheme[K]
  ) {
    setDraft(
      (current) => ({
        ...current,
        [key]: value,
      })
    );

    setSavedMessage("");
  }

  function handleSave() {
    const saved =
      saveProjectThemeForProject(
        projectId,
        draft,
        projectTitle
      );

    setDraft(saved);
    setSavedMessage(
      "Đã lưu thiết kế"
    );
  }

  function handleReset() {
    const confirmed =
      window.confirm(
        "Đưa thiết kế của dự án này về mặc định?"
      );

    if (!confirmed) {
      return;
    }

    const defaults =
      resetProjectThemeForProject(
        projectId,
        projectTitle
      );

    setDraft(defaults);
    setSavedMessage(
      "Đã khôi phục thiết kế mặc định"
    );
  }

  if (!open) {
    return null;
  }

  /* =====================================================
     UI
  ===================================================== */

  return (
    <div className="fixed inset-0 z-[110] flex items-center justify-center bg-black/75 p-5 backdrop-blur-sm">

      <div className="flex max-h-[90vh] w-full max-w-6xl flex-col overflow-hidden rounded-3xl border border-blue-800 bg-[#0b213f] text-white shadow-2xl">

        {/* HEADER */}

        <div className="flex items-center justify-between gap-4 border-b border-blue-800 bg-[#123665] px-6 py-5">

          <div className="flex items-center gap-3">

            <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-cyan-950 text-cyan-300">
              <Palette
                size={22}
              />
            </div>

            <div>
              <h2 className="text-xl font-black">
                Thiết kế bài giảng
              </h2>

              <p className="mt-1 text-xs text-slate-400">
                Thiết lập giao diện riêng cho từng dự án
              </p>
            </div>

          </div>

          <button
            type="button"
            onClick={
              onClose
            }
            className="rounded-xl p-2 text-slate-300 transition hover:bg-blue-900 hover:text-white"
            title="Đóng"
          >
            <X
              size={21}
            />
          </button>

        </div>

        {/* CONTENT */}

        <div className="grid min-h-0 flex-1 overflow-y-auto lg:grid-cols-[420px_1fr]">

          {/* SETTINGS */}

          <div className="border-b border-blue-800 p-6 lg:border-b-0 lg:border-r">

            <div className="space-y-6">

              <label className="block">
                <span className="text-sm font-bold text-slate-200">
                  Tiêu đề hiển thị
                </span>

                <input
                  type="text"
                  value={
                    draft.courseTitle
                  }
                  onChange={(
                    event
                  ) =>
                    updateDraft(
                      "courseTitle",
                      event.target.value
                    )
                  }
                  placeholder={
                    projectTitle
                  }
                  className="mt-2 w-full rounded-xl border border-blue-700 bg-blue-950/70 px-4 py-3 text-white outline-none transition placeholder:text-slate-500 focus:border-cyan-500"
                />
              </label>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Chế độ giao diện
                </div>

                <div className="mt-2 grid grid-cols-2 gap-2">
                  {([
                    [
                      "dark",
                      "Tối",
                    ],
                    [
                      "light",
                      "Sáng",
                    ],
                  ] as Array<[
                    ThemeMode,
                    string,
                  ]>).map(([
                    value,
                    label,
                  ]) => (
                    <button
                      key={
                        value
                      }
                      type="button"
                      onClick={() =>
                        updateDraft(
                          "mode",
                          value
                        )
                      }
                      className={`rounded-xl border px-4 py-3 text-sm font-bold transition ${
                        draft.mode ===
                        value
                          ? "border-cyan-400 bg-cyan-950 text-cyan-200"
                          : "border-blue-800 bg-blue-950/40 text-slate-300 hover:bg-blue-900"
                      }`}
                    >
                      {label}
                    </button>
                  ))}
                </div>
              </div>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Màu sắc
                </div>

                <div className="mt-3 grid grid-cols-2 gap-3">

                  {([
                    [
                      "primaryColor",
                      "Màu chính",
                    ],
                    [
                      "accentColor",
                      "Màu nhấn",
                    ],
                    [
                      "backgroundColor",
                      "Màu nền",
                    ],
                    [
                      "surfaceColor",
                      "Màu khối",
                    ],
                    [
                      "textColor",
                      "Màu chữ",
                    ],
                  ] as Array<[
                    | "primaryColor"
                    | "accentColor"
                    | "backgroundColor"
                    | "surfaceColor"
                    | "textColor",
                    string,
                  ]>).map(([
                    key,
                    label,
                  ]) => (
                    <label
                      key={
                        key
                      }
                      className="rounded-xl border border-blue-800 bg-blue-950/40 p-3"
                    >
                      <span className="text-xs font-bold text-slate-300">
                        {label}
                      </span>

                      <div className="mt-2 flex items-center gap-2">
                        <input
                          type="color"
                          value={
                            safeColor(
                              draft[key],
                              "#2563eb"
                            )
                          }
                          onChange={(
                            event
                          ) =>
                            updateDraft(
                              key,
                              event.target.value
                            )
                          }
                          className="h-10 w-12 cursor-pointer rounded-lg border-0 bg-transparent p-0"
                        />

                        <input
                          type="text"
                          value={
                            draft[key]
                          }
                          onChange={(
                            event
                          ) =>
                            updateDraft(
                              key,
                              event.target.value
                            )
                          }
                          className="min-w-0 flex-1 rounded-lg border border-blue-800 bg-[#07182c] px-2 py-2 font-mono text-xs text-slate-200 outline-none focus:border-cyan-500"
                        />
                      </div>
                    </label>
                  ))}

                </div>
              </div>

              <label className="block">
                <span className="text-sm font-bold text-slate-200">
                  Font chữ
                </span>

                <select
                  value={
                    draft.fontFamily
                  }
                  onChange={(
                    event
                  ) =>
                    updateDraft(
                      "fontFamily",
                      event.target.value as ThemeFont
                    )
                  }
                  className="mt-2 w-full rounded-xl border border-blue-800 bg-blue-950/70 px-4 py-3 text-white outline-none focus:border-cyan-500"
                >
                  <option value="system">
                    System UI
                  </option>
                  <option value="sans">
                    Sans Serif
                  </option>
                  <option value="serif">
                    Serif
                  </option>
                </select>
              </label>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Kiểu nút
                </div>

                <div className="mt-2 grid grid-cols-3 gap-2">
                  {([
                    [
                      "rounded",
                      "Bo góc",
                    ],
                    [
                      "pill",
                      "Viên thuốc",
                    ],
                    [
                      "square",
                      "Vuông",
                    ],
                  ] as Array<[
                    ThemeButtonStyle,
                    string,
                  ]>).map(([
                    value,
                    label,
                  ]) => (
                    <button
                      key={
                        value
                      }
                      type="button"
                      onClick={() =>
                        updateDraft(
                          "buttonStyle",
                          value
                        )
                      }
                      className={`border px-3 py-2.5 text-xs font-bold transition ${
                        value ===
                        "pill"
                          ? "rounded-full"
                          : value ===
                            "square"
                            ? "rounded"
                            : "rounded-xl"
                      } ${
                        draft.buttonStyle ===
                        value
                          ? "border-cyan-400 bg-cyan-950 text-cyan-200"
                          : "border-blue-800 bg-blue-950/40 text-slate-300 hover:bg-blue-900"
                      }`}
                    >
                      {label}
                    </button>
                  ))}
                </div>
              </div>

              <label className="block">
                <div className="flex items-center justify-between gap-4">
                  <span className="text-sm font-bold text-slate-200">
                    Bo góc Player
                  </span>

                  <span className="text-xs text-cyan-300">
                    {draft.playerRadius}px
                  </span>
                </div>

                <input
                  type="range"
                  min="0"
                  max="32"
                  step="2"
                  value={
                    draft.playerRadius
                  }
                  onChange={(
                    event
                  ) =>
                    updateDraft(
                      "playerRadius",
                      Number(
                        event.target.value
                      )
                    )
                  }
                  className="mt-3 w-full"
                />
              </label>

              <div className="space-y-3">

                <label className="flex cursor-pointer items-center justify-between gap-4 rounded-xl border border-blue-800 bg-blue-950/40 px-4 py-3">
                  <div>
                    <div className="text-sm font-bold">
                      Thanh tiến độ
                    </div>
                    <div className="mt-1 text-xs text-slate-400">
                      Hiển thị tiến độ học trong Player
                    </div>
                  </div>

                  <input
                    type="checkbox"
                    checked={
                      draft.showProgress
                    }
                    onChange={(
                      event
                    ) =>
                      updateDraft(
                        "showProgress",
                        event.target.checked
                      )
                    }
                    className="h-5 w-5"
                  />
                </label>

                <label className="flex cursor-pointer items-center justify-between gap-4 rounded-xl border border-blue-800 bg-blue-950/40 px-4 py-3">
                  <div>
                    <div className="text-sm font-bold">
                      Điều hướng trang
                    </div>
                    <div className="mt-1 text-xs text-slate-400">
                      Hiển thị nút Trước / Tiếp theo
                    </div>
                  </div>

                  <input
                    type="checkbox"
                    checked={
                      draft.showPageNavigation
                    }
                    onChange={(
                      event
                    ) =>
                      updateDraft(
                        "showPageNavigation",
                        event.target.checked
                      )
                    }
                    className="h-5 w-5"
                  />
                </label>

              </div>

            </div>

          </div>

          {/* PREVIEW */}

          <div className="p-6">

            <div className="mb-4">
              <div className="text-xs font-black uppercase tracking-[0.16em] text-cyan-300">
                Xem trước thiết kế
              </div>

              <div className="mt-1 text-sm text-slate-400">
                Đây là bản mô phỏng. V11.0B sẽ áp dụng theme vào Preview thật.
              </div>
            </div>

            <div
              className="overflow-hidden border border-white/10 shadow-2xl"
              style={{
                ...previewStyle,
                borderRadius:
                  `${draft.playerRadius}px`,
              }}
            >

              <div
                className="flex items-center justify-between gap-3 border-b border-white/10 px-5 py-4"
                style={{
                  backgroundColor:
                    surfaceColor,
                }}
              >
                <div>
                  <div
                    className="text-xs font-black uppercase tracking-wider"
                    style={{
                      color:
                        accentColor,
                    }}
                  >
                    VLearning Studio
                  </div>

                  <div className="mt-1 font-black">
                    {draft.courseTitle.trim() ||
                      projectTitle}
                  </div>
                </div>

                <div
                  className="rounded-lg px-3 py-1.5 text-xs font-bold text-white"
                  style={{
                    backgroundColor:
                      primaryColor,
                  }}
                >
                  1 / 3
                </div>
              </div>

              <div className="p-5">

                <div
                  className="flex aspect-video items-center justify-center border border-white/10 text-sm"
                  style={{
                    backgroundColor:
                      surfaceColor,
                    borderRadius:
                      `${draft.playerRadius}px`,
                  }}
                >
                  Video bài giảng
                </div>

                {draft.showProgress && (
                  <div className="mt-5">
                    <div className="mb-2 flex justify-between text-xs opacity-70">
                      <span>
                        Tiến độ
                      </span>
                      <span>
                        65%
                      </span>
                    </div>

                    <div className="h-2 overflow-hidden rounded-full bg-black/25">
                      <div
                        className="h-full w-[65%] rounded-full"
                        style={{
                          backgroundColor:
                            accentColor,
                        }}
                      />
                    </div>
                  </div>
                )}

                <div
                  className="mt-5 rounded-xl p-4"
                  style={{
                    backgroundColor:
                      surfaceColor,
                  }}
                >
                  <div className="text-sm font-black">
                    Câu hỏi tương tác
                  </div>

                  <div className="mt-2 text-sm opacity-80">
                    Nội dung câu hỏi sẽ hiển thị tại đây.
                  </div>
                </div>

                {draft.showPageNavigation && (
                  <div className="mt-5 flex justify-end gap-2">
                    <button
                      type="button"
                      className="border border-white/15 px-4 py-2 text-sm font-bold"
                      style={{
                        borderRadius:
                          `${getButtonRadius(
                            draft.buttonStyle
                          )}px`,
                      }}
                    >
                      Trước
                    </button>

                    <button
                      type="button"
                      className="px-4 py-2 text-sm font-bold text-white"
                      style={{
                        backgroundColor:
                          primaryColor,
                        borderRadius:
                          `${getButtonRadius(
                            draft.buttonStyle
                          )}px`,
                      }}
                    >
                      Tiếp theo
                    </button>
                  </div>
                )}

              </div>

            </div>

          </div>

        </div>

        {/* FOOTER */}

        <div className="flex flex-wrap items-center justify-between gap-3 border-t border-blue-800 bg-[#0d294e] px-6 py-4">

          <div className="min-h-5 text-sm font-bold text-emerald-300">
            {savedMessage && (
              <span className="inline-flex items-center gap-2">
                <Check
                  size={16}
                />
                {savedMessage}
              </span>
            )}
          </div>

          <div className="flex flex-wrap gap-2">

            <button
              type="button"
              onClick={
                handleReset
              }
              className="flex items-center gap-2 rounded-xl border border-slate-600 px-4 py-2.5 text-sm font-bold text-slate-200 transition hover:bg-slate-800"
            >
              <RotateCcw
                size={16}
              />
              Mặc định
            </button>

            <button
              type="button"
              onClick={
                onClose
              }
              className="rounded-xl border border-blue-700 px-5 py-2.5 text-sm font-bold transition hover:bg-blue-900"
            >
              Đóng
            </button>

            <button
              type="button"
              onClick={
                handleSave
              }
              className="flex items-center gap-2 rounded-xl bg-gradient-to-r from-cyan-500 to-purple-600 px-5 py-2.5 text-sm font-black transition hover:brightness-110"
            >
              <Check
                size={17}
              />
              Lưu thiết kế
            </button>

          </div>

        </div>

      </div>

    </div>
  );
}
