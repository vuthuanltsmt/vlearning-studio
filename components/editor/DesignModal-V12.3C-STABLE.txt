import {
  Award,
  Check,
  CopyPlus,
  ImagePlus,
  Palette,
  RotateCcw,
  Trash2,
  X,
} from "lucide-react";

import {
  useEffect,
  useMemo,
  useState,
} from "react";

import {
  createDefaultProjectTheme,
  loadProjectThemeForProject,
  resetProjectThemeForProject,
  saveProjectThemeForProject,
} from "@/lib/projectTheme";

import {
  applyBuiltInCertificateTemplate,
  applyCertificateTemplateData,
  BUILT_IN_CERTIFICATE_TEMPLATES,
  deleteSavedCertificateTemplate,
  listSavedCertificateTemplates,
  saveCertificateTemplate,
} from "@/lib/certificateTemplates";

import type {
  SavedCertificateTemplate,
} from "@/lib/certificateTemplates";

import type {
  ProjectTheme,
  ThemeButtonStyle,
  ThemeFont,
  ThemeMode,
} from "@/types/theme";

type Props = {
  open: boolean;
  projectId: string;
  projectTitle: string;
  onClose: () => void;
};

const HEX_COLOR =
  /^#[0-9a-fA-F]{6}$/;

/* =========================================================
   HELPERS
========================================================= */

function getFontFamily(
  font: ThemeFont
) {
  switch (font) {
    case "serif":
      return "Georgia, 'Times New Roman', serif";

    case "sans":
      return "Arial, Helvetica, sans-serif";

    case "system":
    default:
      return "system-ui, -apple-system, BlinkMacSystemFont, 'Segoe UI', sans-serif";
  }
}

function getButtonRadius(
  style: ThemeButtonStyle
) {
  switch (style) {
    case "pill":
      return 999;

    case "square":
      return 4;

    case "rounded":
    default:
      return 12;
  }
}

function safeColor(
  value: string,
  fallback: string
) {
  return HEX_COLOR.test(value)
    ? value
    : fallback;
}

/* =========================================================
   COMPONENT
========================================================= */

export function DesignModal({
  open,
  projectId,
  projectTitle,
  onClose,
}: Props) {
  const [
    draft,
    setDraft,
  ] =
    useState<ProjectTheme>(
      createDefaultProjectTheme(
        projectTitle
      )
    );

  const [
    savedMessage,
    setSavedMessage,
  ] =
    useState("");

  const [
    previewMode,
    setPreviewMode,
  ] =
    useState<
      "player" |
      "certificate"
    >(
      "player"
    );

  const [
    customCertificateTemplates,
    setCustomCertificateTemplates,
  ] =
    useState<
      SavedCertificateTemplate[]
    >([]);

  const [
    newCertificateTemplateName,
    setNewCertificateTemplateName,
  ] =
    useState("");

  /* =====================================================
     LOAD
  ===================================================== */

  useEffect(() => {
    if (!open) {
      return;
    }

    setDraft(
      loadProjectThemeForProject(
        projectId,
        projectTitle
      )
    );

    setSavedMessage("");
    setPreviewMode(
      "player"
    );

    setCustomCertificateTemplates(
      listSavedCertificateTemplates()
    );

    setNewCertificateTemplateName(
      ""
    );
  }, [
    open,
    projectId,
    projectTitle,
  ]);

  /* =====================================================
     ESC
  ===================================================== */

  useEffect(() => {
    if (!open) {
      return;
    }

    function handleKeyDown(
      event: KeyboardEvent
    ) {
      if (
        event.key ===
        "Escape"
      ) {
        onClose();
      }
    }

    window.addEventListener(
      "keydown",
      handleKeyDown
    );

    return () => {
      window.removeEventListener(
        "keydown",
        handleKeyDown
      );
    };
  }, [
    open,
    onClose,
  ]);

  /* =====================================================
     PREVIEW STYLE
  ===================================================== */

  const previewStyle =
    useMemo(
      () => ({
        fontFamily:
          getFontFamily(
            draft.fontFamily
          ),
        backgroundColor:
          safeColor(
            draft.backgroundColor,
            "#07182c"
          ),
        color:
          safeColor(
            draft.textColor,
            "#ffffff"
          ),
      }),
      [
        draft.backgroundColor,
        draft.fontFamily,
        draft.textColor,
      ]
    );

  const surfaceColor =
    safeColor(
      draft.surfaceColor,
      "#102b52"
    );

  const primaryColor =
    safeColor(
      draft.primaryColor,
      "#2563eb"
    );

  const accentColor =
    safeColor(
      draft.accentColor,
      "#8b5cf6"
    );

  const certificatePrimaryColor =
    safeColor(
      draft.certificatePrimaryColor ||
        draft.primaryColor,
      "#2563eb"
    );

  const certificateAccentColor =
    safeColor(
      draft.certificateAccentColor ||
        draft.accentColor,
      "#8b5cf6"
    );

  const certificatePreviewBackground =
    draft.certificateTemplateStyle ===
      "formal"
      ? "#fffbeb"
      : draft.certificateTemplateStyle ===
          "school"
        ? "#f0fdf4"
        : "#ffffff";

  const certificatePreviewCorporate =
    draft.certificateTemplateStyle ===
    "corporate";

  /* =====================================================
     ACTIONS
  ===================================================== */

  function updateDraft<K extends keyof ProjectTheme>(
    key: K,
    value: ProjectTheme[K]
  ) {
    setDraft(
      (current) => ({
        ...current,
        [key]: value,
      })
    );

    setSavedMessage("");
  }

  function applyBuiltInTemplate(
    style:
      ProjectTheme[
        "certificateTemplateStyle"
      ]
  ) {
    setDraft(
      (current) =>
        applyBuiltInCertificateTemplate(
          current,
          style
        )
    );

    setPreviewMode(
      "certificate"
    );

    setSavedMessage("");
  }

  function saveCurrentCertificateTemplate() {
    try {
      const next =
        saveCertificateTemplate(
          newCertificateTemplateName,
          draft
        );

      setCustomCertificateTemplates(
        next
      );

      setNewCertificateTemplateName(
        ""
      );

      setSavedMessage(
        "Đã lưu mẫu chứng nhận cá nhân"
      );
    } catch (error) {
      window.alert(
        error instanceof Error
          ? error.message
          : "Không lưu được mẫu chứng nhận."
      );
    }
  }

  function applySavedCertificateTemplate(
    template:
      SavedCertificateTemplate
  ) {
    setDraft(
      (current) =>
        applyCertificateTemplateData(
          current,
          template.data
        )
    );

    setPreviewMode(
      "certificate"
    );

    setSavedMessage("");
  }

  function removeSavedCertificateTemplate(
    template:
      SavedCertificateTemplate
  ) {
    if (
      !window.confirm(
        `Xóa mẫu "${template.name}"?`
      )
    ) {
      return;
    }

    setCustomCertificateTemplates(
      deleteSavedCertificateTemplate(
        template.id
      )
    );
  }

  function readCertificateImage(
    file: File | undefined,
    key:
      | "certificateLogoDataUrl"
      | "certificateSignatureDataUrl",
    label: string,
    maxBytes: number
  ) {
    if (!file) {
      return;
    }

    if (
      !file.type.startsWith(
        "image/"
      )
    ) {
      window.alert(
        `${label} phải là tệp ảnh.`
      );

      return;
    }

    if (
      file.size >
      maxBytes
    ) {
      window.alert(
        `${label} quá lớn. Vui lòng dùng ảnh nhỏ hơn ${Math.round(
          maxBytes /
            1024
        )} KB.`
      );

      return;
    }

    const reader =
      new FileReader();

    reader.onload =
      () => {
        const result =
          typeof reader.result ===
            "string"
            ? reader.result
            : "";

        updateDraft(
          key,
          result
        );
      };

    reader.onerror =
      () => {
        window.alert(
          `Không đọc được ${label.toLowerCase()}.`
        );
      };

    reader.readAsDataURL(
      file
    );
  }

  function handleSave() {
    try {
      const saved =
        saveProjectThemeForProject(
          projectId,
          draft,
          projectTitle
        );

      setDraft(saved);
      setSavedMessage(
        "Đã lưu thiết kế"
      );
    } catch (
      error
    ) {
      console.error(
        "Không lưu được thiết kế:",
        error
      );

      window.alert(
        "Không lưu được thiết kế. Nếu bạn vừa tải logo/chữ ký, hãy dùng ảnh có dung lượng nhỏ hơn."
      );
    }
  }

  function handleReset() {
    const confirmed =
      window.confirm(
        "Đưa thiết kế của dự án này về mặc định?"
      );

    if (!confirmed) {
      return;
    }

    const defaults =
      resetProjectThemeForProject(
        projectId,
        projectTitle
      );

    setDraft(defaults);
    setSavedMessage(
      "Đã khôi phục thiết kế mặc định"
    );
  }

  if (!open) {
    return null;
  }

     UI
  ===================================================== */

  return (
    <div className="fixed inset-0 z-[110] flex items-center justify-center bg-black/75 p-5 backdrop-blur-sm">

      <div className="flex max-h-[90vh] w-full max-w-7xl flex-col overflow-hidden rounded-3xl border border-blue-800 bg-[#0b213f] text-white shadow-2xl">

        {/* HEADER */}

        <div className="flex items-center justify-between gap-4 border-b border-blue-800 bg-[#123665] px-6 py-5">

          <div className="flex items-center gap-3">

            <div className="flex h-11 w-11 items-center justify-center rounded-xl bg-cyan-950 text-cyan-300">
              <Palette
                size={22}
              />
            </div>

            <div>
              <h2 className="text-xl font-black">
                Thiết kế bài giảng
              </h2>

              <p className="mt-1 text-xs text-slate-400">
                Thiết lập giao diện riêng cho từng dự án
              </p>
            </div>

          </div>

          <button
            type="button"
            onClick={
              onClose
            }
            className="rounded-xl p-2 text-slate-300 transition hover:bg-blue-900 hover:text-white"
            title="Đóng"
          >
            <X
              size={21}
            />
          </button>

        </div>

        {/* CONTENT */}

        <div className="grid min-h-0 flex-1 overflow-hidden lg:grid-cols-[420px_1fr]">

          {/* SETTINGS */}

          <div className="min-h-0 overflow-y-auto border-b border-blue-800 p-6 lg:border-b-0 lg:border-r">

            <div className="space-y-6">

              <label className="block">
                <span className="text-sm font-bold text-slate-200">
                  Tiêu đề hiển thị
                </span>

                <input
                  type="text"
                  value={
                    draft.courseTitle
                  }
                  onChange={(
                    event
                  ) =>
                    updateDraft(
                      "courseTitle",
                      event.target.value
                    )
                  }
                  placeholder={
                    projectTitle
                  }
                  className="mt-2 w-full rounded-xl border border-blue-700 bg-blue-950/70 px-4 py-3 text-white outline-none transition placeholder:text-slate-500 focus:border-cyan-500"
                />
              </label>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Chế độ giao diện
                </div>

                <div className="mt-2 grid grid-cols-2 gap-2">
                  {([
                    [
                      "dark",
                      "Tối",
                    ],
                    [
                      "light",
                      "Sáng",
                    ],
                  ] as Array<[
                    ThemeMode,
                    string,
                  ]>).map(([
                    value,
                    label,
                  ]) => (
                    <button
                      key={
                        value
                      }
                      type="button"
                      onClick={() =>
                        updateDraft(
                          "mode",
                          value
                        )
                      }
                      className={`rounded-xl border px-4 py-3 text-sm font-bold transition ${
                        draft.mode ===
                        value
                          ? "border-cyan-400 bg-cyan-950 text-cyan-200"
                          : "border-blue-800 bg-blue-950/40 text-slate-300 hover:bg-blue-900"
                      }`}
                    >
                      {label}
                    </button>
                  ))}
                </div>
              </div>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Hiển thị trong Preview
                </div>

                <div className="mt-3 space-y-3">
                  <label className="flex cursor-pointer items-center justify-between gap-4 rounded-xl border border-blue-700 bg-blue-950/60 px-4 py-3">
                    <div>
                      <div className="text-sm font-bold text-white">
                        Thanh tiến độ
                      </div>
                      <div className="mt-1 text-xs text-slate-400">
                        Hiển thị tiến độ học trong Player
                      </div>
                    </div>

                    <input
                      type="checkbox"
                      checked={draft.showProgress}
                      onChange={(event) =>
                        updateDraft(
                          "showProgress",
                          event.target.checked
                        )
                      }
                      className="h-5 w-5 accent-cyan-500"
                    />
                  </label>

                  <label className="flex cursor-pointer items-center justify-between gap-4 rounded-xl border border-blue-700 bg-blue-950/60 px-4 py-3">
                    <div>
                      <div className="text-sm font-bold text-white">
                        Điều hướng trang
                      </div>
                      <div className="mt-1 text-xs text-slate-400">
                        Hiển thị nút Trang trước / Trang tiếp theo
                      </div>
                    </div>

                    <input
                      type="checkbox"
                      checked={draft.showPageNavigation}
                      onChange={(event) =>
                        updateDraft(
                          "showPageNavigation",
                          event.target.checked
                        )
                      }
                      className="h-5 w-5 accent-cyan-500"
                    />
                  </label>
                </div>
              </div>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Màu sắc
                </div>

                <div className="mt-3 grid grid-cols-2 gap-3">

                  {([
                    [
                      "primaryColor",
                      "Màu chính",
                    ],
                    [
                      "accentColor",
                      "Màu nhấn",
                    ],
                    [
                      "backgroundColor",
                      "Màu nền",
                    ],
                    [
                      "surfaceColor",
                      "Màu khối",
                    ],
                    [
                      "textColor",
                      "Màu chữ",
                    ],
                  ] as Array<[
                    | "primaryColor"
                    | "accentColor"
                    | "backgroundColor"
                    | "surfaceColor"
                    | "textColor",
                    string,
                  ]>).map(([
                    key,
                    label,
                  ]) => (
                    <label
                      key={
                        key
                      }
                      className="rounded-xl border border-blue-800 bg-blue-950/40 p-3"
                    >
                      <span className="text-xs font-bold text-slate-300">
                        {label}
                      </span>

                      <div className="mt-2 flex items-center gap-2">
                        <input
                          type="color"
                          value={
                            safeColor(
                              draft[key],
                              "#2563eb"
                            )
                          }
                          onChange={(
                            event
                          ) =>
                            updateDraft(
                              key,
                              event.target.value
                            )
                          }
                          className="h-10 w-12 cursor-pointer rounded-lg border-0 bg-transparent p-0"
                        />

                        <input
                          type="text"
                          value={
                            draft[key]
                          }
                          onChange={(
                            event
                          ) =>
                            updateDraft(
                              key,
                              event.target.value
                            )
                          }
                          className="min-w-0 flex-1 rounded-lg border border-blue-800 bg-[#07182c] px-2 py-2 font-mono text-xs text-slate-200 outline-none focus:border-cyan-500"
                        />
                      </div>
                    </label>
                  ))}

                </div>
              </div>

              <label className="block">
                <span className="text-sm font-bold text-slate-200">
                  Font chữ
                </span>

                <select
                  value={
                    draft.fontFamily
                  }
                  onChange={(
                    event
                  ) =>
                    updateDraft(
                      "fontFamily",
                      event.target.value as ThemeFont
                    )
                  }
                  className="mt-2 w-full rounded-xl border border-blue-800 bg-blue-950/70 px-4 py-3 text-white outline-none focus:border-cyan-500"
                >
                  <option value="system">
                    System UI
                  </option>
                  <option value="sans">
                    Sans Serif
                  </option>
                  <option value="serif">
                    Serif
                  </option>
                </select>
              </label>

              <div>
                <div className="text-sm font-bold text-slate-200">
                  Kiểu nút
                </div>

                <div className="mt-2 grid grid-cols-3 gap-2">
                  {([
                    [
                      "rounded",
                      "Bo góc",
                    ],
                    [
                      "pill",
                      "Viên thuốc",
                    ],
                    [
                      "square",
                      "Vuông",
                    ],
                  ] as Array<[
                    ThemeButtonStyle,
                    string,
                  ]>).map(([
                    value,
                    label,
                  ]) => (
                    <button
                      key={
                        value
                      }
                      type="button"
                      onClick={() =>
                        updateDraft(
                          "buttonStyle",
                          value
                        )
                      }
                      className={`border px-3 py-2.5 text-xs font-bold transition ${
                        value ===
                        "pill"
                          ? "rounded-full"
                          : value ===
                            "square"

                        draft.buttonStyle ===

                          : "border-blue-800 bg-blue-950/40 text-slate-300 hover:bg-blue-900"

                    >
                      {label}

                  ))}

              </div>

              <label className="block">
                <div className="flex items-center justify-between gap-4">
                  <span className="text-sm font-bold text-slate-200">
                    Bo góc Player
                  </span>

                  <span className="text-xs text-cyan-300">
                    {draft.playerRadius}px
                  </span>
                </div>

                <input
                  type="range"
                  min="0"
                  max="32"
                  step="2"

                    draft.playerRadius
                  }
                  onChange={(
                    event

                    updateDraft(
                      "playerRadius",
                      Number(
                        event.target.value
                      )

                  }
                  className="mt-3 w-full"
                />
              </label>

              {/* V12.3B CERTIFICATE DESIGN */}

              <div className="border-t border-blue-800 pt-6">

                <div className="flex items-center gap-2">
                  <Award
                    size={19}
                    className="text-amber-300"
                  />

                  <div>
                    <div className="text-sm font-black text-white">
                      Chứng nhận hoàn thành
                    </div>

                      Tùy chỉnh mẫu chứng nhận cho Preview và SCORM.
                    </div>
                  </div>
                </div>

                <div className="mt-4 space-y-4">

                  {/* V12.3C TEMPLATE LIBRARY */}

                  <div className="rounded-2xl border border-blue-800 bg-blue-950/35 p-4">

                    <div className="flex items-center gap-2">
                      <CopyPlus
                        size={16}
                        className="text-cyan-300"
                      />

                      <div className="text-xs font-black uppercase tracking-[0.12em] text-cyan-200">
                        Thư viện mẫu
                      </div>
                    </div>

                    <div className="mt-3 grid grid-cols-2 gap-2">

                        (template) => {
                          const active =
                            draft.certificateTemplateStyle ===
                            template.id;

                          return (
                            <button
                              key={
                                template.id
                              }
                              type="button"
                              onClick={() =>
                                applyBuiltInTemplate(
                                  template.id
                                )
                              }
                              className={`rounded-xl border p-3 text-left transition ${
                                active
                                  ? "border-cyan-400 bg-cyan-950/60"
                                  : "border-blue-800 bg-[#07182c]/60 hover:border-blue-600"
                              }`}
                            >
                              <div className="flex items-center gap-2">
                                <span
                                  className="h-3 w-3 rounded-full"
                                  style={{
                                    backgroundColor:
                                      template.primaryColor,
                                    boxShadow:
                                      `0 0 0 3px ${template.accentColor}55`,
                                  }}
                                />

                                <span className="text-xs font-black text-white">
                                  {template.name}
                                </span>
                              </div>

                              <div className="mt-2 text-[10px] leading-4 text-slate-400">
                                {template.description}
                              </div>
                            </button>
                          );

                    <div className="mt-4 border-t border-blue-800 pt-4">
                      <div className="text-xs font-black text-slate-200">
                        Mẫu cá nhân
                      </div>

                      <div className="mt-2 flex gap-2">

                          type="text"
                          value={
                            newCertificateTemplateName
                          }
                          onChange={(
                            event
                          ) =>
                            setNewCertificateTemplateName(
                              event.target.value

                          type="button"
                          onClick={
                            saveCurrentCertificateTemplate
                          }
                          className="rounded-xl bg-cyan-700 px-3 py-2 text-xs font-black text-white hover:bg-cyan-600"

                          Lưu mẫu

                      0 ? (

                    <span className="text-xs font-bold text-slate-300">

                      type="text"
                      value={
                        draft.certificateKicker

                        event

                        updateDraft(
                          "certificateKicker",
